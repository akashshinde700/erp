   100→function isTypingTarget(t: EventTarget | null): boolean {
   101→  if (!(t instanceof HTMLElement)) return false

   105→function readJsonArray(key: string): string[] {
   106→  try {
   107→    const raw = window.localStorage.getItem(key)
   108→    const parsed = raw ? JSON.parse(raw) : []
   109→    return Array.isArray(parsed) ? parsed.filter((k) => typeof k === 'string') : []
   110→  } catch { return [] }
   111→}
   112→
   113→function persistJson(key: string, value: string[]) {
   114→  try { window.localStorage.setItem(key, JSON.stringify(value)) } catch { /* private mode */ }
   115→}

   286→  function goToModule(key: string, preset?: { q?: string }) {
   287→    // Filter preset: sessionStorage covers not-yet-mounted modules, the live event covers the already-mounted one
   288→    if (preset && typeof preset.q === 'string' && preset.q.length > 0) {
   289→      try {
   290→        window.sessionStorage.setItem('erp_filter_preset', JSON.stringify({ module: key, q: preset.q, ts: Date.now() }))
   291→      } catch { /* private mode */ }
   292→      window.dispatchEvent(new CustomEvent('erp:filter-preset', { detail: { module: key, q: preset.q } }))
   293→    }
   294→    setActiveKey(key)
   295→    setPaletteOpen(false)
   296→    setPaletteQuery('') // programmatic close bypasses onOpenChange — clear stale query/results
   297→    setSearchResults([])
   298→    setNavOpen(false)
   299→    setRecents((prev) => {
   300→      const nx = [key, ...prev.filter((k) => k !== key)].slice(0, 6)
   301→      persistJson(RECENT_KEY, nx)
   302→      return nx
   303→    })
   304→  }

   321→          : undefined)
   322→      }
   323→    }
   324→    window.addEventListener('erp:goto', handler)
   325→    return () => window.removeEventListener('erp:goto', handler)

   344→  // ---------- keyboard shortcuts engine ----------

   346→    if (!session) return

   348→      if ((e.metaKey || e.ctrlKey) && e.key.toLowerCase() === 'k') {
   349→        e.preventDefault()

   351→        return

   353→      if (paletteOpen || helpOpen || pwdOpen) return
   354→      if (isTypingTarget(e.target) || e.metaKey || e.ctrlKey || e.altKey) return
   355→
   356→      // g + key sequence (Gmail-style)
   357→      const now = Date.now()
   358→      if (goPendingRef.current && now - goPendingRef.current < 1500) {
   359→        goPendingRef.current = 0
   360→        const target = GO_KEYS[e.key.toLowerCase()]
   361→        if (target) {
   362→          e.preventDefault()
   363→          const def = findModule(target)
   364→          if (def && canAccessModule(session.user.role, target)) goToModule(target)
   365→          else if (def) toast.error(`${def.label} is not available for the ${ROLE_LABEL[session.user.role] || session.user.role} role`)
   366→          return
   367→        }
   368→      }
   369→      if (e.key === 'g') { goPendingRef.current = Date.now(); return }
   370→      if (e.key === '?') { e.preventDefault(); setHelpOpen((o) => !o); return }
   371→      if (e.key === '/') { e.preventDefault(); setPaletteOpen(true); return }
   372→      if (e.key === '[' || e.key === ']') {
   373→        e.preventDefault()
   374→        const dir = e.key === ']' ? 1 : -1
   375→        const idx = visibleModules.findIndex((m) => m.key === activeKey)
   376→        const nextIdx = (idx + dir + visibleModules.length) % visibleModules.length
   377→        const target = visibleModules[nextIdx]
   378→        if (target) goToModule(target.key)
   379→      }
   380→    }
   381→    window.addEventListener('keydown', onKey)
   382→    return () => window.removeEventListener('keydown', onKey)
   383→  }, [session, paletteOpen, helpOpen, pwdOpen, visibleModules, activeKey])
