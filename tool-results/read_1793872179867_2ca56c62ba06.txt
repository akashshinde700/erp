   584→                                        .inner_join(SalesOrderItem)

   597→                                                & (SalesOrder.docstatus == 1)
   598→                                                & (PackedItem.item_code == production_item)

   600→                                )

   602→                                if self.sales_order_item:
   603→                                        packed_so_query = packed_so_query.where(

   852→        def get_batch_capacity(self):
   853→                batches = frappe.get_all(
   854→                        "Batch",
   855→                        filters={"reference_doctype": self.doctype, "reference_name": self.name},
   856→                        fields=["name", "qty_to_produce"],
   857→                        order_by="creation",
   858→                )
   859→                serial_no = frappe.qb.DocType("Serial No")
   860→                used = dict(
   861→                        frappe.qb.from_(serial_no)
   862→                        .select(serial_no.batch_no, Count(serial_no.name))
   863→                        .where(
   864→                                (serial_no.work_order == self.name)
   865→                                & serial_no.batch_no.isin([d.name for d in batches] or [""])
   866→                        )
   867→                        .groupby(serial_no.batch_no)
   868→                        .run()
   869→                )
   870→                return [[d.name, cint(d.qty_to_produce) - cint(used.get(d.name))] for d in batches]

   872→        def create_batch_for_finished_good(self, total_qty=None):

   891→                        make_batch(
   892→                                frappe._dict(
   893→                                        {
   894→                                                "item": self.production_item,

  1022→        def validate_qty(self):

  1043→

  1171→                                        "time_in_mins": source.time_in_mins,
  1172→                                        "fixed_time": source.fixed_time,
  1173→                                        "batch_size": source.batch_size,
  1174→                                }
  1175→                        )
  1176→                        return row
  1177→
  1178→                if flt(operation.completed_qty) and flt(operation.actual_operation_time):
  1179→                        completed_qty = flt(operation.completed_qty) * flt(self.qty) / get_operation_qty(self, operation)
  1180→                        time_per_unit = flt(operation.actual_operation_time) / completed_qty

  1295→def get_work_order_allocated_qty(scio_item_name: str, rm_item_codes: list[str]) -> dict[str, float]:
  1296→        """Qty of each raw material held by the submitted Work Orders of an inward order item: a closed order the qty it used or still holds, any other its required qty."""
  1297→        work_order = frappe.qb.DocType("Work Order")
  1298→        work_order_item = frappe.qb.DocType("Work Order Item")
  1299→        return dict(
  1300→                frappe.qb.from_(work_order_item)
  1301→                .join(work_order)
  1302→                .on(work_order.name == work_order_item.parent)
  1303→                .select(
  1304→                        work_order_item.item_code,
  1305→                        Sum(
  1306→                                Case()
  1307→                                .when(
  1308→                                        (work_order.status == "Closed")
  1309→                                        & (work_order_item.transferred_qty > work_order_item.consumed_qty),
  1310→                                        work_order_item.transferred_qty - work_order_item.returned_qty,
  1311→                                )
  1312→                                .when(
  1313→                                        work_order.status == "Closed",
  1314→                                        work_order_item.consumed_qty - work_order_item.returned_qty,
  1315→                                )
  1316→                                .else_(work_order_item.required_qty)
  1317→                        ),
  1318→                )
  1319→                .where(
  1320→                        (work_order.docstatus == 1)
  1321→                        & (work_order.subcontracting_inward_order_item == scio_item_name)
  1322→                        & (work_order_item.item_code.isin(rm_item_codes))
  1323→                )
  1324→                .groupby(work_order_item.item_code)
  1325→                .run()
  1326→        )
