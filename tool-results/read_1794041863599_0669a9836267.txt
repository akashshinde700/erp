   294→   294→export function ErpApp() {
   295→   295→  const { view, params, setView } = useErpUi()
   296→   296→
   297→   297→  // ---------- organization login (SaaS) ----------
   298→   298→  // undefined = session probe in flight · null = signed out · info = signed in.
   299→   299→  // Platform Admin is a SEPARATE console (platformMode) — never inside the ERP.
   300→   300→  // A SUPER_ADMIN session opens the console directly (like the reference SaaS).
   301→   301→  const [session, setSession] = useState<SessionInfo | null | undefined>(undefined)
   302→   302→  const [platformMode, setPlatformMode] = useState(false)
   303→   303→
   304→   304→  // Boot routing uses the PUBLIC status probe (no sensitive data): it tells
   305→   305→  // first-boot (no company) apart from signed-out without ever touching the
   306→   306→  // company profile. The full profile (bank details included) is fetched ONLY
   307→   307→  // for a signed-in operating-org session — never anonymously.
   308→   308→  const status = useApi<AuthStatus | null>('/api/auth/status')
   309→   309→  const company = useApi<Company | null>(session === undefined ? null : session ? '/api/masters/company' : null)
   310→   310→  const [manualSetup, setManualSetup] = useState(false)
   311→   311→  const [drawerOpen, setDrawerOpen] = useState(false)
   312→   312→  const mounted = useMounted()
   313→   313→
   314→   314→  useEffect(() => {
   315→   315→    let cancelled = false
   316→   316→    api<SessionInfo | null>('/api/auth/session')
   317→   317→      .then((s) => {
   318→   318→        if (!cancelled) setSession(s)
   319→   319→      })
   320→   320→      .catch(() => {
   321→   321→        if (!cancelled) setSession(null)
   322→   322→      })
   323→   323→    return () => {
   324→   324→      cancelled = true
   325→   325→    }
   326→   326→  }, [])
   327→   327→
   328→   328→  const logout = async () => {
   329→   329→    try {
   330→   330→      await apiPost('/api/auth/logout', {})
   331→   331→      toast.success('Signed out')
   332→   332→    } catch {
   333→   333→      toast.error('Could not sign out — try again')
   334→   334→    }
   335→   335→    setSession(null)
   336→   336→    setView('dashboard')
   337→   337→  }
   338→   338→
   339→   339→  useEffect(() => {
   340→   340→    window.scrollTo({ top: 0 })
   341→   341→  }, [view])
   342→   342→
   343→   343→  // ---------- global keyboard: palette (Ctrl/⌘+K, /), g+key goto, ? help, Alt+S/Alt+P ----------
   344→   344→  const [paletteOpen, setPaletteOpen] = useState(false)
   345→   345→  const [helpOpen, setHelpOpen] = useState(false)
   346→   346→  const gotoArmed = useRef(0)
   347→   347→
   348→   348→  const openSalesCreate = useCallback(() => {
   349→   349→    setCreatePreset({ type: 'sales-invoice' })
   350→   350→    setView('sales')
   351→   351→  }, [setView])
   352→   352→  const openPurchaseCreate = useCallback(() => {
   353→   353→    setCreatePreset({ type: 'purchase-invoice' })
   354→   354→    setView('purchase')
   355→   355→  }, [setView])
   356→   356→
   357→   357→  useEffect(() => {
   358→   358→    const onKey = (e: KeyboardEvent) => {
   359→   359→      // Ctrl/⌘ + K → palette (always)
   360→   360→      if ((e.ctrlKey || e.metaKey) && e.key.toLowerCase() === 'k') {
   361→   361→        e.preventDefault()
   362→   362→        setPaletteOpen(true)
   363→   363→        return
   364→   364→      }
   365→   365→      if (isTypingTarget(e)) return
   366→   366→      // Alt+S / Alt+P → new documents from anywhere
   367→   367→      if (e.altKey && e.key.toLowerCase() === 's') {
   368→   368→        e.preventDefault()
   369→   369→        openSalesCreate()
   370→   370→        return
   371→   371→      }
   372→   372→      if (e.altKey && e.key.toLowerCase() === 'p') {
   373→   373→        e.preventDefault()
   374→   374→        openPurchaseCreate()
   375→   375→        return
   376→   376→      }
   377→   377→      // "?" → shortcuts help
   378→   378→      if (e.key === '?') {
   379→   379→        e.preventDefault()
   380→   380→        setHelpOpen(true)
   381→   381→        return
   382→   382→      }
   383→   383→      // "/" → palette
   384→   384→      if (e.key === '/') {
   385→   385→        e.preventDefault()
   386→   386→        setPaletteOpen(true)
   387→   387→        return
   388→   388→      }
   389→   389→      // g + key → module (Gmail style, 1.5 s window)
   390→   390→      if (e.key.toLowerCase() === 'g' && !e.ctrlKey && !e.metaKey && !e.altKey) {
   391→   391→        gotoArmed.current = Date.now()
   392→   392→        return
   393→   393→      }
   394→   394→      const goto = GOTO_KEYS[e.key.toLowerCase()]
   395→   395→      if (goto && Date.now() - gotoArmed.current < 1500 && !e.ctrlKey && !e.metaKey && !e.altKey) {
   396→   396→        e.preventDefault()
   397→   397→        gotoArmed.current = 0
   398→   398→        setView(goto.view)
   399→   399→      }
   400→   400→    }
   401→   401→    window.addEventListener('keydown', onKey)
   402→   402→    return () => window.removeEventListener('keydown', onKey)
   403→   403→  }, [setView, openSalesCreate, openPurchaseCreate])
   404→   404→
   405→   405→  // Track recent views for the palette
   406→   406→  useEffect(() => {
   407→   407→    pushRecent(view)
   408→   408→  }, [view])
   409→   409→
   410→   410→  // Boot probe: session + status must both settle before deciding. The
   411→   411→  // company profile loads lazily behind the session.
   412→   412→  if (session === undefined || (status.data === null && !status.error)) {
   413→   413→    return (
   414→   414→      <div className="flex min-h-screen flex-col gap-4 p-6">
   415→   415→        <Skeleton className="h-14 w-full rounded-lg" />
   416→   416→        <div className="flex flex-1 gap-4">
   417→   417→          <Skeleton className="hidden h-96 w-60 rounded-lg md:block" />
   418→   418→          <Skeleton className="h-96 flex-1 rounded-lg" />
   419→   419→        </div>
   420→   420→      </div>
   421→   421→    )
   422→   422→  }
   423→   423→
   424→   424→  // Status endpoint unreachable → honest offline/error screen (never a
   425→   425→  // wrong setup wizard).
   426→   426→  if (status.error && status.data === null) {
   427→   427→    return (
   428→   428→      <div className="flex min-h-screen flex-col items-center justify-center gap-4 p-6 text-center">
