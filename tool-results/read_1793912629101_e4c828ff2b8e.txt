   201→   201→  function replaceRow(org: OrgRow) {
   202→   202→    // PATCH returns the plain org row — keep the list-only firm fields (linkedFirms / firmOf)
   203→   203→    orgs.setData((d) => (d ? { items: d.items.map((r) => (r.id === org.id ? { ...r, ...org, linkedFirms: org.linkedFirms ?? r.linkedFirms, firmOf: org.firmOf !== undefined ? org.firmOf : r.firmOf } : r)) } : d))
   204→   204→  }
   205→   205→
   206→   206→  async function act(row: OrgRow, body: Record<string, unknown>, msg?: string) {
   207→   207→    setBusy(row.id + String(body.action))
   208→   208→    try {
   209→   209→      const d = await apiPatch<{ org: OrgRow; password?: string; email?: string }>(`/api/platform/orgs/${row.id}`, body)
   210→   210→      replaceRow(d.org)
   211→   211→      if (d.password && d.email) {
   212→   212→        setShownPw((m) => ({ ...m, [row.id]: d.password! }))
   213→   213→        setCreds({ org: row.name, email: d.email, password: d.password })
   214→   214→      } else if (msg) toast.success(msg)
   215→   215→      return true
   216→   216→    } catch (e) {
   217→   217→      toast.error(e instanceof Error ? e.message : 'Action failed')
   218→   218→      return false
   219→   219→    } finally {
   220→   220→      setBusy(null)
   221→   221→    }
   222→   222→  }
   223→   223→
   224→   224→  async function create() {
   225→   225→    if (!form.name.trim() || !form.adminEmail.trim()) { toast.error('Company name and admin email are required'); return }
   226→   226→    if (form.password && form.password.length < 8) { toast.error('Password must be at least 8 characters (or leave blank to generate)'); return }
   227→   227→    setSaving(true)
   228→   228→    try {
   229→   229→      const d = await apiPost<{ org: OrgRow; password: string }>('/api/platform/orgs', { ...form, trialDays: Number(form.trialDays || 0) })
   230→   230→      orgs.setData((prev) => ({ items: [d.org, ...(prev?.items ?? [])] }))
   231→   231→      setShownPw((m) => ({ ...m, [d.org.id]: d.password }))
   232→   232→      setCreds({ org: d.org.name, email: d.org.admin?.email || form.adminEmail, password: d.password })
   233→   233→      setCreateOpen(false)
   234→   234→      setForm(EMPTY_CREATE)
   235→   235→    } catch (e) {
   236→   236→      toast.error(e instanceof Error ? e.message : 'Failed to create organization')
   237→   237→    } finally {
   238→   238→      setSaving(false)
   239→   239→    }
   240→   240→  }

   270→   270→    } finally {
   271→   271→      setBusy(null)
   272→   272→    }

   300→   300→              <TableHead className="min-w-40">Company</TableHead>

   303→   303→              <TableHead>Status</TableHead>

   477→   477→interface PlanForm { name: string; priceMonthly: string; priceYearly: string; maxUsers: string; maxWarehouses: string; maxItems: string }
   478→   478→const EMPTY_PLAN: PlanForm = { name: '', priceMonthly: '0', priceYearly: '0', maxUsers: '0', maxWarehouses: '0', maxItems: '0' }
   479→   479→
   480→   480→function Plans() {
   481→   481→  const plans = useList<{ items: PlanRow[] }>('/api/platform/plans')
   482→   482→  const rows = plans.data?.items ?? []
   483→   483→  const [open, setOpen] = React.useState(false)
   484→   484→  const [editId, setEditId] = React.useState<string | null>(null)
   485→   485→  const [form, setForm] = React.useState<PlanForm>(EMPTY_PLAN)
   486→   486→  const [saving, setSaving] = React.useState(false)
   487→   487→  const setF = (k: keyof PlanForm) => (e: React.ChangeEvent<HTMLInputElement>) => setForm((f) => ({ ...f, [k]: e.target.value }))
   488→   488→
   489→   489→  async function save() {
   490→   490→    if (!form.name.trim()) { toast.error('Plan name is required'); return }
   491→   491→    setSaving(true)
   492→   492→    try {
   493→   493→      const body = { ...form, name: form.name.trim() }
   494→   494→      if (editId) await apiPatch(`/api/platform/plans/${editId}`, body)
   495→   495→      else await apiPost('/api/platform/plans', body)
   496→   496→      toast.success(editId ? 'Plan updated' : 'Plan created')
   497→   497→      setOpen(false)
   498→   498→      await plans.refetch()
   499→   499→    } catch (e) {
   500→   500→      toast.error(e instanceof Error ? e.message : 'Failed to save plan')
   501→   501→    } finally {
   502→   502→      setSaving(false)
   503→   503→    }
   504→   504→  }
   505→   505→
   506→   506→  async function remove(p: PlanRow) {
   507→   507→    try {
   508→   508→      await apiDelete(`/api/platform/plans/${p.id}`)
   509→   509→      toast.success('Plan deleted')
   510→   510→      await plans.refetch()
   511→   511→    } catch (e) {
   512→   512→      toast.error(e instanceof Error ? e.message : 'Failed to delete plan')
   513→   513→    }
   514→   514→  }
   515→   515→

   517→   517→    <div>
   518→   518→      <div className="mb-4 flex flex-wrap items-center justify-between gap-3">

   524→   524→          <Plus className="mr-1.5 h-4 w-4" />Plan
   525→   525→        </Button>
   526→   526→      </div>
   527→   527→      <Card className="overflow-hidden p-0">
   528→   528→        <Table>
   529→   529→          <TableHeader>
   530→   530→            <TableRow>
   531→   531→              <TableHead>Plan</TableHead>
   532→   532→              <TableHead className="text-right">Monthly</TableHead>
   533→   533→              <TableHead className="text-right">Yearly</TableHead>
   534→   534→              <TableHead className="text-right">Users</TableHead>
   535→   535→              <TableHead className="text-right">Warehouses</TableHead>
   536→   536→              <TableHead className="text-right">Items</TableHead>
   537→   537→              <TableHead className="text-right">Organizations</TableHead>
   538→   538→              <TableHead>Actions</TableHead>
   539→   539→            </TableRow>
   540→   540→          </TableHeader>
   541→   541→          <TableBody>
   542→   542→            {!plans.loading && !rows.length && (
