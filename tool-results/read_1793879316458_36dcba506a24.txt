   127→function csvEscape(v: string | number): string {
   128→  const s = String(v ?? '')
   129→  return /[",\n]/.test(s) ? `"${s.replaceAll('"', '""')}"` : s
   130→}

   132→export function downloadCsv(filename: string, headers: string[], rows: (string | number)[][]) {
   133→  const csv = [headers, ...rows].map((r) => r.map(csvEscape).join(',')).join('\n')
   134→  const blob = new Blob(['\ufeff' + csv], { type: 'text/csv;charset=utf-8;' })
   135→  const url = URL.createObjectURL(blob)
   136→  const a = document.createElement('a')
   137→  a.href = url
   138→  a.download = `${filename}-${new Date().toISOString().slice(0, 10)}.csv`
   139→  a.click()
   140→  URL.revokeObjectURL(url)
   141→}

   145→/**
   146→ * Builds a real Excel workbook (one sheet) from a 2-D array — first data row is
   147→ * the header row. Numbers stay number cells (Excel can sum/filter them) and each
   148→ * column gets a width derived from its longest content (capped). Browser-only:
   149→ * XLSX.writeFile triggers the file download.
   150→ */
   151→export function downloadXlsx(filename: string, headers: string[], rows: (string | number)[][]) {
   152→  const ws = XLSX.utils.aoa_to_sheet([headers, ...rows])
   153→  ws['!cols'] = headers.map((h, col) => {
   154→    let max = String(h ?? '').length
   155→    for (const row of rows) {
   156→      const len = String(row[col] ?? '').length
   157→      if (len > max) max = len
   158→    }
   159→    return { wch: Math.min(Math.max(max + 2, 10), 45) }
   160→  })
   161→  const wb = XLSX.utils.book_new()
   162→  XLSX.utils.book_append_sheet(wb, ws, 'Data')
   163→  XLSX.writeFile(wb, `${filename}.xlsx`)
   164→}

   183→/** Escape a string for safe use inside a CSS attribute selector (ids are plain cuids — defensive only). */
   184→function cssAttrEscape(s: string): string {
   185→  return typeof CSS !== 'undefined' && typeof CSS.escape === 'function'
   186→    ? CSS.escape(s)
   187→    : s.replace(/["\\]/g, '\\$&')
   188→}
   189→
   190→/**
   191→ * Reads the deep-link written by the global search palette (sessionStorage key
   192→ * 'erp_deep_link': { module, id, ts }) and returns the record id that should be
   193→ * highlighted in this module's table — or null. The entry is consumed (removed)
   194→ * on first read and the highlight auto-clears after ~8s, making it a one-shot pulse.
   195→ * Pass the module's key (e.g. 'items', 'sales-orders'); deep links for other
   196→ * modules are ignored (and consumed) by the first mounted module that reads them.
   197→ */
   198→export function useRowHighlight(moduleKey: string | null | undefined): string | null {
   199→  const [highlightId, setHighlightId] = React.useState<string | null>(null)
   200→
   201→  // Read once per mount. The sessionStorage key is removed on every successful
   202→  // read, so effect re-runs (moduleKey change without a remount) are no-ops.
   203→  React.useEffect(() => {
   204→    try {
   205→      const raw = sessionStorage.getItem('erp_deep_link')
   206→      if (!raw) return
   207→      const d = JSON.parse(raw) as { module?: unknown; id?: unknown; ts?: unknown }
   208→      sessionStorage.removeItem('erp_deep_link')
   209→      if (d && d.module === moduleKey && typeof d.id === 'string' && d.id.trim() !== '') {
   210→        setHighlightId(String(d.id))
   211→      }
   212→    } catch {
   213→      // Malformed / unavailable deep link — best-effort UX nicety, ignore.
   214→    }
   215→  }, [moduleKey])
   216→
   217→  // Auto-clear the pulse so a stale highlight never sticks around.
   218→  React.useEffect(() => {
   219→    if (!highlightId) return
   220→    const t = setTimeout(() => setHighlightId(null), 8000)
   221→    return () => clearTimeout(t)
   222→  }, [highlightId])
   223→
   224→  // Same-module deep link while already mounted (no remount happens): the palette
   225→  // also broadcasts 'erp:deep-link' so we re-highlight live.
   226→  React.useEffect(() => {
   227→    const handler = (e: Event) => {
   228→      try {
   229→        const d = (e as CustomEvent<{ module?: unknown; id?: unknown }>).detail
   230→        if (d && d.module === moduleKey && typeof d.id === 'string' && d.id.trim() !== '') {
   231→          setHighlightId(null) // restart the pulse even for the same row
   232→          window.setTimeout(() => setHighlightId(String(d.id)), 0)
   233→        }
   234→      } catch {
   235→        // best-effort UX nicety — ignore
   236→      }
   237→    }
   238→    window.addEventListener('erp:deep-link', handler)
   239→    return () => window.removeEventListener('erp:deep-link', handler)
   240→  }, [moduleKey])
   241→
   242→  return highlightId
   243→}

   245→// ---------- Saved views (per-module localStorage search presets) ----------
   246→
   247→interface SavedView { name: string; q: string; ts: number }
   248→
   249→const VIEWS_STORAGE_PREFIX = 'manuflow:views:'
   250→
   251→/** Reads + sanitizes the saved-view list for a storage key. Private-mode safe (returns [] on any failure). */
   252→function loadSavedViews(key: string): SavedView[] {
   253→  try {
   254→    const raw = localStorage.getItem(VIEWS_STORAGE_PREFIX + key)
   255→    if (!raw) return []
   256→    const parsed: unknown = JSON.parse(raw)
   257→    if (!Array.isArray(parsed)) return []
   258→    return parsed
   259→      .filter((v): v is SavedView =>
   260→        !!v && typeof v === 'object' &&
   261→        typeof (v as SavedView).name === 'string' && typeof (v as SavedView).q === 'string')
   262→      .map((v) => ({ name: v.name, q: v.q, ts: typeof v.ts === 'number' ? v.ts : 0 }))
   263→  } catch {
   264→    return []
   265→  }
   266→}
   267→
   268→/** Writes the saved-view list for a storage key. Private-mode safe (silently no-ops). */
   269→function persistSavedViews(key: string, views: SavedView[]) {
   270→  try {
   271→    localStorage.setItem(VIEWS_STORAGE_PREFIX + key, JSON.stringify(views))
   272→  } catch {
   273→    // Storage unavailable (private mode / quota) — views stay in-memory for this session.
   274→  }
   275→}

   279→export interface Column<T> {
   280→  key: string
   281→  header: string
   282→  render: (row: T) => React.ReactNode
   283→  className?: string
   284→  hideOnMobile?: boolean
   285→  /** Override extractor for the value behind this column — used by DataTable column sorting (`sortable`) and the totals footer (`sum`). Needed for computed columns whose key has no raw row field. */
   286→  sortValue?: (row: T) => string | number
   287→  /** Includes this column in the totals footer row: finite numeric values are summed (client mode: all filtered rows; server mode: current page rows). */
   288→  sum?: boolean
   289→  /** Footer total format — 'money' renders via fmtMoney, 'num' (default) via fmtNum. */
   290→  sumFormat?: 'money' | 'num'
   291→}
   292→
   293→/** Raw value behind a column for sorting/summing: the column's sortValue override when set, else the row's own field under the column key. */
   294→function columnValue<T>(row: T, col: Column<T>): unknown {
   295→  return col.sortValue ? col.sortValue(row) : (row as Record<string, unknown>)[col.key]
   296→}
   297→
   298→/**
   299→ * Three-way compare for extracted column values: numbers compare numerically,
   300→ * strings lexicographically (ISO date strings order correctly), mixed types
   301→ * fall back to a String() compare. undefined/null never reach this — the sort
   302→ * comparator pulls empties to the end in both directions (empties-always-last).
   303→ */
   304→function compareSortValues(a: unknown, b: unknown): number {
   305→  if (typeof a === 'number' && typeof b === 'number') return a - b
   306→  if (typeof a === 'string' && typeof b === 'string') return a.localeCompare(b)
   307→  const sa = String(a)
   308→  const sb = String(b)
   309→  return sa < sb ? -1 : sa > sb ? 1 : 0
   310→}

   312→/**
   313→ * Desktop/tablet vs phone, resolved after hydration.
   314→ *
   315→ * DataTable renders exactly one tree — never both — so screen readers, DOM queries and text search see a
   316→ * single copy of every record. The server snapshot says "desktop", which is also the pre-hydration paint:
   317→ * the table is the default document, and phones swap to record cards the moment React runs.
   318→ */
   319→const DESKTOP_QUERY = '(min-width: 768px)'
   320→function subscribeDesktop(onChange: () => void) {
   321→  const mql = window.matchMedia(DESKTOP_QUERY)
   322→  mql.addEventListener('change', onChange)
   323→  return () => mql.removeEventListener('change', onChange)
   324→}
   325→function useIsDesktop(): boolean {
   326→  return React.useSyncExternalStore(
   327→    subscribeDesktop,
   328→    () => window.matchMedia(DESKTOP_QUERY).matches,
   329→    () => true,
   330→  )
   331→}

   333→export function DataTable<T extends { id: string }>({
   334→  columns, rows, loading, emptyText = 'No records yet', searchPlaceholder, onRowClick, maxHeight = 'auto', pageSize = 12, csvExport, xlsxExport, highlightId, moduleKey, savedViewsKey, serverPagination, sortable, mobileView = 'cards',
   335→}: {
   336→  columns: Column<T>[]
   337→  rows: T[]
   338→  loading?: boolean
   339→  emptyText?: string
   340→  searchPlaceholder?: string
   341→  onRowClick?: (row: T) => void
   342→  maxHeight?: string
   343→  pageSize?: number
   344→  /** When provided, shows an "Export CSV" button exporting all filtered rows. */
   345→  csvExport?: { filename: string; headers?: string[]; pick: (row: T) => (string | number)[] }
   346→  /** When provided, shows an "Excel" button (beside the CSV one) exporting the same filtered rows as a real .xlsx. Headers default to the column labels — identical columns to the CSV. */
   347→  xlsxExport?: { filename: string; headers?: string[]; pick: (row: T) => (string | number)[] }
   348→  /** Deep-linked record id (from useRowHighlight) — page-jumps to + scrolls the row into view and highlights it. */
   349→  highlightId?: string | null
   350→  /** Opts into the dashboard filter-preset channel: on mount consumes sessionStorage 'erp_filter_preset' ({ module, q, ts }) when its module matches, and live-listens for 'erp:filter-preset' CustomEvents (detail { module, q }). Pass the module's key (e.g. 'items'). No effect when omitted. */
   351→  moduleKey?: string
   352→  /** Enables the "Views" saved-search control. Views persist in localStorage under `manuflow:views:<savedViewsKey>` — pass `${companyId}:<moduleKey>` so views are scoped per company + module. No effect when omitted. */
   353→  savedViewsKey?: string
   354→  /** Opts the whole table into column sorting: every column with usable values (a sortValue extractor, or at least one row with a non-null raw value under its key) gets a clickable header cycling asc → desc → off. Client mode sorts the full filtered set before pagination (filter → sort → paginate); server mode only reorders the current page slice (the API owns the real order). undefined/null values sort last in both directions. Sort state persists across refetches. No effect when omitted. */
   355→  sortable?: boolean
   356→  /**
   357→   * How a phone (< 768px) shows the records:
   358→   *  • 'cards' (default) — one card per record: primary identifier, secondary fields as label/value
   359→   *    pairs, actions in the card header. Every non-`hideOnMobile` column is still shown, so nothing is
   360→   *    silently dropped; the desktop table is unchanged.
   361→   *  • 'scroll' — the full table with horizontal scrolling, for wide numeric matrices where the column
   362→   *    alignment itself carries meaning (a comparison across many periods, for example).
   363→   */
   364→  mobileView?: 'cards' | 'scroll'
   365→  /** Opts into server-side pagination: `rows` must then be exactly the page slice the API returned (already filtered + paginated). The local text filter and internal page state are bypassed — Prev/Next and the "Showing …" footer are driven by `page`/`onPageChange`/`total`, and (with `onSearchChange`) the search text is pushed to the module debounced by 300ms (mount run skipped). Omit for the default fully client-side behavior (zero change). */
   366→  serverPagination?: {
   367→    /** 0-based page index, controlled by the module (Prev → onPageChange(page - 1), Next → onPageChange(page + 1)). */
   368→    page: number
   369→    /** Rows per server page — the module must fetch exactly this many. */
   370→    pageSize: number
   371→    /** Full filtered row count from the server (across all pages). */
   372→    total: number
   373→    /** Called when the user pages (Prev/Next) or a highlight page-jump targets another page. */
   374→    onPageChange: (page: number) => void
   375→    /** Debounced (300ms) search-text push; the module should store it as its server-side q filter and reset to page 0. */
   376→    onSearchChange?: (q: string) => void
   377→  }
   378→}) {
   379→  const [q, setQ] = React.useState('')
   380→  const [appliedView, setAppliedView] = React.useState<string | null>(null)
   381→  const [page, setPage] = React.useState(0)
   382→  const wrapperRef = React.useRef<HTMLDivElement | null>(null)
   383→
   384→  const filtered = React.useMemo(() => {
   385→    if (serverPagination) return rows // server mode: the API already filtered + sliced — rows are the page as-is
   386→    if (!q.trim()) return rows
   387→    const needle = q.toLowerCase()
   388→    return rows.filter((r) => JSON.stringify(r).toLowerCase().includes(needle))
   389→  }, [rows, q, serverPagination])
   390→
   391→  React.useEffect(() => setPage(0), [q, rows.length])
   392→
   393→  // ---- Column sorting (sortable) ----
   394→  // Active sort: { key, dir } cycling asc → desc → off per header click; null
   395→  // = unsorted. Persists across refetches — deliberately no reset on data change.
   396→  const [sort, setSort] = React.useState<{ key: string; dir: 'asc' | 'desc' } | null>(null)
   397→
   398→  // Columns with usable values: a column is sortable when it declares a
   399→  // sortValue extractor or at least one display row has a non-null raw value
   400→  // under its key (keeps 'actions'-style render-only columns inert).
   401→  const sortableKeys = React.useMemo(() => {
   402→    const set = new Set<string>()
   403→    if (!sortable) return set
   404→    for (const c of columns) {
   405→      if (c.sortValue) { set.add(c.key); continue }
   406→      for (const row of filtered) {
   407→        const v = (row as Record<string, unknown>)[c.key]
   408→        if (v !== undefined && v !== null) { set.add(c.key); break }
   409→      }
   410→    }
   411→    return set
   412→  }, [sortable, columns, filtered])
   413→
   414→  // Display pipeline: filter (above) → sort (here) → paginate (pageRows
   415→  // below). Sorting is opt-in; without `sortable` (or an active sort) this is
   416→  // `filtered` verbatim. Server mode sorts nothing — `rows` are the API's page
   417→  // slice, so header clicks only reorder the current page.
   418→  const displayRows = React.useMemo(() => {
   419→    if (serverPagination || !sortable || !sort) return filtered
   420→    const col = columns.find((c) => c.key === sort.key)
   421→    if (!col) return filtered
   422→    const dirMul = sort.dir === 'asc' ? 1 : -1
   423→    return filtered
   424→      .map((row, i) => ({ row, i, v: columnValue(row, col) }))
   425→      .sort((a, b) => {
   426→        const aEmpty = a.v === undefined || a.v === null
   427→        const bEmpty = b.v === undefined || b.v === null
   428→        if (aEmpty !== bEmpty) return aEmpty ? 1 : -1 // empties always last, asc + desc
   429→        const c = compareSortValues(a.v, b.v)
   430→        return c !== 0 ? dirMul * c : a.i - b.i // stable ties keep the filtered order
   431→      })
   432→      .map((x) => x.row)
   433→  }, [filtered, columns, sortable, sort, serverPagination])
   434→
   435→  // Deep-link: page-jump to + scroll the highlighted row into view. Rows may
   436→  // arrive async (table can be loading when the highlight first applies), so
   437→  // attempts are retried across a few ticks; state kept in a ref keyed by the
   438→  // highlight id makes each jump a strict one-shot that never fights the
   439→  // user's own paging/filtering afterwards.
   440→  const jumpStateRef = React.useRef<{ id: string | null; jumped: boolean; scrolled: boolean }>({ id: null, jumped: false, scrolled: false })
   441→
   442→  // Latest serverPagination config, readable inside effects below without
   443→  // re-triggering them on every render (the object is recreated per render).
   444→  // Synced in a bare effect (runs after every render, before the consumers
   445→  // declared after it) so effects always read the freshest config.
   446→  const serverPaginationRef = React.useRef(serverPagination)
   447→  React.useEffect(() => {
   448→    serverPaginationRef.current = serverPagination
   449→  })
   450→
   451→  React.useEffect(() => {
   452→    if (!highlightId) return
   453→    const st = jumpStateRef.current
   454→    if (st.id !== highlightId) {
   455→      st.id = highlightId
   456→      st.jumped = false
   457→      st.scrolled = false
   458→    }
