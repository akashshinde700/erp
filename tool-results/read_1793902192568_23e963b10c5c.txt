    92→function GstinHint({ value }: { value: string }) {
    93→  if (!value.trim()) return <p className="text-xs text-muted-foreground">Optional for B2C / unregistered.</p>
    94→  const ok = gstinChecksumValid(value)
    95→  return (
    96→    <p className={`flex items-center gap-1 text-xs font-medium ${ok ? 'text-primary' : 'text-destructive'}`}>
    97→      {ok ? <CheckCircle2 className="size-3.5" aria-hidden /> : <XCircle className="size-3.5" aria-hidden />}
    98→      {ok ? 'Checksum valid' : 'Invalid GSTIN — checksum failed (format: 27ABCDE1234F1Z5)'}
    99→    </p>
   100→  )
   101→}

   129→function PartiesTab() {
   130→  const list = useApi<Party[]>('/api/masters/parties')
   131→  const setView = useErpUi((s) => s.setView)
   132→  const [search, setSearch] = useState('')
   133→  const [dialogOpen, setDialogOpen] = useState(false)
   134→  const [editing, setEditing] = useState<Party | null>(null)
   135→  const [viewing, setViewing] = useState<Party | null>(null)
   136→
   137→  const parties = (list.data ?? []).filter((p) => {
   138→    const q = search.trim().toLowerCase()
   139→    if (!q) return true
   140→    return p.name.toLowerCase().includes(q) || p.code.toLowerCase().includes(q) || (p.gstin ?? '').toLowerCase().includes(q)
   141→  })
   142→
   143→  const toggleActive = async (p: Party) => {
   144→    try {
   145→      await apiDelete(`/api/masters/parties/${p.id}${p.active ? '' : '?activate=1'}`)
   146→      toast.success(`${p.code} ${p.active ? 'deactivated' : 'reactivated'}`)
   147→      list.reload()
   148→    } catch (e) {
   149→      toastError(e)
   150→    }
   151→  }
   152→
   153→  return (
   154→    <div className="space-y-4">
   155→      <div className="flex flex-wrap items-center gap-2">
   156→        <div className="relative min-w-52 flex-1">
   157→          <Search className="absolute left-2.5 top-1/2 size-4 -translate-y-1/2 text-muted-foreground" aria-hidden />
   158→          <Input
   159→            value={search}
   160→            onChange={(e) => setSearch(e.target.value)}
   161→            placeholder="Search name / code / GSTIN…"
   162→            className="h-11 pl-8"
   163→            aria-label="Search parties"
   164→          />
   165→        </div>
   166→        <Button
   167→          onClick={() => {
   168→            setEditing(null)
   169→            setDialogOpen(true)
   170→          }}
   171→          className="h-11 gap-2"
   172→        >
   173→          <Plus className="size-4" aria-hidden /> New Party
   174→        </Button>
   175→      </div>
   176→
   177→      <Card className="p-0">
   178→        <CardContent className="p-0">
   179→          {list.loading ? (
   180→            <TableSkeleton rows={8} cols={6} />
   181→          ) : parties.length === 0 ? (
   182→            <div className="p-6">
   183→              <EmptyState title="No parties" hint="Create customers, suppliers and jobworkers here." icon={Building2} />
   184→            </div>
   185→          ) : (
   186→            <div className="max-h-[62vh] overflow-auto erp-scroll">
   187→              <Table>
   188→                <TableHeader className="sticky top-0 z-10 bg-card">
   189→                  <TableRow>
   190→                    <TableHead>Code</TableHead>
   191→                    <TableHead>Name</TableHead>
   192→                    <TableHead>Types</TableHead>
   193→                    <TableHead>GSTIN</TableHead>
   194→                    <TableHead>State</TableHead>
   195→                    <TableHead>City</TableHead>
   196→                    <TableHead className="text-right">Opening Bal</TableHead>
   197→                    <TableHead>Status</TableHead>
   198→                    <TableHead className="w-36" />
   199→                  </TableRow>
   200→                </TableHeader>
   201→                <TableBody>
   202→                  {parties.map((p) => (
   203→                    <TableRow key={p.id} className="cursor-pointer" onClick={() => setViewing(p)}>
   204→                      <TableCell className="font-mono text-xs font-medium">{p.code}</TableCell>
   205→                      <TableCell className="max-w-44 truncate font-medium">{p.name}</TableCell>
   206→                      <TableCell>
   207→                        <div className="flex flex-wrap gap-1">

   216→                      <TableCell className="text-xs">{stateName(p.stateCode)}</TableCell>

   219→                        <Money value={p.openingBal} plain />

   269→      <PartyViewDialog party={viewing} onOpenChange={(o) => !o && setViewing(null)} onNavigate={setView} />

   272→}

   278→function PartyViewDialog({
   279→  party,
   280→  onOpenChange,
   281→  onNavigate,
   282→}: {
   283→  party: Party | null
   284→  onOpenChange: (o: boolean) => void
   285→  onNavigate: (view: import('./store').ErpView) => void
   286→}) {
   287→  const open = !!party
   288→  const types = party?.types.split(',').filter(Boolean) ?? []
   289→  const isCustomer = types.includes('CUSTOMER')
   290→  const isSupplier = types.includes('SUPPLIER')
   291→
   292→  const sales = useApi<{ id: string; invoiceNo: string; date: string; grandTotal: number; paidAmount: number; status: string }[] | null>(
   293→    party && isCustomer ? `/api/sales-invoices?partyId=${party.id}` : null
   294→  )
   295→  const purchases = useApi<{ id: string; invoiceNo: string; date: string; grandTotal: number; paidAmount: number; status: string }[] | null>(
   296→    party && isSupplier ? `/api/purchase-invoices?partyId=${party.id}` : null
   297→  )
   298→  const payments = useApi<{ id: string; paymentNo: string; date: string; type: string; amount: number; mode: string | null; status: string }[] | null>(
   299→    party ? `/api/payments?partyId=${party.id}` : null
   300→  )
   301→
   302→  if (!party) return null
   303→
   304→  const close = () => onOpenChange(false)
   305→
   306→  const newInvoice = () => {
   307→    setCreatePreset({ type: 'sales-invoice', partyId: party.id })
   308→    close()
   309→    onNavigate('sales')
   310→  }
   311→  const newPurchase = () => {
   312→    setCreatePreset({ type: 'purchase-invoice', partyId: party.id })
   313→    close()
   314→    onNavigate('purchase')
   315→  }
   316→  const receivePayment = () => {
   317→    setCreatePreset({ type: 'payment', partyId: party.id, paymentType: 'RECEIPT' })
   318→    close()
   319→    onNavigate('accounting')
   320→  }
   321→  const makePayment = () => {
   322→    setCreatePreset({ type: 'payment', partyId: party.id, paymentType: 'PAYMENT' })
   323→    close()
   324→    onNavigate('accounting')
   325→  }
   326→  const openLedger = () => {
   327→    setCreatePreset({ type: 'party-ledger', partyId: party.id })
   328→    close()
   329→    onNavigate('accounting')
   330→  }
   331→
   332→  const salesRows = sales.data ?? []
   333→  const purchaseRows = purchases.data ?? []
   334→  const paymentRows = payments.data ?? []
   335→  const receivable = salesRows
   336→    .filter((i) => i.status === 'POSTED')
   337→    .reduce((s, i) => s + i.grandTotal - i.paidAmount, 0)
   338→  const payable = purchaseRows
   339→    .filter((i) => i.status === 'POSTED')
   340→    .reduce((s, i) => s + i.grandTotal - i.paidAmount, 0)
   341→
   342→  const invLine = (i: { id: string; invoiceNo: string; date: string; grandTotal: number; paidAmount: number; status: string }) => (
   343→    <TableRow key={i.id}>
   344→      <TableCell className="font-medium tabular-nums">{i.invoiceNo}</TableCell>
   345→      <TableCell className="whitespace-nowrap text-xs">{dateStr(i.date)}</TableCell>
   346→      <TableCell className="text-right tabular-nums">₹ {money(i.grandTotal)}</TableCell>
   347→      <TableCell className="text-right tabular-nums text-muted-foreground">₹ {money(i.paidAmount)}</TableCell>
   348→      <TableCell className="text-right font-medium tabular-nums">₹ {money(i.grandTotal - i.paidAmount)}</TableCell>
   349→      <TableCell>
   350→        <Badge variant={i.status === 'POSTED' ? 'default' : 'destructive'} className="text-[10px]">
   351→          {i.status}
   352→        </Badge>
   353→      </TableCell>
   354→    </TableRow>
   355→  )
   356→
   357→  return (
   358→    <Dialog open={open} onOpenChange={onOpenChange}>
   359→      <DialogContent className="flex max-h-[92vh] w-full max-w-3xl flex-col overflow-hidden sm:max-w-3xl">
   360→        <DialogHeader>
   361→          <DialogTitle className="flex flex-wrap items-center gap-2">
   362→            {party.name}
   363→            <span className="font-mono text-xs font-normal text-muted-foreground">{party.code}</span>
   364→            {types.map((t) => (
   365→              <Badge key={t} variant={t === 'CUSTOMER' ? 'default' : t === 'SUPPLIER' ? 'secondary' : 'outline'} className="text-[10px]">
   366→                {t}
   367→              </Badge>
   368→            ))}
   369→          </DialogTitle>
   370→          <DialogDescription>
   371→            {party.gstin ? `${party.gstin} · ` : 'Unregistered · '}
   372→            {stateName(party.stateCode)}
   373→            {party.city ? ` · ${party.city}` : ''}
   374→            {party.pincode ? ` ${party.pincode}` : ''}
   375→            {party.phone ? ` · ${party.phone}` : ''}
   376→          </DialogDescription>
   377→        </DialogHeader>
   378→
   379→        {/* One-click next actions — the Mapsoft-style flow buttons */}
   380→        <div className="flex flex-wrap gap-2">
   381→          {isCustomer ? (
   382→            <Button size="sm" className="h-10 gap-2" onClick={newInvoice}>
   383→              <ReceiptText className="size-4" aria-hidden /> New Invoice
   384→            </Button>
   385→          ) : null}
   386→          {isSupplier ? (
   387→            <Button size="sm" variant="outline" className="h-10 gap-2" onClick={newPurchase}>
   388→              <ShoppingCart className="size-4" aria-hidden /> New Purchase
   389→            </Button>
   390→          ) : null}
   391→          {isCustomer ? (
   392→            <Button size="sm" variant="outline" className="h-10 gap-2" onClick={receivePayment}>
   393→              <HandCoins className="size-4" aria-hidden /> Receive Payment
   394→            </Button>
   395→          ) : null}
   396→          {isSupplier ? (
   397→            <Button size="sm" variant="outline" className="h-10 gap-2" onClick={makePayment}>
   398→              <HandCoins className="size-4" aria-hidden /> Pay Supplier
   399→            </Button>
   400→          ) : null}
   401→          <Button size="sm" variant="ghost" className="h-10 gap-2" onClick={openLedger}>
   402→            <BookOpen className="size-4" aria-hidden /> Party Ledger
   403→          </Button>
   404→        </div>
   405→
   406→        <div className="grid grid-cols-3 gap-2">
   407→          <div className="rounded-md border bg-muted/30 p-2.5 text-center">
   408→            <p className="text-[10px] font-medium uppercase tracking-wide text-muted-foreground">Receivable</p>
   409→            <p className="text-sm font-semibold tabular-nums text-primary">₹ {money(Math.max(0, receivable))}</p>
   410→          </div>
   411→          <div className="rounded-md border bg-muted/30 p-2.5 text-center">
   412→            <p className="text-[10px] font-medium uppercase tracking-wide text-muted-foreground">Payable</p>
   413→            <p className="text-sm font-semibold tabular-nums text-destructive">₹ {money(Math.max(0, payable))}</p>
   414→          </div>
   415→          <div className="rounded-md border bg-muted/30 p-2.5 text-center">
   416→            <p className="text-[10px] font-medium uppercase tracking-wide text-muted-foreground">Vouchers</p>
   417→            <p className="text-sm font-semibold tabular-nums">{salesRows.length + purchaseRows.length + paymentRows.length}</p>
   418→          </div>
   419→        </div>
   420→
   421→        <div className="min-h-0 flex-1 overflow-hidden">
   422→          <Tabs defaultValue={isCustomer ? 'sales' : 'purchases'}>
   423→            <TabsList className="h-9">
