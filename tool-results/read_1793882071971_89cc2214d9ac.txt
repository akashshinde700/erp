    93→    try {
    94→      await apiDelete(`/api/payments/${p.id}`)

    96→        description: 'Allocations reversed and invoice paid amounts rolled back.',

   100→      toastError(e)

   102→  }
   103→
   104→  return (
   105→    <div className="space-y-4">
   106→      <div className="flex items-center justify-between gap-2">
   107→        <p className="text-sm text-muted-foreground">
   108→          {payments.length} voucher{payments.length === 1 ? '' : 's'} — receipts & payments with allocation.
   109→        </p>
   110→        <Button onClick={() => setOpen(true)} className="h-11 gap-2">
   111→          <Plus className="size-4" aria-hidden /> New Payment
   112→        </Button>
   113→      </div>
   114→
   115→      <Card className="p-0">
   116→        <CardContent className="p-0">
   117→          {list.loading ? (
   118→            <TableSkeleton rows={6} cols={6} />
   119→          ) : payments.length === 0 ? (
   120→            <div className="p-6">

   122→                title="No payments yet"
   123→                hint="Record receipts from customers or payments to suppliers, allocated against open invoices."

   126→                    <Plus className="size-4" aria-hidden /> New Payment
   127→                  </Button>
   128→                }
   129→              />
   130→            </div>
   131→          ) : (
   132→            <div className="max-h-[62vh] overflow-auto erp-scroll">
   133→              <Table>

   135→                  <TableRow>
   136→                    <TableHead>Voucher No</TableHead>
   137→                    <TableHead>Date</TableHead>
   138→                    <TableHead>Type</TableHead>
   139→                    <TableHead>Party</TableHead>
   140→                    <TableHead className="text-right">Amount</TableHead>

   142→                    <TableHead className="text-right">Allocated</TableHead>

   225→function NewPaymentDialog({
   226→  open,
   227→  onOpenChange,
   228→  onCreated,
   229→}: {
   230→  open: boolean
   231→  onOpenChange: (o: boolean) => void
   232→  onCreated: () => void
   233→}) {
   234→  const { parties } = useMasters()
   235→  const [type, setType] = useState<'RECEIPT' | 'PAYMENT'>('RECEIPT')
   236→  const [partyId, setPartyId] = useState('')
   237→  const [date, setDate] = useState(todayISO())
   238→  const [amount, setAmount] = useState('')
   239→  const [mode, setMode] = useState('BANK')
   240→  const [refNo, setRefNo] = useState('')
   241→  const [notes, setNotes] = useState('')
   242→  const [allocs, setAllocs] = useState<Record<string, string>>({})
   243→  const [saving, setSaving] = useState(false)
   244→
   245→  const eligible = parties.filter((p) =>
   246→    p.active && partyTypes(p).includes(type === 'RECEIPT' ? 'CUSTOMER' : 'SUPPLIER')
   247→  )
   248→
   249→  const invPath = partyId
   250→    ? type === 'RECEIPT'
   251→      ? `/api/sales-invoices?partyId=${partyId}&status=POSTED`
   252→      : `/api/purchase-invoices?partyId=${partyId}&status=POSTED`
   253→    : null
   254→  const invoices = useApi<AnyInvoice[]>(invPath)
   255→
   256→  const openInvoices = useMemo(
   257→    () => (invoices.data ?? []).filter((i) => i.grandTotal - i.paidAmount > 0.004),
   258→    [invoices.data]
   259→  )
   260→
   261→  const amountN = num(amount)
   262→  const allocated = openInvoices.reduce((s, inv) => s + num(allocs[inv.invoiceNo] ?? ''), 0)
   263→  const over = allocated > amountN + 0.004
   264→  const perInvoiceOver = openInvoices.some((inv) => {
   265→    const a = num(allocs[inv.invoiceNo] ?? '')
   266→    return a > inv.grandTotal - inv.paidAmount + 0.004
   267→  })
   268→
   269→  const onTypeChange = (t: 'RECEIPT' | 'PAYMENT') => {
   270→    setType(t)
   271→    setPartyId('')
   272→    setAllocs({})
   273→  }
   274→
   275→  const submit = async () => {
   276→    if (!partyId) {
   277→      toast.error('Select a party')
   278→      return
   279→    }
   280→    if (amountN <= 0) {
   281→      toast.error('Amount must be greater than zero')
   282→      return
   283→    }
   284→    if (over || perInvoiceOver) {
   285→      toast.error('Fix allocations first — total cannot exceed the payment amount, single lines cannot exceed outstanding')
   286→      return
   287→    }
   288→    setSaving(true)
   289→    try {
   290→      const res = await apiPost<Payment>('/api/payments', {
   291→        type,
   292→        partyId,
   293→        date,
   294→        amount: amountN,
   295→        mode,
   296→        refNo: refNo.trim() || undefined,
   297→        notes: notes.trim() || undefined,
   298→        allocations: openInvoices
   299→          .filter((inv) => num(allocs[inv.invoiceNo] ?? '') > 0)
   300→          .map((inv) => ({
   301→            invoiceType: type === 'RECEIPT' ? 'SALES' : 'PURCHASE',
   302→            invoiceNo: inv.invoiceNo,
   303→            amount: num(allocs[inv.invoiceNo]),
   304→          })),
   305→      })
   306→      toast.success(`${res.paymentNo} posted`, {
   307→        description: `${type === 'RECEIPT' ? 'Receipt' : 'Payment'} of ₹ ${money(res.amount)} — GL double-entry committed.`,
   308→      })
   309→      onCreated()
   310→      onOpenChange(false)
   311→      setPartyId('')
   312→      setAmount('')
   313→      setRefNo('')
   314→      setNotes('')
   315→      setAllocs({})
   316→    } catch (e) {
   317→      toastError(e)
   318→    } finally {
   319→      setSaving(false)
   320→    }
   321→  }
   322→
   323→  return (
   324→    <Dialog open={open} onOpenChange={onOpenChange}>
   325→      <DialogContent className="flex max-h-[92vh] w-full max-w-2xl flex-col overflow-hidden">
   326→        <DialogHeader>
   327→          <DialogTitle>New Payment</DialogTitle>
   328→          <DialogDescription>
   329→            Receipts (money in from customers) or payments (money out to suppliers), optionally allocated against
   330→            open invoices.
   331→          </DialogDescription>
   332→        </DialogHeader>
   333→
   334→        <div className="min-h-0 flex-1 space-y-4 overflow-y-auto erp-scroll-y pr-1">
   335→          <div className="grid grid-cols-2 gap-2" role="group" aria-label="Payment type">
   336→            <Button
   337→              type="button"
   338→              variant={type === 'RECEIPT' ? 'default' : 'outline'}
   339→              onClick={() => onTypeChange('RECEIPT')}
   340→              className="h-11 gap-2"
   341→            >
   342→              <ArrowDownCircle className="size-4" aria-hidden /> RECEIPT
   343→            </Button>
   344→            <Button
   345→              type="button"
   346→              variant={type === 'PAYMENT' ? 'default' : 'outline'}
   347→              onClick={() => onTypeChange('PAYMENT')}
   348→              className="h-11 gap-2"
   349→            >
   350→              <ArrowUpCircle className="size-4" aria-hidden /> PAYMENT
   351→            </Button>
   352→          </div>
   353→
   354→          <div className="grid gap-4 sm:grid-cols-2">
   355→            <div className="space-y-1.5">
   356→              <Label>{type === 'RECEIPT' ? 'Customer' : 'Supplier'}</Label>
   357→              <Select value={partyId || undefined} onValueChange={(v) => { setPartyId(v); setAllocs({}) }}>
   358→                <SelectTrigger className="h-11 w-full">
   359→                  <SelectValue placeholder="Select party…" />
   360→                </SelectTrigger>
   361→                <SelectContent>
   362→                  {eligible.map((p) => (
   363→                    <SelectItem key={p.id} value={p.id}>
   364→                      {p.code} — {p.name}
   365→                    </SelectItem>
   366→                  ))}
   367→                </SelectContent>
   368→              </Select>
   369→            </div>
   370→            <div className="space-y-1.5">
   371→              <Label htmlFor="pay-date">Date</Label>
   372→              <Input id="pay-date" type="date" value={date} onChange={(e) => setDate(e.target.value)} className="h-11" />
   373→            </div>
   374→            <div className="space-y-1.5">
   375→              <Label htmlFor="pay-amount">Amount ₹</Label>
   376→              <Input
   377→                id="pay-amount"
   378→                type="number"
   379→                min="0"
   380→                step="any"
   381→                value={amount}
   382→                onChange={(e) => setAmount(e.target.value)}
   383→                placeholder="0"
   384→                className="h-11 text-right tabular-nums"
   385→              />
   386→            </div>
   387→            <div className="space-y-1.5">
   388→              <Label>Mode</Label>
   389→              <Select value={mode} onValueChange={setMode}>
   390→                <SelectTrigger className="h-11 w-full">
   391→                  <SelectValue />
   392→                </SelectTrigger>
   393→                <SelectContent>
   394→                  {PAYMENT_MODES.map((m) => (
   395→                    <SelectItem key={m} value={m}>
   396→                      {m}
   397→                    </SelectItem>
   398→                  ))}
   399→                </SelectContent>
   400→              </Select>
   401→            </div>
   402→            <div className="space-y-1.5">
   403→              <Label htmlFor="pay-ref">Reference No</Label>
   404→              <Input id="pay-ref" value={refNo} onChange={(e) => setRefNo(e.target.value)} placeholder="Cheque / UTR — optional" className="h-11" />
   405→            </div>
   406→            <div className="space-y-1.5">
   407→              <Label htmlFor="pay-notes">Notes</Label>
   408→              <Input id="pay-notes" value={notes} onChange={(e) => setNotes(e.target.value)} placeholder="Optional" className="h-11" />
   409→            </div>
   410→          </div>
   411→
   412→          {partyId ? (
   413→            <div className="space-y-2 rounded-md border p-3">
   414→              <div className="flex items-center justify-between">
   415→                <p className="text-sm font-medium">Allocate against open invoices</p>
   416→                <p className={`text-xs tabular-nums ${over ? 'font-semibold text-destructive' : 'text-muted-foreground'}`}>
   417→                  Allocated ₹ {money(allocated)} / ₹ {money(amountN)}
   418→                </p>
   419→              </div>
   420→              {invoices.loading ? (
   421→                <p className="text-xs text-muted-foreground">Loading open invoices…</p>
   422→              ) : openInvoices.length === 0 ? (
   423→                <p className="text-xs text-muted-foreground">No open invoices for this party.</p>
   424→              ) : (
   425→                <div className="max-h-56 overflow-y-auto erp-scroll-y">
   426→                  <Table>
   427→                    <TableHeader>
   428→                      <TableRow>
   429→                        <TableHead>Invoice</TableHead>
   430→                        <TableHead>Date</TableHead>
   431→                        <TableHead className="text-right">Outstanding</TableHead>
   432→                        <TableHead className="w-32 text-right">Allocate ₹</TableHead>
   433→                      </TableRow>
   434→                    </TableHeader>
   435→                    <TableBody>
   436→                      {openInvoices.map((inv) => {
   437→                        const outstanding = inv.grandTotal - inv.paidAmount
   438→                        const a = num(allocs[inv.invoiceNo] ?? '')
   439→                        const bad = a > outstanding + 0.004
   440→                        return (
   441→                          <TableRow key={inv.id}>
   442→                            <TableCell className="font-medium tabular-nums">{inv.invoiceNo}</TableCell>
   443→                            <TableCell className="whitespace-nowrap text-xs">{dateStr(inv.date)}</TableCell>
   444→                            <TableCell className="text-right tabular-nums">
   445→                              <Money value={outstanding} plain />
   446→                            </TableCell>
   447→                            <TableCell>
   448→                              <Input
   449→                                type="number"
   450→                                min="0"
   451→                                max={outstanding}
   452→                                step="any"
   453→                                value={allocs[inv.invoiceNo] ?? ''}
   454→                                onChange={(e) => setAllocs((s) => ({ ...s, [inv.invoiceNo]: e.target.value }))}
   455→                                placeholder="0"
   456→                                className={`h-11 text-right tabular-nums ${bad ? 'border-destructive focus-visible:ring-destructive' : ''}`}
   457→                                aria-label={`Allocate to ${inv.invoiceNo}`}
   458→                              />
   459→                            </TableCell>
   460→                          </TableRow>
   461→                        )
   462→                      })}
   463→                    </TableBody>
   464→                  </Table>
   465→                </div>
   466→              )}
   467→              {(over || perInvoiceOver) ? (
   468→                <p className="text-xs font-medium text-destructive">
   469→                  Σ allocations cannot exceed the payment amount, and each line cannot exceed its outstanding.
   470→                </p>
   471→              ) : null}
   472→            </div>
   473→          ) : (
   474→            <p className="text-xs text-muted-foreground">Select a party to see open invoices for allocation.</p>
   475→          )}
   476→        </div>
   477→
   478→        <DialogFooter className="gap-2 border-t pt-3">
   479→          <Button variant="outline" onClick={() => onOpenChange(false)} disabled={saving} className="h-11">
   480→            Cancel
   481→          </Button>
   482→          <Button onClick={submit} disabled={saving || over || perInvoiceOver} className="h-11 gap-2">
   483→            {saving ? <Loader2 className="size-4 animate-spin" aria-hidden /> : null}
   484→            {saving ? 'Posting…' : 'Post Payment'}
   485→          </Button>
   486→        </DialogFooter>
   487→      </DialogContent>
   488→    </Dialog>
   489→  )
   490→}

   494→function PartyLedgerTab() {
   495→  const { parties } = useMasters()
   496→  const [partyId, setPartyId] = useState('')
   497→  const ledger = useApi<PartyLedger | null>(partyId ? `/api/reports?type=party-ledger&partyId=${partyId}` : null)
   498→
   499→  const sorted = useMemo(() => [...parties].sort((a, b) => a.name.localeCompare(b.name)), [parties])
   500→  const party: Party | undefined = parties.find((p) => p.id === partyId)
   501→  const data = ledger.data
   502→
   503→  const totalDr = data ? data.rows.reduce((s, r) => s + r.debit, 0) : 0
   504→  const totalCr = data ? data.rows.reduce((s, r) => s + r.credit, 0) : 0
   505→
   506→  return (
   507→    <div className="space-y-4">
   508→      <div className="grid gap-2 sm:max-w-md">
   509→        <Label>Party</Label>
   510→        <Select value={partyId || undefined} onValueChange={setPartyId}>
   511→          <SelectTrigger className="h-11 w-full">
   512→            <SelectValue placeholder="Select party…" />
   513→          </SelectTrigger>
   514→          <SelectContent className="max-h-64">
   515→            {sorted.map((p) => (
   516→              <SelectItem key={p.id} value={p.id}>
   517→                {p.code} — {p.name}
   518→              </SelectItem>
   519→            ))}
   520→          </SelectContent>
   521→        </Select>
   522→      </div>
   523→
   524→      {!partyId ? (
   525→        <EmptyState title="Pick a party" hint="The ledger shows every voucher with a running balance." icon={BookOpen} />
   526→      ) : ledger.loading ? (
   527→        <Card className="p-0"><CardContent className="p-0"><TableSkeleton rows={8} cols={5} /></CardContent></Card>
   528→      ) : !data ? (
   529→        <EmptyState title="No ledger data" hint={ledger.error ?? 'Nothing to show.'} icon={BookOpen} />
   530→      ) : (
   531→        <>
   532→          <div className="grid gap-4 sm:grid-cols-4">
   533→            <Card className="p-4">
   534→              <p className="text-xs font-medium uppercase tracking-wide text-muted-foreground">Opening</p>
   535→              <p className="mt-1 text-lg font-semibold tabular-nums">₹ {money(data.openingBal)}</p>
