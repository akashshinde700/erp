    51→    51→export function AccountingView() {
    52→    52→  const [tab, setTab] = useState<'payments' | 'ledger'>('payments')

    54→    54→  const [ledgerPartyId, setLedgerPartyId] = useState('')
    55→    55→
    56→    56→  // Cross-module presets: "Receive Payment" from an invoice detail opens the
    57→    57→  // payment dialog prefilled; "Party ledger" jumps straight to the ledger tab.
    58→    58→  useEffect(() => {
    59→    59→    const apply = (p: CreatePreset) => {
    60→    60→      if (p.type === 'payment') {
    61→    61→        setTab('payments')
    62→    62→        setPaymentPreset({
    63→    63→          partyId: p.partyId,
    64→    64→          paymentType: p.paymentType,
    65→    65→          amount: p.amount,
    66→    66→          allocations: p.allocations,
    67→    67→          stamp: Date.now(),
    68→    68→        })
    69→    69→      } else if (p.type === 'party-ledger') {
    70→    70→        setTab('ledger')
    71→    71→        setLedgerPartyId(p.partyId ?? '')
    72→    72→      }
    73→    73→    }

   275→   275→function NewPaymentDialog({
   276→   276→  open,
   277→   277→  onOpenChange,
   278→   278→  onCreated,
   279→   279→  preset,
   280→   280→}: {
   281→   281→  open: boolean
   282→   282→  onOpenChange: (o: boolean) => void
   283→   283→  onCreated: () => void
   284→   284→  preset?: PaymentPreset | null
   285→   285→}) {
   286→   286→  const { parties } = useMasters()
   287→   287→  const [type, setType] = useState<'RECEIPT' | 'PAYMENT'>('RECEIPT')
   288→   288→  const [partyId, setPartyId] = useState('')
   289→   289→  const [date, setDate] = useState(todayISO())
   290→   290→  const [amount, setAmount] = useState('')
   291→   291→  const [mode, setMode] = useState('BANK')
   292→   292→  const [refNo, setRefNo] = useState('')
   293→   293→  const [notes, setNotes] = useState('')
   294→   294→  const [allocs, setAllocs] = useState<Record<string, string>>({})
   295→   295→  const [saving, setSaving] = useState(false)
   296→   296→
   297→   297→  const eligible = parties.filter((p) =>
   298→   298→    p.active && partyTypes(p).includes(type === 'RECEIPT' ? 'CUSTOMER' : 'SUPPLIER')
   299→   299→  )
   300→   300→
   301→   301→  const invPath = partyId
   302→   302→    ? type === 'RECEIPT'
   303→   303→      ? `/api/sales-invoices?partyId=${partyId}&status=POSTED`
   304→   304→      : `/api/purchase-invoices?partyId=${partyId}&status=POSTED`
   305→   305→    : null
   306→   306→  const invoices = useApi<AnyInvoice[]>(invPath)
   307→   307→
   308→   308→  const openInvoices = useMemo(
   309→   309→    () => (invoices.data ?? []).filter((i) => i.grandTotal - i.paidAmount > 0.004),
   310→   310→    [invoices.data]
   311→   311→  )
   312→   312→
   313→   313→  const amountN = num(amount)
   314→   314→  const allocated = openInvoices.reduce((s, inv) => s + num(allocs[inv.invoiceNo] ?? ''), 0)
   315→   315→  const over = allocated > amountN + 0.004
   316→   316→  const perInvoiceOver = openInvoices.some((inv) => {
   317→   317→    const a = num(allocs[inv.invoiceNo] ?? '')
   318→   318→    return a > inv.grandTotal - inv.paidAmount + 0.004
   319→   319→  })
   320→   320→
   321→   321→  const onTypeChange = (t: 'RECEIPT' | 'PAYMENT') => {
   322→   322→    setType(t)
   323→   323→    setPartyId('')
   324→   324→    setAllocs({})
   325→   325→  }
   326→   326→
   327→   327→  // Preset from an invoice's "Receive/Make Payment" button: party + type +
   328→   328→  // amount + allocation pre-ticked for that invoice.
   329→   329→  useEffect(() => {
   330→   330→    if (!open || !preset) return
   331→   331→    if (preset.paymentType) {
   332→   332→      setType(preset.paymentType)
   333→   333→    }
   334→   334→    if (preset.partyId) {
   335→   335→      setPartyId(preset.partyId)
   336→   336→    }
   337→   337→    if (preset.amount !== undefined) {
   338→   338→      setAmount(String(preset.amount))
   339→   339→    }
   340→   340→    if (preset.allocations?.length) {
   341→   341→      const a: Record<string, string> = {}
   342→   342→      for (const alloc of preset.allocations) a[alloc.invoiceNo] = String(alloc.amount)
   343→   343→      setAllocs(a)
   344→   344→    }
   345→   345→  }, [open, preset])
   346→   346→
   347→   347→  const submit = async () => {
   348→   348→    if (!partyId) {
   349→   349→      toast.error('Select a party')
   350→   350→      return
   351→   351→    }
   352→   352→    if (amountN <= 0) {
   353→   353→      toast.error('Amount must be greater than zero')
   354→   354→      return
   355→   355→    }
   356→   356→    if (over || perInvoiceOver) {
   357→   357→      toast.error('Fix allocations first — total cannot exceed the payment amount, single lines cannot exceed outstanding')
   358→   358→      return
   359→   359→    }
   360→   360→    setSaving(true)
   361→   361→    try {
   362→   362→      const res = await apiPost<Payment>('/api/payments', {
   363→   363→        type,
   364→   364→        partyId,
   365→   365→        date,
   366→   366→        amount: amountN,
   367→   367→        mode,
   368→   368→        refNo: refNo.trim() || undefined,
   369→   369→        notes: notes.trim() || undefined,
   370→   370→        allocations: openInvoices
   371→   371→          .filter((inv) => num(allocs[inv.invoiceNo] ?? '') > 0)
   372→   372→          .map((inv) => ({
   373→   373→            invoiceType: type === 'RECEIPT' ? 'SALES' : 'PURCHASE',
   374→   374→            invoiceNo: inv.invoiceNo,
   375→   375→            amount: num(allocs[inv.invoiceNo]),
   376→   376→          })),
   377→   377→      })
   378→   378→      toast.success(`${res.paymentNo} posted`, {
   379→   379→        description: `${type === 'RECEIPT' ? 'Receipt' : 'Payment'} of ₹ ${money(res.amount)} — GL double-entry committed.`,
   380→   380→      })
   381→   381→      onCreated()
   382→   382→      onOpenChange(false)
   383→   383→      setPartyId('')
   384→   384→      setAmount('')
   385→   385→      setRefNo('')
   386→   386→      setNotes('')
   387→   387→      setAllocs({})
   388→   388→    } catch (e) {
   389→   389→      toastError(e)
   390→   390→    } finally {
   391→   391→      setSaving(false)
   392→   392→    }
   393→   393→  }
   394→   394→
   395→   395→  return (
   396→   396→    <Dialog open={open} onOpenChange={onOpenChange}>
   397→   397→      <DialogContent className="flex max-h-[92vh] w-full max-w-2xl flex-col overflow-hidden">
   398→   398→        <DialogHeader>
   399→   399→          <DialogTitle>New Payment</DialogTitle>
   400→   400→          <DialogDescription>
   401→   401→            Receipts (money in from customers) or payments (money out to suppliers), optionally allocated against
   402→   402→            open invoices.
   403→   403→          </DialogDescription>
   404→   404→        </DialogHeader>
   405→   405→
   406→   406→        <div className="min-h-0 flex-1 space-y-4 overflow-y-auto erp-scroll-y pr-1">
   407→   407→          <div className="grid grid-cols-2 gap-2" role="group" aria-label="Payment type">
   408→   408→            <Button
   409→   409→              type="button"
   410→   410→              variant={type === 'RECEIPT' ? 'default' : 'outline'}
   411→   411→              onClick={() => onTypeChange('RECEIPT')}
   412→   412→              className="h-11 gap-2"
   413→   413→            >
   414→   414→              <ArrowDownCircle className="size-4" aria-hidden /> RECEIPT
   415→   415→            </Button>
   416→   416→            <Button
   417→   417→              type="button"
   418→   418→              variant={type === 'PAYMENT' ? 'default' : 'outline'}
   419→   419→              onClick={() => onTypeChange('PAYMENT')}
   420→   420→              className="h-11 gap-2"
   421→   421→            >
   422→   422→              <ArrowUpCircle className="size-4" aria-hidden /> PAYMENT
   423→   423→            </Button>
   424→   424→          </div>
   425→   425→
   426→   426→          <div className="grid gap-4 sm:grid-cols-2">
   427→   427→            <div className="space-y-1.5">
   428→   428→              <Label>{type === 'RECEIPT' ? 'Customer' : 'Supplier'}</Label>
   429→   429→              <Select value={partyId} onValueChange={(v) => { setPartyId(v); setAllocs({}) }}>
   430→   430→                <SelectTrigger className="h-11 w-full">
   431→   431→                  <SelectValue placeholder="Select party…" />
   432→   432→                </SelectTrigger>
   433→   433→                <SelectContent>
   434→   434→                  {eligible.map((p) => (
   435→   435→                    <SelectItem key={p.id} value={p.id}>
   436→   436→                      {p.code} — {p.name}
   437→   437→                    </SelectItem>
   438→   438→                  ))}
   439→   439→                </SelectContent>
   440→   440→              </Select>
   441→   441→            </div>
   442→   442→            <div className="space-y-1.5">
   443→   443→              <Label htmlFor="pay-date">Date</Label>
   444→   444→              <Input id="pay-date" type="date" value={date} onChange={(e) => setDate(e.target.value)} className="h-11" />
   445→   445→            </div>
   446→   446→            <div className="space-y-1.5">
   447→   447→              <Label htmlFor="pay-amount">Amount ₹</Label>
   448→   448→              <Input
   449→   449→                id="pay-amount"
   450→   450→                type="number"
   451→   451→                min="0"
   452→   452→                step="any"
   453→   453→                value={amount}
   454→   454→                onChange={(e) => setAmount(e.target.value)}
   455→   455→                placeholder="0"
   456→   456→                className="h-11 text-right tabular-nums"
   457→   457→              />
   458→   458→            </div>
   459→   459→            <div className="space-y-1.5">
   460→   460→              <Label>Mode</Label>
