   114→export function PlatformView({ onExit }: { onExit: () => void }) {
   115→  const [key, setKey] = useState('')
   116→  const [authed, setAuthed] = useState(false)
   117→  const [checking, setChecking] = useState(false)
   118→
   119→  useEffect(() => {
   120→    const saved = sessionStorage.getItem(KEY_STORE)
   121→    if (saved) {
   122→      setKey(saved)
   123→      void verify(saved)
   124→    }
   125→  }, [])
   126→
   127→  const verify = async (k: string) => {
   128→    setChecking(true)
   129→    try {
   130→      await api('/api/platform/auth', { method: 'POST', body: JSON.stringify({ key: k }) })
   131→      sessionStorage.setItem(KEY_STORE, k)
   132→      setAuthed(true)
   133→    } catch (e) {
   134→      sessionStorage.removeItem(KEY_STORE)
   135→      toastError(e)
   136→    } finally {
   137→      setChecking(false)
   138→    }
   139→  }
   140→
   141→  if (!authed) {
   142→    return (
   143→      <div className="flex min-h-dvh flex-col items-center justify-center bg-muted/30 p-4 pb-[max(1.5rem,env(safe-area-inset-bottom))]">
   144→        <div className="w-full max-w-md space-y-6">
   145→          <div className="space-y-3 text-center">
   146→            <span className="mx-auto flex size-14 items-center justify-center rounded-2xl bg-primary text-primary-foreground">
   147→              <KeyRound className="size-7" aria-hidden />
   148→            </span>
   149→            <div className="space-y-1">
   150→              <h1 className="text-2xl font-bold tracking-tight">Platform Admin</h1>
   151→              <p className="text-sm text-muted-foreground">Plans, organizations, subscriptions, inquiries — the control plane.</p>
   152→            </div>
   153→          </div>
   154→          <Card className="p-0">
   155→            <CardContent className="space-y-4 p-6">
   156→              <div className="space-y-1.5">
   157→                <Label htmlFor="platform-key">Platform admin key</Label>
   158→                <Input
   159→                  id="platform-key"
   160→                  type="password"
   161→                  value={key}
   162→                  onChange={(e) => setKey(e.target.value)}
   163→                  onKeyDown={(e) => {
   164→                    if (e.key === 'Enter' && key) void verify(key)
   165→                  }}
   166→                  placeholder="forge-admin"
   167→                  className="h-11"
   168→                  autoFocus
   169→                />
   170→                <p className="text-xs text-muted-foreground">
   171→                  Default is <span className="font-mono">forge-admin</span> — change it with the{' '}
   172→                  <span className="font-mono">PLATFORM_ADMIN_KEY</span> environment variable.
   173→                </p>
   174→              </div>
   175→              <Button onClick={() => void verify(key)} disabled={!key || checking} className="h-11 w-full gap-2">
   176→                {checking ? <Loader2 className="size-4 animate-spin" aria-hidden /> : <KeyRound className="size-4" aria-hidden />}
   177→                Unlock console
   178→              </Button>
   179→            </CardContent>
   180→          </Card>
   181→          <Button variant="outline" onClick={onExit} className="h-11 w-full gap-2">
   182→            <ArrowLeft className="size-4" aria-hidden /> Back to ERP
   183→          </Button>
   184→        </div>
   185→      </div>
   186→    )
   187→  }
   188→
   189→  return <Console key={key} apiKey={key} onAuthLost={() => setAuthed(false)} onExit={onExit} />
   190→}

   196→function Console({ apiKey, onAuthLost, onExit }: { apiKey: string; onAuthLost: () => void; onExit: () => void }) {
   197→  const [tab, setTab] = useState<ConsoleTab>('orgs')
   198→  const [data, setData] = useState<OrgsResponse | null>(null)
   199→  const [plans, setPlans] = useState<Plan[] | null>(null)
   200→  const [inquiries, setInquiries] = useState<InquiriesResponse | null>(null)
   201→  const [posts, setPosts] = useState<BlogPost[] | null>(null)
   202→  const [loading, setLoading] = useState(true)
   203→
   204→  const reload = useCallback(async () => {
   205→    setLoading(true)
   206→    try {
   207→      const [orgs, pl, inq, blog] = await Promise.all([
   208→        platformApi<OrgsResponse>('/api/platform/orgs', undefined, apiKey),
   209→        platformApi<Plan[]>('/api/platform/plans', undefined, apiKey),
   210→        platformApi<InquiriesResponse>('/api/platform/inquiries', undefined, apiKey),
   211→        platformApi<{ posts: BlogPost[] }>('/api/platform/blog', undefined, apiKey),
   212→      ])
   213→      setData(orgs)
   214→      setPlans(pl)
   215→      setInquiries(inq)
   216→      setPosts(blog.posts)
   217→    } catch (e) {
   218→      toastError(e)
   219→      onAuthLost()
   220→    } finally {
   221→      setLoading(false)
   222→    }
   223→  }, [apiKey, onAuthLost])
   224→
   225→  useEffect(() => {
   226→    void reload()
   227→  }, [reload])
   228→
   229→  const orgAction = async (id: string, body: Record<string, unknown>, okMsg: string) => {
   230→    try {
   231→      await platformApi(`/api/platform/orgs/${id}`, { method: 'PATCH', body: JSON.stringify(body) }, apiKey)
   232→      toast.success(okMsg)
   233→      void reload()
   234→    } catch (e) {
   235→      toastError(e)
   236→    }
   237→  }
   238→
   239→  const logout = () => {
   240→    sessionStorage.removeItem(KEY_STORE)
   241→    onAuthLost()
   242→  }
   243→
   244→  const pill = (t: ConsoleTab, label: React.ReactNode, count?: number) => (
   245→    <button
   246→      type="button"
   247→      onClick={() => setTab(t)}
   248→      aria-current={tab === t ? 'page' : undefined}
   249→      className={cn(
   250→        'inline-flex h-10 min-w-11 items-center justify-center gap-1.5 whitespace-nowrap rounded-full border px-4 text-sm font-medium transition-colors',
   251→        tab === t
   252→          ? 'border-primary bg-primary text-primary-foreground'
   253→          : 'border-border bg-card text-foreground hover:bg-accent'
   254→      )}
   255→    >
   256→      {label}
   257→      {typeof count === 'number' ? <span className="tabular-nums opacity-70">({count})</span> : null}
   258→    </button>
   259→  )
   260→
   261→  const newCount = inquiries?.newCount ?? 0
   262→
   263→  return (
   264→    <div className="flex min-h-dvh flex-col bg-muted/30">
   265→      <header className="border-b bg-card">
   266→        <div className="mx-auto flex w-full max-w-6xl flex-col gap-3 px-4 py-4 pb-[max(1rem,env(safe-area-inset-top))] sm:flex-row sm:items-center sm:justify-between">
   267→          <div className="min-w-0 space-y-0.5">
   268→            <h1 className="truncate text-xl font-bold tracking-tight">Platform Admin</h1>
   269→            <p className="truncate text-sm text-muted-foreground">
   270→              <span className="font-mono">{apiKey}</span> · ForgeERP control plane
   271→            </p>
   272→          </div>
   273→          <nav aria-label="Platform Admin sections" className="flex flex-wrap items-center gap-2">
   274→            {pill('orgs', 'Organizations', data?.orgs.length)}
   275→            {pill('plans', 'Plans', plans?.length)}
   276→            {pill('inquiries', 'Inquiries', newCount > 0 ? newCount : undefined)}
   277→            {pill('blog', 'Blog', posts?.length)}
   278→            <span className="mx-0.5 hidden h-6 w-px bg-border sm:block" aria-hidden />
   279→            <Button variant="outline" onClick={onExit} className="h-10 gap-2 rounded-full px-4">
   280→              <ArrowLeft className="size-4" aria-hidden /> Back to ERP
   281→            </Button>
   282→            <Button variant="outline" onClick={logout} className="h-10 gap-2 rounded-full px-4 text-destructive hover:text-destructive">
   283→              <LogOut className="size-4" aria-hidden /> Logout
   284→            </Button>
   285→          </nav>
   286→        </div>
   287→      </header>
   288→
   289→      <main className="mx-auto w-full max-w-6xl flex-1 px-4 py-6">
   290→        {loading && !data ? (
   291→          <TableSkeleton rows={5} cols={5} />
   292→        ) : tab === 'orgs' ? (
   293→          <OrgsTab data={data} plans={plans ?? []} apiKey={apiKey} onAction={orgAction} reload={reload} />
   294→        ) : tab === 'plans' ? (
   295→          <PlansTab plans={plans ?? []} apiKey={apiKey} reload={reload} />
   296→        ) : tab === 'inquiries' ? (
   297→          <InquiriesTab inquiries={inquiries} apiKey={apiKey} reload={reload} />
   298→        ) : (
   299→          <BlogTab posts={posts ?? []} apiKey={apiKey} reload={reload} />
   300→        )}
   301→      </main>
   302→
   303→      <footer className="mt-auto border-t bg-card pb-[max(0.75rem,env(safe-area-inset-bottom))]">
   304→        <div className="mx-auto flex max-w-6xl flex-col items-center justify-between gap-1 px-4 py-3 text-center text-xs text-muted-foreground sm:flex-row sm:text-left">
   305→          <span>
   306→            <span className="font-medium text-foreground">ForgeERP</span> Platform — plans, subscriptions &amp; enforcement
   307→          </span>
   308→          <span>Single-tenant runtime · the ERP operates the first organization</span>
   309→        </div>
   310→      </footer>
   311→    </div>
   312→  )
   313→}

   319→function OrgStatusBadge({ o }: { o: OrgRow }) {
   320→  const now = Date.now()
   321→  const paidOk = o.paidUntil ? new Date(o.paidUntil).getTime() > now : false
   322→  const trialOk = o.trialEndsAt ? new Date(o.trialEndsAt).getTime() > now : false
   323→  if (o.saasStatus === 'SUSPENDED')
   324→    return (
   325→      <Badge variant="destructive" className="font-normal">
   326→        suspended
   327→      </Badge>
   328→    )
   329→  if (paidOk)
   330→    return (
   331→      <Badge className="border-transparent bg-emerald-100 font-normal text-emerald-700 dark:bg-emerald-500/15 dark:text-emerald-400">
   332→        active
   333→      </Badge>
   334→    )
   335→  if (trialOk)
   336→    return (
   337→      <Badge className="border-transparent bg-amber-100 font-normal text-amber-700 dark:bg-amber-500/15 dark:text-amber-400">
   338→        trial
   339→      </Badge>
   340→    )
   341→  if (o.paidUntil || o.trialEndsAt)
   342→    return (
   343→      <Badge variant="destructive" className="font-normal">
   344→        expired
   345→      </Badge>
   346→    )
   347→  return (
   348→    <Badge variant="secondary" className="font-normal">
   349→      free · unlimited
   350→    </Badge>
   351→  )
   352→}

   354→function OrgsTab({
   355→  data,
   356→  plans,
   357→  apiKey,
   358→  onAction,
   359→  reload,

   376→    try {
   377→      await platformApi(
   378→        '/api/platform/orgs',
   379→        {
   380→          method: 'POST',
   381→          body: JSON.stringify({
   382→            name: form.name,
   383→            gstin: form.gstin || undefined,
   384→            city: form.city || undefined,

   386→            planId: form.planId === 'none' ? null : form.planId,
   387→            trialDays: Number(form.trialDays) || 0,
   388→          }),
   389→        },
   390→        apiKey

   392→      toast.success(`Organization "${form.name}" created`)

   434→            maxParties: custom.parties === '' ? null : Math.max(0, Math.trunc(Number(custom.parties) || 0)),

   514→              {stats.active} active · {stats.suspended} suspended · MRR ₹{money(stats.mrr)}

   519→          <Plus className="size-4" aria-hidden /> Create organization
   520→        </Button>
   521→      </div>
   522→
   523→      {orgs.length === 0 ? (
   524→        <Card className="p-0">
   525→          <CardContent className="p-6">
   526→            <EmptyState
   527→              title="No organizations"
   528→              hint="Create the first organization — it becomes a real Company row you can manage here."
   529→              action={
   530→                <Button onClick={() => setCreateOpen(true)} className="mt-2 h-11 gap-2">
   531→                  <Plus className="size-4" aria-hidden /> New organization
   532→                </Button>
   533→              }
   534→            />
   535→          </CardContent>
   536→        </Card>
   537→      ) : (
   538→        <Card className="p-0">
   539→          <CardContent className="p-0">
   540→            <div className="max-h-[64dvh] overflow-auto erp-scroll [&_[data-slot=table-container]]:overflow-visible">
   541→              <Table>
   542→                <TableHeader className="sticky top-0 z-10 bg-card">
   543→                  <TableRow>
   544→                    <TableHead>Organization</TableHead>
   545→                    <TableHead className="hidden lg:table-cell">Contact</TableHead>
   546→                    <TableHead>Plan</TableHead>
   547→                    <TableHead>Status</TableHead>

   930→
   931→  const openNew = () => {
   932→    setEditing(null)

   934→    setFeatures(PLAN_FEATURE_CATALOG.map((f) => f.key))

   995→      <div className="flex flex-wrap items-center justify-between gap-3">

  1003→        <Card className="p-0">
  1004→          <CardContent className="p-6">
  1005→            <EmptyState
  1006→              title="No plans yet"

  1008→              action={
  1009→                <Button onClick={openNew} className="mt-2 h-11 gap-2">
  1010→                  <Plus className="size-4" aria-hidden /> Add plan

  1034→                      <AlertDialog>
  1035→                        <AlertDialogTrigger asChild>
  1036→                          <Button
  1037→                            size="sm"
  1038→                            variant="outline"

  1588→function BlogTab({ posts, apiKey, reload }: { posts: BlogPost[]; apiKey: string; reload: () => Promise<void> }) {

  1590→  const [editing, setEditing] = useState<BlogPost | null>(null)
  1591→  const [form, setForm] = useState({ title: '', slug: '', excerpt: '', body: '', status: 'draft' })
  1592→  const [slugTouched, setSlugTouched] = useState(false)
  1593→  const [saving, setSaving] = useState(false)
  1594→
  1595→  const openNew = () => {
  1596→    setEditing(null)
  1597→    setForm({ title: '', slug: '', excerpt: '', body: '', status: 'draft' })
  1598→    setSlugTouched(false)
  1599→    setOpen(true)
  1600→  }
  1601→
  1602→  const openEdit = (p: BlogPost) => {
  1603→    setEditing(p)
  1604→    setForm({
  1605→      title: p.title,
  1606→      slug: p.slug,
  1607→      excerpt: p.excerpt ?? '',
  1608→      body: p.body ?? '',
  1609→      status: p.status === 'published' ? 'published' : 'draft',
  1610→    })
  1611→    setSlugTouched(true)
  1612→    setOpen(true)
  1613→  }
  1614→
  1615→  const save = async () => {
  1616→    setSaving(true)
  1617→    try {
  1618→      const body = {
  1619→        title: form.title.trim(),
  1620→        slug: form.slug.trim() ? slugify(form.slug) : undefined,
  1621→        excerpt: form.excerpt.trim() || undefined,
  1622→        body: form.body.trim() || undefined,
  1623→        status: form.status,
  1624→      }
  1625→      if (editing) await platformApi(`/api/platform/blog/${editing.id}`, { method: 'PATCH', body: JSON.stringify(body) }, apiKey)
  1626→      else await platformApi('/api/platform/blog', { method: 'POST', body: JSON.stringify(body) }, apiKey)
  1627→      toast.success(editing ? 'Post updated' : 'Post created')
  1628→      setOpen(false)
  1629→      void reload()
  1630→    } catch (e) {
  1631→      toastError(e)
  1632→    } finally {
  1633→      setSaving(false)
  1634→    }
  1635→  }
  1636→
  1637→  const del = async (p: BlogPost) => {
  1638→    try {
  1639→      await platformApi(`/api/platform/blog/${p.id}`, { method: 'DELETE' }, apiKey)
  1640→      toast.success(`"${p.title}" deleted`)

  1643→      toastError(e)
  1644→    }
  1645→  }

  1653→        </Button>

  1655→
  1656→      {posts.length === 0 ? (

  1658→          <CardContent className="p-6">

  1663→                <Button onClick={openNew} className="mt-2 h-11 gap-2">
  1664→                  <Plus className="size-4" aria-hidden /> New post
  1665→                </Button>
  1666→              }
  1667→            />
  1668→          </CardContent>
  1669→        </Card>
  1670→      ) : (
  1671→        <div className="space-y-3">
  1672→          {posts.map((p) => (
  1673→            <Card key={p.id} className="p-0">
  1674→              <CardContent className="flex flex-wrap items-start justify-between gap-3 p-4">

  1679→                      <Badge className="border-transparent bg-emerald-100 font-normal text-emerald-700 dark:bg-emerald-500/15 dark:text-emerald-400">
  1680→                        published
  1681→                      </Badge>
  1682→                    ) : (
  1683→                      <Badge variant="secondary" className="font-normal">
  1684→                        draft
  1685→                      </Badge>
  1686→                    )}
  1687→                  </div>

  1690→                  <p className="text-xs text-muted-foreground/80">

  1696→                    <Pencil className="size-3.5" aria-hidden /> Edit

  1719→                          Delete

  1825→function BackupCard() {

  1833→      const res = await api<{ restored: boolean }>('/api/backup/restore', {
  1834→        method: 'POST',
  1835→        body: JSON.stringify({ confirm: 'RESTORE', snapshot }),

  1837→      if (res.restored) {
  1838→        toast.success('Snapshot restored', { description: 'Reload the page to see the restored data.' })
  1839→      }
  1840→    } catch (e) {
  1841→      toastError(e)

  1843→      setRestoring(false)
  1844→      setConfirmText('')
