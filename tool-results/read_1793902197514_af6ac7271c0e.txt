    58→const KEY_STORE = 'forge_platform_key'
    59→
    60→export function PlatformView() {
    61→  const [key, setKey] = useState('')
    62→  const [authed, setAuthed] = useState(false)
    63→  const [checking, setChecking] = useState(false)
    64→
    65→  useEffect(() => {
    66→    const saved = sessionStorage.getItem(KEY_STORE)
    67→    if (saved) {
    68→      setKey(saved)
    69→      void verify(saved)
    70→    }
    71→  }, [])
    72→
    73→  const verify = async (k: string) => {
    74→    setChecking(true)
    75→    try {
    76→      await api('/api/platform/auth', { method: 'POST', body: JSON.stringify({ key: k }) })
    77→      sessionStorage.setItem(KEY_STORE, k)
    78→      setAuthed(true)
    79→    } catch (e) {
    80→      sessionStorage.removeItem(KEY_STORE)
    81→      toastError(e)
    82→    } finally {
    83→      setChecking(false)
    84→    }
    85→  }
    86→
    87→  if (!authed) {
    88→    return (
    89→      <div className="mx-auto max-w-md space-y-4 py-10">
    90→        <SectionHeading title="Platform Admin" description="SaaS control plane — plans, organizations, subscriptions, backups." />
    91→        <Card className="p-0">
    92→          <CardContent className="space-y-4 p-6">
    93→            <div className="flex size-12 items-center justify-center rounded-xl bg-primary text-primary-foreground">
    94→              <KeyRound className="size-6" aria-hidden />
    95→            </div>
    96→            <div className="space-y-1.5">
    97→              <Label htmlFor="platform-key">Platform admin key</Label>
    98→              <Input
    99→                id="platform-key"
   100→                type="password"
   101→                value={key}
   102→                onChange={(e) => setKey(e.target.value)}
   103→                onKeyDown={(e) => {
   104→                  if (e.key === 'Enter' && key) void verify(key)
   105→                }}
   106→                placeholder="forge-admin"
   107→                className="h-11"
   108→              />
   109→              <p className="text-xs text-muted-foreground">
   110→                Default is <span className="font-mono">forge-admin</span> — change it with the <span className="font-mono">PLATFORM_ADMIN_KEY</span> environment variable.
   111→              </p>
   112→            </div>
   113→            <Button onClick={() => void verify(key)} disabled={!key || checking} className="h-11 w-full gap-2">
   114→              {checking ? <Loader2 className="size-4 animate-spin" aria-hidden /> : <KeyRound className="size-4" aria-hidden />}
   115→              Unlock console
   116→            </Button>
   117→          </CardContent>
   118→        </Card>
   119→      </div>
   120→    )
   121→  }
   122→
   123→  return <Console key={key} apiKey={key} onAuthLost={() => setAuthed(false)} />
   124→}

   128→function Console({ apiKey, onAuthLost }: { apiKey: string; onAuthLost: () => void }) {
   129→  const [tab, setTab] = useState<'orgs' | 'plans'>('orgs')
   130→  const [data, setData] = useState<OrgsResponse | null>(null)
   131→  const [plans, setPlans] = useState<Plan[] | null>(null)
   132→  const [loading, setLoading] = useState(true)
   133→
   134→  const reload = useCallback(async () => {
   135→    setLoading(true)
   136→    try {
   137→      const [orgs, pl] = await Promise.all([
   138→        platformApi<OrgsResponse>('/api/platform/orgs', undefined, apiKey),
   139→        platformApi<Plan[]>('/api/platform/plans', undefined, apiKey),
   140→      ])
   141→      setData(orgs)
   142→      setPlans(pl)
   143→    } catch (e) {
   144→      toastError(e)
   145→      onAuthLost()
   146→    } finally {
   147→      setLoading(false)
   148→    }
   149→  }, [apiKey, onAuthLost])
   150→
   151→  useEffect(() => {
   152→    void reload()
   153→  }, [reload])
   154→
   155→  const orgAction = async (id: string, body: Record<string, unknown>, okMsg: string) => {
   156→    try {
   157→      await platformApi(`/api/platform/orgs/${id}`, { method: 'PATCH', body: JSON.stringify(body) }, apiKey)
   158→      toast.success(okMsg)
   159→      void reload()
   160→    } catch (e) {
   161→      toastError(e)
   162→    }
   163→  }
   164→
   165→  const stats = data?.stats
   166→
   167→  return (
   168→    <div className="space-y-6">
   169→      <SectionHeading
   170→        title="Platform Admin"
   171→        description="Plans, organizations, subscriptions, backups. The ERP operates the first organization; suspension/expiry blocks its business mutations for real."
   172→        actions={
   173→          <Button variant="outline" onClick={() => void reload()} className="h-11 gap-2">
   174→            <RefreshCw className="size-4" aria-hidden /> Refresh
   175→          </Button>
   176→        }
   177→      />
   178→
   179→      {loading && !data ? (
   180→        <TableSkeleton rows={4} cols={4} />
   181→      ) : stats ? (
   182→        <div className="grid gap-4 sm:grid-cols-2 lg:grid-cols-4">
   183→          <KpiCard title="Organizations" value={String(stats.total)} icon={Building2} hint={`${stats.active} active · ${stats.suspended} suspended`} />
   184→          <KpiCard title="MRR" value={`₹ ${money(stats.mrr)}`} icon={IndianRupee} tone="positive" hint="Active orgs × monthly plan price" />
   185→          <KpiCard title="Documents" value={String(stats.salesInvoices)} icon={DatabaseBackup} hint={`${stats.parties} parties · ${stats.items} items`} />
   186→          <KpiCard title="Warehouses" value={String(stats.warehouses)} icon={Building2} hint="Across the operating org" />
   187→        </div>
   188→      ) : null}
   189→
   190→      <div className="flex gap-2">
   191→        <Button variant={tab === 'orgs' ? 'default' : 'outline'} onClick={() => setTab('orgs')} className="h-11">
   192→          Organizations ({data?.orgs.length ?? 0})
   193→        </Button>
   194→        <Button variant={tab === 'plans' ? 'default' : 'outline'} onClick={() => setTab('plans')} className="h-11">
   195→          Plans ({plans?.length ?? 0})
   196→        </Button>
   197→      </div>
   198→
   199→      {tab === 'orgs' ? <OrgsTable data={data} plans={plans ?? []} apiKey={apiKey} onAction={orgAction} reload={reload} /> : null}
   200→      {tab === 'plans' ? <PlansTable plans={plans ?? []} apiKey={apiKey} reload={reload} /> : null}
   201→
   202→      <BackupCard />
   203→    </div>
   204→  )
   205→}

   213→  onAction,
   214→  reload,

   216→  data: OrgsResponse | null
   217→  plans: Plan[]
   218→  apiKey: string

   226→  const [trialDays, setTrialDays] = useState('14')

   231→    try {
   232→      await platformApi(
   233→        '/api/platform/orgs',
   234→        {
   235→          method: 'POST',
   236→          body: JSON.stringify({
   237→            name,
   238→            gstin: gstin || undefined,
   239→            planId: planId === 'none' ? null : planId,

   242→        },
   243→        apiKey

   247→      setName('')
   248→      setGstin('')

   251→      toastError(e)
   252→    } finally {
   253→      setCreating(false)
   254→    }
   255→  }
   256→
   257→  const del = async (id: string, orgName: string) => {
   258→    try {
   259→      await platformApi(`/api/platform/orgs/${id}?confirmName=${encodeURIComponent(orgName)}`, { method: 'DELETE' }, apiKey)
   260→      toast.success(`${orgName} deleted`)
   261→      void reload()
   262→    } catch (e) {
   263→      toastError(e)
   264→    }
   265→  }
   266→
   267→  if (!data || data.orgs.length === 0) {
   268→    return (
   269→      <Card className="p-0">
   270→        <CardContent className="p-6">
   271→          <EmptyState
   272→            title="No organizations"
   273→            hint="Create the first organization — it becomes a real Company row you can manage here."
   274→            action={
   275→              <Button onClick={() => setCreateOpen(true)} className="mt-2 h-11 gap-2">
   276→                <Plus className="size-4" aria-hidden /> New organization

   278→            }
   279→          />

   281→      </Card>
   282→    )

   288→    const trialOk = o.trialEndsAt ? new Date(o.trialEndsAt).getTime() > now : false
   289→    if (o.saasStatus === 'SUSPENDED') return <Badge variant="destructive">Suspended</Badge>
   290→    if (paidOk) return <Badge variant="default">Paid</Badge>
   291→    if (trialOk) return <Badge variant="outline" className="border-amber-400 text-amber-600 dark:text-amber-400">Trial</Badge>
   292→    if (o.paidUntil || o.trialEndsAt) return <Badge variant="destructive">Expired</Badge>

   298→      <Card className="p-0">

   301→            <p className="text-sm font-semibold">Organizations</p>
   302→            <Button size="sm" className="h-10 gap-2" onClick={() => setCreateOpen(true)}>

   308→              <TableHeader className="sticky top-0 z-10 bg-card">

   312→                  <TableHead>Status</TableHead>

   315→                  <TableHead className="text-right">Actions</TableHead>
   316→                </TableRow>
   317→              </TableHeader>

   319→                {data.orgs.map((o) => (
   320→                  <TableRow key={o.id}>
   321→                    <TableCell>
   322→                      <span className="block font-medium">{o.name}</span>
   323→                      <span className="block text-xs text-muted-foreground">
   324→                        {o.gstin ?? 'No GSTIN'}
   325→                        {o.operating ? ' · operating (opens in the ERP)' : ''}
   326→                      </span>
   327→                    </TableCell>
   328→                    <TableCell>
   329→                      <Select
   330→                        value={o.plan?.id ?? 'none'}
   331→                        onValueChange={(v) => void onAction(o.id, { action: 'setPlan', planId: v === 'none' ? null : v }, 'Plan updated')}
   332→                      >
   333→                        <SelectTrigger className="h-9 w-40" aria-label={`Plan for ${o.name}`}>
   334→                          <SelectValue />
   335→                        </SelectTrigger>
   336→                        <SelectContent>
   337→                          <SelectItem value="none">No plan (unlimited)</SelectItem>
   338→                          {plans.map((p) => (
   339→                            <SelectItem key={p.id} value={p.id}>
   340→                              {p.name}
   341→                            </SelectItem>
   342→                          ))}
   343→                        </SelectContent>
   344→                      </Select>
   345→                      {o.customized ? <span className="mt-1 block text-[11px] text-muted-foreground">custom limits</span> : null}
   346→                    </TableCell>
   347→                    <TableCell>{stateBadge(o)}</TableCell>
   348→                    <TableCell className="text-xs tabular-nums">
   349→                      {o.operating ? (
   350→                        <>
   351→                          {o.usage.parties}/{o.limits.parties || '∞'} parties · {o.usage.items}/{o.limits.items || '∞'} items ·{' '}

   486→function PlansTable({ plans, apiKey, reload }: { plans: Plan[]; apiKey: string; reload: () => Promise<void> }) {
   487→  const [open, setOpen] = useState(false)
   488→  const [editing, setEditing] = useState<Plan | null>(null)
   489→  const [form, setForm] = useState({ name: '', priceMonthly: '0', priceYearly: '0', maxItems: '0', maxWarehouses: '0', maxParties: '0' })
   490→  const [saving, setSaving] = useState(false)
   491→
   492→  const openNew = () => {
   493→    setEditing(null)
   494→    setForm({ name: '', priceMonthly: '0', priceYearly: '0', maxItems: '0', maxWarehouses: '0', maxParties: '0' })
   495→    setOpen(true)
   496→  }
   497→  const openEdit = (p: Plan) => {
   498→    setEditing(p)
   499→    setForm({
   500→      name: p.name,
   501→      priceMonthly: String(p.priceMonthly),
   502→      priceYearly: String(p.priceYearly),
   503→      maxItems: String(p.maxItems),
   504→      maxWarehouses: String(p.maxWarehouses),
   505→      maxParties: String(p.maxParties),
   506→    })
   507→    setOpen(true)
   508→  }
   509→
   510→  const save = async () => {
   511→    setSaving(true)
   512→    try {
   513→      const body = {
   514→        name: form.name.trim(),
   515→        priceMonthly: Number(form.priceMonthly) || 0,
   516→        priceYearly: Number(form.priceYearly) || 0,
   517→        maxItems: Math.max(0, Math.trunc(Number(form.maxItems) || 0)),
   518→        maxWarehouses: Math.max(0, Math.trunc(Number(form.maxWarehouses) || 0)),
   519→        maxParties: Math.max(0, Math.trunc(Number(form.maxParties) || 0)),
   520→      }
   521→      if (editing) await platformApi(`/api/platform/plans/${editing.id}`, { method: 'PATCH', body: JSON.stringify(body) }, apiKey)
   522→      else await platformApi('/api/platform/plans', { method: 'POST', body: JSON.stringify(body) }, apiKey)
   523→      toast.success(editing ? 'Plan updated' : 'Plan created')
   524→      setOpen(false)
   525→      void reload()
   526→    } catch (e) {
   527→      toastError(e)
   528→    } finally {
   529→      setSaving(false)
   530→    }
   531→  }
   532→
   533→  const del = async (p: Plan) => {
   534→    try {
   535→      await platformApi(`/api/platform/plans?id=${p.id}`, { method: 'DELETE' }, apiKey)
   536→      toast.success(`${p.name} deleted`)
   537→      void reload()
   538→    } catch (e) {
   539→      toastError(e)
   540→    }
   541→  }
   542→
   543→  return (
   544→    <>
   545→      <Card className="p-0">
   546→        <CardContent className="p-0">
   547→          <div className="flex flex-wrap items-center justify-between gap-2 p-4">
   548→            <p className="text-sm font-semibold">Plans (0 = unlimited)</p>
   549→            <Button size="sm" className="h-10 gap-2" onClick={openNew}>
   550→              <Plus className="size-4" aria-hidden /> New plan
   551→            </Button>
   552→          </div>
   553→          <div className="overflow-auto erp-scroll">
   554→            <Table>

   556→                <TableRow>
   557→                  <TableHead>Name</TableHead>
   558→                  <TableHead className="text-right">₹ / month</TableHead>
   559→                  <TableHead className="text-right">₹ / year</TableHead>
   560→                  <TableHead className="text-right">Items</TableHead>
   561→                  <TableHead className="text-right">Warehouses</TableHead>
   562→                  <TableHead className="text-right">Parties</TableHead>
   563→                  <TableHead className="text-right">Orgs</TableHead>
   564→                  <TableHead className="text-right">Actions</TableHead>
   565→                </TableRow>
   566→              </TableHeader>
   567→              <TableBody>
   568→                {plans.map((p) => (
   569→                  <TableRow key={p.id}>
   570→                    <TableCell className="font-medium">{p.name}</TableCell>
   571→                    <TableCell className="text-right tabular-nums">{money(p.priceMonthly)}</TableCell>
   572→                    <TableCell className="text-right tabular-nums">{money(p.priceYearly)}</TableCell>
   573→                    <TableCell className="text-right tabular-nums">{p.maxItems || '∞'}</TableCell>
   574→                    <TableCell className="text-right tabular-nums">{p.maxWarehouses || '∞'}</TableCell>
   575→                    <TableCell className="text-right tabular-nums">{p.maxParties || '∞'}</TableCell>
   576→                    <TableCell className="text-right tabular-nums">{p._count?.companies ?? 0}</TableCell>
   577→                    <TableCell>
   578→                      <div className="flex items-center justify-end gap-1.5">
   579→                        <Button size="sm" variant="outline" className="h-8" onClick={() => openEdit(p)}>
   580→                          Edit
   581→                        </Button>
   582→                        <AlertDialog>
   583→                          <AlertDialogTrigger asChild>
   584→                            <Button size="sm" variant="ghost" className="h-8 text-destructive hover:text-destructive" disabled={(p._count?.companies ?? 0) > 0}>

   595→                              <AlertDialogAction onClick={() => void del(p)} className="bg-destructive text-destructive-foreground hover:bg-destructive/90">

   598→                            </AlertDialogFooter>

   600→                        </AlertDialog>
   601→                      </div>
   602→                    </TableCell>
   603→                  </TableRow>
   604→                ))}
   605→              </TableBody>
   606→            </Table>
   607→          </div>
   608→        </CardContent>
   609→      </Card>
   610→
   611→      <Dialog open={open} onOpenChange={setOpen}>
   612→        <DialogContent className="sm:max-w-lg">
   613→          <DialogHeader>
   614→            <DialogTitle>{editing ? `Edit plan — ${editing.name}` : 'New plan'}</DialogTitle>
   615→            <DialogDescription>Limits of 0 mean unlimited. Prices are in ₹ (0 = free).</DialogDescription>
   616→          </DialogHeader>
   617→          <div className="grid gap-3 sm:grid-cols-2">
   618→            <div className="space-y-1.5 sm:col-span-2">
   619→              <Label htmlFor="plan-name">Name</Label>
   620→              <Input id="plan-name" value={form.name} onChange={(e) => setForm({ ...form, name: e.target.value })} className="h-11" placeholder="Growth" />
   621→            </div>
   622→            <div className="space-y-1.5">
   623→              <Label htmlFor="plan-m">₹ / month</Label>
   624→              <Input id="plan-m" type="number" min={0} value={form.priceMonthly} onChange={(e) => setForm({ ...form, priceMonthly: e.target.value })} className="h-11" />
   625→            </div>
   626→            <div className="space-y-1.5">
   627→              <Label htmlFor="plan-y">₹ / year</Label>

   659→function BackupCard() {
   660→  const [restoring, setRestoring] = useState(false)
   661→  const [confirmText, setConfirmText] = useState('')
   662→
   663→  const restore = async (file: File) => {
   664→    setRestoring(true)
   665→    try {
   666→      const snapshot = JSON.parse(await file.text())
   667→      const res = await api<{ restored: boolean }>('/api/backup/restore', {
   668→        method: 'POST',
   669→        body: JSON.stringify({ confirm: 'RESTORE', snapshot }),
   670→      })
   671→      if (res.restored) {
   672→        toast.success('Snapshot restored', { description: 'Reload the page to see the restored data.' })
   673→      }
   674→    } catch (e) {
   675→      toastError(e)
   676→    } finally {
   677→      setRestoring(false)
   678→      setConfirmText('')
   679→    }
   680→  }
   681→
   682→  return (
   683→    <Card className="p-0">
   684→      <CardContent className="space-y-3 p-4">
   685→        <div className="flex flex-wrap items-center justify-between gap-2">
   686→          <div>
   687→            <p className="text-sm font-semibold">Data & backup</p>
   688→            <p className="text-xs text-muted-foreground">
   689→              Full JSON snapshot of every table. Restore is destructive and transactional — everything is replaced inside one transaction.
   690→            </p>
   691→          </div>
   692→          <Button variant="outline" className="h-10 gap-2" onClick={() => downloadUrl('/api/backup')}>
   693→            <DatabaseBackup className="size-4" aria-hidden /> Download backup
   694→          </Button>
   695→        </div>
   696→        <div className="flex flex-wrap items-end gap-3 rounded-lg border border-dashed p-3">
   697→          <div className="min-w-44 flex-1 space-y-1.5">
   698→            <Label htmlFor="restore-confirm">Restore — type RESTORE to enable</Label>
   699→            <Input id="restore-confirm" value={confirmText} onChange={(e) => setConfirmText(e.target.value.toUpperCase())} placeholder="RESTORE" className="h-11" />
   700→          </div>
   701→          <Label
   702→            className={`inline-flex h-11 cursor-pointer items-center gap-2 rounded-md border px-4 text-sm font-medium ${
   703→              confirmText === 'RESTORE' ? 'border-destructive text-destructive hover:bg-destructive/10' : 'pointer-events-none opacity-50'
   704→            }`}
   705→          >
   706→            {restoring ? <Loader2 className="size-4 animate-spin" aria-hidden /> : <DatabaseBackup className="size-4" aria-hidden />}
   707→            {restoring ? 'Restoring…' : 'Pick snapshot file…'}
   708→            <input
   709→              type="file"
   710→              accept="application/json"
   711→              className="sr-only"
   712→              disabled={confirmText !== 'RESTORE' || restoring}
   713→              onChange={(e) => {
   714→                const f = e.target.files?.[0]
   715→                if (f) void restore(f)
   716→                e.target.value = ''
   717→              }}
   718→            />
   719→          </Label>
   720→        </div>
   721→      </CardContent>
   722→    </Card>
   723→  )
   724→}
