    11→// v1.10.83 - failures that are nobody's expected answer go to the Problem
    12→// log: no answer at all (server stopped) and 5xx. 4xx is left alone - a 409
    13→// duplicate number or a 404 is an answer the caller handles. Method, path and
    14→// status only, never the request body (it holds invoices and clients).
    15→const logApiProblem = (options, url, status, message) => {
    16→  logProblem({ kind: 'api', message, api: { method: options.method || 'GET', path: url, status } });
    17→};
    18→
    19→async function apiFetch(url, options = {}) {
    20→  let res;
    21→  try {
    22→    res = await fetch(url, {
    23→      headers: { 'Content-Type': 'application/json' },
    24→      ...options,
    25→    });
    26→  } catch (e) {
    27→    logApiProblem(options, url, 0, e?.message || 'Network error');
    28→    if (e && typeof e === 'object') e.problemLogged = true;
    29→    throw e;
    30→  }
    31→  if (!res.ok) {
    32→    // Preserve server-provided error message (used e.g. for 409 duplicate
    33→    // invoice number). Attach status so callers can branch on it.
    34→    let body = null;
    35→    try { body = await res.json(); } catch { /* non-JSON body */ }
    36→    const err = new Error(body?.error || `API error: ${res.status}`);
    37→    err.status = res.status;
    38→    err.body = body;
    39→    if (res.status >= 500) {
    40→      logApiProblem(options, url, res.status, err.message);
    41→      err.problemLogged = true;
    42→    }
    43→    throw err;
    44→  }
    45→  return res.json();
    46→}

   127→export const applyStockChanges = async (changes) => {

   129→  if (!ids.length) return [];
   130→  const products = await getAllProducts();
   131→  const saved = [];
   132→  for (const next of stockUpdates(products, changes)) { // services left out (#121)

   134→    saved.push(next);
   135→  }
   136→  return saved;

   229→// ---- Invoice counter ----
   230→// Uses the atomic /meta/:key/increment endpoint so two concurrent saves can't both
   231→// read 5 and both write 6 (= duplicate invoice numbers, which is a GST audit failure).
   232→//
   233→// Since v1.6.8: pass { peek: true } to preview the number WITHOUT reserving it.
   234→// Used by InvoiceGenerator to show the next number on form mount — the atomic
   235→// reservation happens only when the user actually saves. Cancelled forms no
   236→// longer burn counter values, so CA-audited businesses keep gapless sequences.
   237→// v1.10.78 (#93) - pass invoiceType: a tax invoice starts at least from
   238→// "Start numbering from"; other types (and receipts, which pass none) from 1.
   239→// v1.10.79 (#95) - pass businessKey: a business with its own numbering counts
   240→// in its own series, with its own prefix (Print Settings' per-type prefixes
   241→// belong to the shared series).
   242→// `invoiceDate` (YYYY-MM-DD) picks the financial year printed in the number;
   243→// without it, today's. A back-dated invoice (31 March, made on 2 April) took
   244→// the new year's label, so its number disagreed with its own date.
   245→export const getNextInvoiceNumber = async (prefix = 'INV', { peek = false, explicitPrefix = false, invoiceType = null, businessKey = '', invoiceDate = '' } = {}) => {
   246→  const settings = await getNumberSeries(businessKey);
   247→  const typePrefix = settings.own ? (INVOICE_TYPES[invoiceType]?.prefix || prefix) : prefix;
   248→  const key = counterKeyFor(typePrefix, settings, businessKey);
   249→  const start = invoiceType ? startFor(invoiceType, settings) : 1;
   250→  let next;
   251→  if (peek) {
   252→    // Encoded: a key can hold a business name ("M/S Sharma", "Shop #12").
   253→    const { value: current } = await apiFetch(`${API}/meta/${encodeURIComponent(key)}`);
   254→    // Same rule the server uses when the number is reserved.
   255→    next = nextCounterValue(current, start);
   256→  } else {
   257→    const inc = await apiFetch(`${API}/meta/${encodeURIComponent(key)}/increment`, { method: 'POST', body: JSON.stringify({ start }) });
   258→    next = inc.value;
   259→  }
   260→
   261→  // v1.10.10 — When `explicitPrefix` is true, the caller (per-type
   262→  // prefix override from Print Settings) wants THEIR prefix used as-is
   263→  // instead of the global brandPrefix from Invoice Number Settings.
   264→  // v1.10.53 — the financial year label is India's April-March FY.
   265→  // v1.10.79 — formatting lives in utils (formatDocNumber), shared with the
   266→  // server's recurring invoices.
   267→  return formatDocNumber({
   268→    settings, typePrefix, next,
   269→    explicitPrefix: settings.own ? false : explicitPrefix,
   270→    fyLabel: getFinancialYearLabel(invoiceDate ? parseLocalDate(invoiceDate) : new Date()),
   271→  });
   272→};

   406→// The next PO number for a business, PO/26-27/0001. peek shows it without
   407→// using it up; otherwise it is reserved on the server's atomic counter.
   408→export const getNextPoNumber = async ({ peek = false, businessKey = '', date = '' } = {}) => {
   409→  const key = encodeURIComponent(poCounterKey(businessKey));
   410→  const next = peek
   411→    ? nextCounterValue((await apiFetch(`${API}/meta/${key}`)).value)
   412→    : (await apiFetch(`${API}/meta/${key}/increment`, { method: 'POST', body: JSON.stringify({}) })).value;
   413→  return formatPoNumber(next, date);
   414→};

   486→// localStorage keys that are part of the "user's data" and should ride along in any
   487→// backup. Each key is documented with what it stores and whether losing it matters.
   488→// Exact keys we back up + restore. Wildcard patterns handled separately
   489→// below because localStorage doesn't have a native "keys matching pattern"
   490→// API — we iterate localStorage and match.
   491→const EXPORTABLE_LOCALSTORAGE_KEYS = [
   492→  'gst_customUnits',                 // user-defined units for line items
   493→  'gst_regionMode',                  // 'india' | 'international' | 'both'
   494→  'gst_enabledModules',              // map of disabled feature toggles
   495→  'gst_businessType',                // v1.10.83 — business type picked in Settings / first run
   496→  'gst_offlineMode',                 // v1.10.83 — Offline mode: no automatic internet use
   497→  'gst_menuOrder',                   // v1.10.83 — sidebar order

   499→  'gst_printSettings',               // v1.8.4 — thermal printer defaults (font, weight, all-caps, etc.)
   500→  'gst_itrCalcInputs',               // v1.7.0 — regime calculator inputs
   501→  'gst_itrPresumptive',              // v1.8.0 — presumptive taxation state
   502→  'gst_itrAdvanceTax',               // v1.8.0 — advance tax state
   503→  'gst_stockAlertSettings',          // v1.6.3 — low-stock threshold + on/off
   504→  'freegstbill_invoiceOptions',      // per-invoice display preference defaults
   505→  'freegstbill_theme',               // light/dark (was written as 'theme' in the old whitelist — typo)
   506→  'freegstbill_onboarded',           // skip welcome wizard on next launch
   507→  'freegstbill_dismissedUpdate',     // version the user dismissed the update banner for
   508→  'freegstbill_pwa_dismissed_at',    // 14-day PWA-install-banner cooldown timestamp

   511→// Keys matched by prefix. Currently used for per-profile last-used payment
   512→// account (gst_lastUsedAccountId_<profileId>) so multi-account preferences
   513→// survive backup/restore.
   514→const EXPORTABLE_LOCALSTORAGE_PREFIXES = [
   515→  'gst_lastUsedAccountId_',
   516→];
   517→
   518→const collectLocalStorage = () => {
   519→  const out = {};
   520→  EXPORTABLE_LOCALSTORAGE_KEYS.forEach(k => {
   521→    try { const v = localStorage.getItem(k); if (v !== null) out[k] = v; } catch { /* sandboxed */ }
   522→  });
   523→  // Prefix-matched keys — iterate all localStorage keys and include any hit.
   524→  try {
   525→    for (let i = 0; i < localStorage.length; i++) {
   526→      const k = localStorage.key(i);
   527→      if (!k) continue;
   528→      if (EXPORTABLE_LOCALSTORAGE_PREFIXES.some(p => k.startsWith(p))) {
   529→        out[k] = localStorage.getItem(k);
   530→      }
   531→    }
   532→  } catch { /* sandboxed */ }
   533→  return out;
   534→};
   535→
   536→const isRestorable = (k) => EXPORTABLE_LOCALSTORAGE_KEYS.includes(k)
   537→  || EXPORTABLE_LOCALSTORAGE_PREFIXES.some(p => k.startsWith(p));
   538→
   539→const restoreLocalStorage = (map) => {
   540→  if (!map || typeof map !== 'object') return;
   541→  Object.entries(map).forEach(([k, v]) => {
   542→    if (!isRestorable(k)) return; // ignore foreign keys — never let a backup smuggle unrelated writes in
   543→    try { localStorage.setItem(k, v); } catch { /* ignore */ }
   544→  });
   545→};

   548→// frontend doesn't have to ship its own copy of package.json. Falls back to 'unknown'

   550→// window before the user opens the app.

   597→  if (sel.products)       data.products = all.products;
   598→  // v1.10.83 (#115) - production entries travel with products: stock is the result of both.
   599→  if (sel.products && all.productions) data.productions = all.productions;
   600→  if (sel.expenses)       data.expenses = all.expenses;
   601→  if (sel.expenses)       data.cashBook = all.cashBook; // v1.10.83 - the cash book travels with expenses
   602→  if (sel.recurring)      data.recurring = all.recurring;

   604→  if (sel.purchases)      data.purchases = all.purchases;
   605→  if (sel.purchases)      data.purchaseOrders = all.purchaseOrders; // v1.10.83 (#112) - travel with purchase bills
   606→  if (sel.meta)           data.meta = all.meta; // includes regionMode, enabledModules, etc. on server
   607→  if (sel.localStorage)   data.localStorage = collectLocalStorage();

   612→// Inspect a backup file without committing — returns counts so the UI can show
   613→// what's in it before the user picks what to restore.
   614→export const inspectBackup = (jsonString) => {
   615→  let data;
   616→  try { data = JSON.parse(jsonString); }
   617→  catch { throw new Error('Not a valid JSON file'); }
   618→  return {
   619→    valid: !!data && (data.__freegstbill_backup || data.bills || data.profile),
   620→    exportedAt: data.exportedAt || null,
   621→    version: data.version || null,
   622→    counts: {
   623→      profile: data.profile && Object.keys(data.profile).length > 0 ? 1 : 0,
   624→      profiles: Array.isArray(data.profiles) ? data.profiles.length : 0,
   625→      bills: Array.isArray(data.bills) ? data.bills.length : 0,
   626→      clients: Array.isArray(data.clients) ? data.clients.length : 0,
   627→      termsTemplates: Array.isArray(data.termsTemplates) ? data.termsTemplates.length : 0,
   628→      products: Array.isArray(data.products) ? data.products.length : 0,
   629→      expenses: Array.isArray(data.expenses) ? data.expenses.length : 0,
   630→      purchases: Array.isArray(data.purchases) ? data.purchases.length : 0,
   631→      recurring: Array.isArray(data.recurring) ? data.recurring.length : 0,
   632→      receipts: Array.isArray(data.receipts) ? data.receipts.length : 0,
   633→      meta: data.meta ? Object.keys(data.meta).length : 0,
   634→      localStorage: data.localStorage ? Object.keys(data.localStorage).length : 0,
   635→    },
   636→    raw: data,
   637→  };

   640→// Selective import. `selection` is the same shape as for exportAllData.
   641→export const importData = async (jsonString, selection) => {
   642→  const inspected = typeof jsonString === 'string' ? inspectBackup(jsonString) : { raw: jsonString };
   643→  const data = inspected.raw;
   644→  const sel = selection || { profile: true, profiles: true, bills: true, clients: true, products: true, expenses: true, purchases: true, recurring: true, receipts: true, termsTemplates: true, meta: true, localStorage: true };
   645→
   646→  // Build a filtered payload — never touch collections the user didn't tick.
   647→  const payload = {};
   648→  if (sel.profile && data.profile)               payload.profile = data.profile;
   649→  if (sel.profiles && data.profiles)             payload.profiles = data.profiles;
   650→  if (sel.bills && data.bills)                   payload.bills = data.bills;
   651→  if (sel.clients && data.clients)               payload.clients = data.clients;
   652→  if (sel.termsTemplates && data.termsTemplates) payload.termsTemplates = data.termsTemplates;
   653→  if (sel.products && data.products)             payload.products = data.products;
   654→  if (sel.products && data.productions)          payload.productions = data.productions;
   655→  if (sel.expenses && data.expenses)             payload.expenses = data.expenses;
   656→  if (sel.expenses && data.cashBook)             payload.cashBook = data.cashBook;
   657→  if (sel.recurring && data.recurring)           payload.recurring = data.recurring;
   658→  if (sel.receipts && data.receipts)             payload.receipts = data.receipts;
   659→  if (sel.purchases && data.purchases)           payload.purchases = data.purchases;
   660→  if (sel.purchases && data.purchaseOrders)      payload.purchaseOrders = data.purchaseOrders;
   661→  if (sel.meta && data.meta)                     payload.meta = data.meta;
   662→
   663→  const result = await apiFetch(`${API}/import`, { method: 'POST', body: JSON.stringify(payload) });
   664→
   665→  if (sel.localStorage && data.localStorage) restoreLocalStorage(data.localStorage);
   666→
   667→  return result;
