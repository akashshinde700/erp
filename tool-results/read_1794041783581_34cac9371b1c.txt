    76→    76→async function stockAvail(tx: Tx, itemId: string, warehouseId?: string): Promise<number> {
    77→    77→  const rows = await tx.stockLedger.findMany({
    78→    78→    where: warehouseId ? { itemId, warehouseId, isCancelled: false } : { itemId, isCancelled: false },
    79→    79→    select: { qtyIn: true, qtyOut: true },
    80→    80→  })
    81→    81→  return r2(rows.reduce((s, r) => s + r.qtyIn - r.qtyOut, 0))
    82→    82→}

    84→    84→async function avgCost(tx: Tx, itemId: string, warehouseId: string): Promise<number> {
    85→    85→  const rows = await tx.stockLedger.findMany({
    86→    86→    where: { itemId, warehouseId, isCancelled: false },
    87→    87→    select: { qtyIn: true, rate: true },
    88→    88→  })
    89→    89→  let qty = 0
    90→    90→  let value = 0
    91→    91→  for (const r of rows) {
    92→    92→    qty += r.qtyIn
    93→    93→    value += r.qtyIn * r.rate
    94→    94→  }
    95→    95→  return qty > EPS ? r2(value / qty) : 0
    96→    96→}

   194→   194→type ProjectWithStats = Prisma.ProjectGetPayload<{ include: typeof projectInclude }> & {
   195→   195→  stats: {
   196→   196→    bomCount: number
   197→   197→    woCount: number
   198→   198→    taskTotal: number

   207→   207→function withStats(p: Prisma.ProjectGetPayload<{ include: typeof projectInclude }>): ProjectWithStats {
   208→   208→  const tasks = p.workOrders.flatMap((w) => w.tasks)
   209→   209→  const taskTotal = tasks.filter((t) => t.status !== 'CANCELLED').length
   210→   210→  const taskDone = tasks.filter((t) => t.status === 'COMPLETED').length
   211→   211→  const actualCost = r2(tasks.reduce((s, t) => s + t.labourCost + t.materialCost, 0))
   212→   212→  const labourCost = r2(tasks.reduce((s, t) => s + t.labourCost, 0))
   213→   213→  const materialCost = r2(tasks.reduce((s, t) => s + t.materialCost, 0))
   214→   214→  return {
   215→   215→    ...p,
   216→   216→    stats: {
   217→   217→      bomCount: p.boms.filter((b) => b.status !== 'CANCELLED').length,
   218→   218→      woCount: p.workOrders.filter((w) => w.status !== 'CANCELLED').length,
   219→   219→      taskTotal,
   220→   220→      taskDone,
   221→   221→      progressPct: taskTotal > 0 ? Math.round((taskDone / taskTotal) * 100) : 0,
   222→   222→      actualCost,
   223→   223→      labourCost,
   224→   224→      materialCost,
   225→   225→    },
   226→   226→  }

   234→   234→export async function getProjectDetail(id: string) {
   235→   235→  const p = await db.project.findUnique({
   236→   236→    where: { id },
   237→   237→    include: {
   238→   238→      party: true,
   239→   239→      boms: {
   240→   240→        orderBy: { createdAt: 'asc' },
   241→   241→        include: { items: { include: { item: { select: { id: true, code: true, name: true, uom: true, costRate: true } } } } },
   242→   242→      },
   243→   243→      workOrders: {
   244→   244→        orderBy: { createdAt: 'asc' },
   245→   245→        include: {
   246→   246→          lines: { include: { item: { select: { code: true, name: true, uom: true } } } },
   247→   247→          tasks: { orderBy: [{ seq: 'asc' }, { createdAt: 'asc' }], include: { employee: { select: { id: true, code: true, name: true, machineNo: true } } } },
   248→   248→        },
   249→   249→      },
   250→   250→    },
   251→   251→  })
   252→   252→  if (!p) throw new BusinessError('Project not found', 404)
   253→   253→
   254→   254→  // MRP-style stock check per BOM line (gross requirement vs available stock).
   255→   255→  const tx = db as unknown as Tx
   256→   256→  const boms = await Promise.all(
   257→   257→    p.boms.map(async (b) => {
   258→   258→      const items = await Promise.all(
   259→   259→        b.items.map(async (bi) => {
   260→   260→          const avail = await stockAvail(tx, bi.itemId)
   261→   261→          const grossQty = r2(bi.qty * (1 + bi.wastagePct / 100))
   262→   262→          return { ...bi, avail, grossQty, shortage: r2(Math.max(0, grossQty - avail)) }
   263→   263→        })
   264→   264→      )
   265→   265→      return { ...b, items }
   266→   266→    })
   267→   267→  )
   268→   268→
   269→   269→  const tasks = p.workOrders.flatMap((w) => w.tasks)
   270→   270→  const taskTotal = tasks.filter((t) => t.status !== 'CANCELLED').length
   271→   271→  const taskDone = tasks.filter((t) => t.status === 'COMPLETED').length
   272→   272→  const labourCost = r2(tasks.reduce((s, t) => s + t.labourCost, 0))
   273→   273→  const materialCost = r2(tasks.reduce((s, t) => s + t.materialCost, 0))
   274→   274→
   275→   275→  return {
   276→   276→    ...p,
   277→   277→    boms,
   278→   278→    stats: {
   279→   279→      bomCount: p.boms.filter((b) => b.status !== 'CANCELLED').length,
   280→   280→      woCount: p.workOrders.filter((w) => w.status !== 'CANCELLED').length,
   281→   281→      taskTotal,
   282→   282→      taskDone,
   283→   283→      progressPct: taskTotal > 0 ? Math.round((taskDone / taskTotal) * 100) : 0,
   284→   284→      labourCost,
   285→   285→      materialCost,
   286→   286→      actualCost: r2(labourCost + materialCost),
   287→   287→      estCost: p.estCost,
   288→   288→      // Project-wise P&L view: contract value vs cost so far
   289→   289→      budget: p.budget,
   290→   290→      margin: r2(p.budget - labourCost - materialCost),
   291→   291→      pendingTasks: taskTotal - taskDone,
   292→   292→    },
   293→   293→  }
   294→   294→}

   384→   384→export async function cancelProject(id: string) {
   385→   385→  return db.$transaction(async (tx) => {
   386→   386→    const p = await tx.project.findUnique({
   387→   387→      where: { id },
   388→   388→      include: { workOrders: { select: { id: true, status: true }, include: { tasks: { select: { status: true } } } } },
   389→   389→    })
   390→   390→    if (!p) throw new BusinessError('Project not found', 404)
   391→   391→    if (p.status === 'COMPLETED') throw new BusinessError('Completed projects cannot be cancelled')
   392→   392→    if (p.status === 'CANCELLED') throw new BusinessError('Project is already cancelled')
   393→   393→    const hasProduction = p.workOrders.some((w) => w.tasks.some((t) => t.status === 'COMPLETED'))
   394→   394→    if (hasProduction) {
   395→   395→      throw new BusinessError('Production has started on this project (completed tasks exist) — it cannot be cancelled', 409)
   396→   396→    }
   397→   397→    // Cascade-cancel dangling WOs/tasks (none wrote stock: no COMPLETED tasks).
   398→   398→    for (const w of p.workOrders) {
   399→   399→      if (w.status !== 'CANCELLED') {
   400→   400→        await tx.productionTask.updateMany({
   401→   401→          where: { workOrderId: w.id, status: { in: [...TASK_ACTIVE] } },
   402→   402→          data: { status: 'CANCELLED' },
   403→   403→        })
   404→   404→        await tx.workOrder.update({ where: { id: w.id }, data: { status: 'CANCELLED' } })
   405→   405→      }
   406→   406→    }
   407→   407→    return tx.project.update({ where: { id }, data: { status: 'CANCELLED' } })
   408→   408→  })
   409→   409→}

   435→   435→function computeBomLine(l: BomLineInput, costRate: number): Omit<Prisma.BomItemCreateWithoutBomInput, 'item'> {
   436→   436→  const qty = l.qty
   437→   437→  const wastagePct = l.wastagePct ?? 0
   438→   438→  const rate = l.rate && l.rate > 0 ? l.rate : costRate
   439→   439→  const estCost = r2(qty * (1 + wastagePct / 100) * rate)
   440→   440→  return {
   441→   441→    grade: l.grade ?? null,
   442→   442→    thickness: r2(l.thickness ?? 0),
   443→   443→    length: r2(l.length ?? 0),
   444→   444→    width: r2(l.width ?? 0),
   445→   445→    qty,
   446→   446→    uom: l.uom,
   447→   447→    wastagePct,
   448→   448→    rate,
   449→   449→    estCost,
   450→   450→    drawingRef: l.drawingRef ?? null,
   451→   451→    remarks: l.remarks ?? null,
   452→   452→  }
   453→   453→}

   455→   455→/** Recompute project.estCost = Σ approved BOM estCost (call inside a tx after mutation). */
   456→   456→async function refreshProjectEstCost(tx: Tx, projectId: string) {
   457→   457→  const boms = await tx.bom.findMany({
   458→   458→    where: { projectId, status: 'APPROVED' },
   459→   459→    select: { items: { select: { estCost: true } } },
   460→   460→  })
   461→   461→  const est = r2(boms.reduce((s, b) => s + b.items.reduce((x, i) => x + i.estCost, 0), 0))
   462→   462→  await tx.project.update({ where: { id: projectId }, data: { estCost: est } })
   463→   463→  return est
   464→   464→}

   650→   650→export async function createWorkOrder(input: WorkOrderInput) {
   651→   651→  return db.$transaction(async (tx) => {
   652→   652→    const project = await tx.project.findUnique({ where: { id: input.projectId } })
   653→   653→    if (!project) throw new BusinessError('Project not found', 404)
   654→   654→    if (project.status === 'COMPLETED' || project.status === 'CANCELLED') {
   655→   655→      throw new BusinessError(`Project is ${project.status} — work orders need an active project`)
   656→   656→    }
   657→   657→
   658→   658→    let lines: WoLineInput[] = input.lines ?? []
   659→   659→    let bomId: string | null = null
   660→   660→
   661→   661→    if (input.bomId) {
   662→   662→      const bom = await tx.bom.findUnique({ where: { id: input.bomId }, include: { items: true } })
   663→   663→      if (!bom) throw new BusinessError('BOM not found', 404)
   664→   664→      if (bom.projectId !== project.id) throw new BusinessError('BOM belongs to a different project', 400)
   665→   665→      if (bom.status !== 'APPROVED') throw new BusinessError('Only APPROVED BOMs can generate work orders')
   666→   666→      bomId = bom.id
   667→   667→      // WO quantity = BOM net + wastage allowance (the fabrication gross-up).
   668→   668→      lines = bom.items.map((bi) => ({
   669→   669→        itemId: bi.itemId,
   670→   670→        description: bom.productName,
   671→   671→        plannedQty: r2(bi.qty * (1 + bi.wastagePct / 100)),
   672→   672→        uom: bi.uom,
   673→   673→        bomItemId: bi.id,
   674→   674→      }))
   675→   675→    }
   676→   676→
   677→   677→    if (!lines.length) throw new BusinessError('Work order needs lines (pick a BOM or add lines manually)')
   678→   678→
   679→   679→    const items = await tx.item.findMany({ where: { id: { in: lines.map((l) => l.itemId) } } })
   680→   680→    const itemMap = new Map(items.map((i) => [i.id, i]))
   681→   681→    for (const l of lines) {
   682→   682→      const it = itemMap.get(l.itemId)
   683→   683→      if (!it) throw new BusinessError(`Item ${l.itemId} not found`, 404)
   684→   684→      if (!(l.plannedQty > 0)) throw new BusinessError(`Planned quantity must be positive for "${it.name}"`)
   685→   685→    }
   686→   686→
   687→   687→    const date = iso(input.date)
   688→   688→    const woNo = await nextDocNo(tx, 'WO', date)
   689→   689→    const wo = await tx.workOrder.create({
   690→   690→      data: {
   691→   691→        woNo,
   692→   692→        projectId: project.id,
   693→   693→        bomId,
   694→   694→        date,
   695→   695→        startDate: input.startDate ? iso(input.startDate) : null,
   696→   696→        endDate: input.endDate ? iso(input.endDate) : null,
   697→   697→        priority: input.priority ?? 'NORMAL',
   698→   698→        notes: input.notes ?? null,
   699→   699→        lines: {
   700→   700→          create: lines.map((l) => ({
   701→   701→            // Nested writes need relation syntax for the optional BOM link
   702→   702→            ...(l.bomItemId ? { bomItem: { connect: { id: l.bomItemId } } } : {}),
   703→   703→            description: l.description ?? null,
   704→   704→            plannedQty: l.plannedQty,
   705→   705→            uom: l.uom,
   706→   706→            item: { connect: { id: l.itemId } },
   707→   707→          })),
   708→   708→        },
   709→   709→      },
   710→   710→      include: { lines: true },
   711→   711→    })
   712→   712→
   713→   713→    // First work order kicks a PENDING project into RUNNING (the plan started).
   714→   714→    if (project.status === 'PENDING') {
   715→   715→      await tx.project.update({ where: { id: project.id }, data: { status: 'RUNNING' } })
   716→   716→    }
   717→   717→    return wo
   718→   718→  })

   898→   898→export interface CompleteTaskInput {
   899→   899→  actualQty: number
   900→   900→  rejectionQty?: number
   901→   901→  hours?: number
   902→   902→  labourRate?: number
   903→   903→  materialItemId?: string | null
   904→   904→  materialQty?: number
   905→   905→  notes?: string | null
   906→   906→  completedAt?: string | Date
   907→   907→}
   908→   908→
   909→   909→/**
   910→   910→ * Completing a task is the ONE place manufacturing touches the ledgers:
   911→   911→ * material consumption writes an append-only MFG_ISSUE row to the stock
   912→   912→ * ledger (availability-guarded, valued at moving average) inside the SAME
   913→   913→ * transaction as the task update. Completed tasks are final.
   914→   914→ */
   915→   915→export async function completeTask(id: string, input: CompleteTaskInput) {
   916→   916→  return db.$transaction(async (tx) => {
   917→   917→    const task = await tx.productionTask.findUnique({ where: { id }, include: { materialItem: true } })
   918→   918→    if (!task) throw new BusinessError('Task not found', 404)
   919→   919→    if (task.status !== 'IN_PROGRESS' && task.status !== 'ASSIGNED') {
   920→   920→      throw new BusinessError(`Task is ${task.status}`)
   921→   921→    }
   922→   922→    if (!(input.actualQty > 0)) throw new BusinessError('Actual quantity must be positive')
   923→   923→    const rejectionQty = input.rejectionQty ?? 0
   924→   924→    if (rejectionQty < 0) throw new BusinessError('Rejection quantity cannot be negative')
   925→   925→    if (task.plannedQty > 0 && input.actualQty + rejectionQty > task.plannedQty * TASK_TOLERANCE + EPS) {
   926→   926→      throw new BusinessError(
   927→   927→        `Over-production: actual ${input.actualQty} + rejection ${rejectionQty} exceeds planned ${task.plannedQty} (10% tolerance)`
   928→   928→      )
   929→   929→    }
   930→   930→    const hours = input.hours ?? 0
   931→   931→    if (hours < 0 || hours > 24 * 30) throw new BusinessError('Production hours look wrong')
   932→   932→    const labourRate = input.labourRate && input.labourRate > 0 ? input.labourRate : task.labourRate
   933→   933→    const labourCost = r2(hours * labourRate)
   934→   934→    const completedAt = iso(input.completedAt)
   935→   935→
   936→   936→    // ---- material consumption → stock ledger (atomic with this task) ----
   937→   937→    let materialItemId: string | null = task.materialItemId
   938→   938→    let materialQty = task.materialQty
   939→   939→    let materialCost = task.materialCost
   940→   940→    if (input.materialItemId || (input.materialQty && input.materialQty > 0)) {
   941→   941→      const itemId = input.materialItemId ?? task.materialItemId
   942→   942→      const qty = input.materialQty ?? 0
   943→   943→      if (!itemId) throw new BusinessError('Pick the material item being consumed')
   944→   944→      if (!(qty > 0)) throw new BusinessError('Material consumption quantity must be positive')
   945→   945→      const mat = await tx.item.findUnique({ where: { id: itemId } })
   946→   946→      if (!mat) throw new BusinessError('Material item not found', 404)
   947→   947→      if (!mat.isStockItem) throw new BusinessError(`"${mat.name}" is not a stock item`)
   948→   948→      const warehouseId = await defaultWarehouseId(tx)
   949→   949→      const avail = await stockAvail(tx, itemId, warehouseId)
   950→   950→      if (avail + EPS < qty) {
   951→   951→        throw new BusinessError(
   952→   952→          `Insufficient stock for "${mat.name}": available ${avail} ${mat.uom}, consumption ${qty} ${mat.uom} — receive material first (GRN/Purchase)`
   953→   953→        )
   954→   954→      }
   955→   955→      const rate = await avgCost(tx, itemId, warehouseId)
   956→   956→      materialItemId = itemId
   957→   957→      materialQty = qty
   958→   958→      materialCost = r2(qty * rate)
   959→   959→      await tx.stockLedger.create({
   960→   960→        data: {
   961→   961→          postingDate: completedAt,
   962→   962→          voucherType: 'MFG_ISSUE',
   963→   963→          voucherNo: task.taskNo,
   964→   964→          itemId,
   965→   965→          warehouseId,
   966→   966→          qtyIn: 0,
   967→   967→          qtyOut: qty,
   968→   968→          rate,
   969→   969→          remarks: `Consumed on ${task.taskNo} (${OPERATION_LABEL[task.operation] ?? task.operation})`,
   970→   970→        },
   971→   971→      })
   972→   972→    }
   973→   973→
   974→   974→    return tx.productionTask.update({
   975→   975→      where: { id },
   976→   976→      data: {
   977→   977→        actualQty: input.actualQty,
   978→   978→        rejectionQty,
   979→   979→        hours,
   980→   980→        labourRate,
   981→   981→        labourCost,
   982→   982→        materialItemId,
   983→   983→        materialQty,
   984→   984→        materialCost,
   985→   985→        status: 'COMPLETED',
   986→   986→        startedAt: task.startedAt ?? completedAt,
   987→   987→        completedAt,
   988→   988→        notes: input.notes ?? task.notes,
   989→   989→      },
   990→   990→    })
   991→   991→  })
   992→   992→}
