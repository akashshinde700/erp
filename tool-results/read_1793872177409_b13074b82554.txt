    11→    11→// v1.10.83 - failures that are nobody's expected answer go to the Problem
    12→    12→// log: no answer at all (server stopped) and 5xx. 4xx is left alone - a 409
    13→    13→// duplicate number or a 404 is an answer the caller handles. Method, path and
    14→    14→// status only, never the request body (it holds invoices and clients).
    15→    15→const logApiProblem = (options, url, status, message) => {
    16→    16→  logProblem({ kind: 'api', message, api: { method: options.method || 'GET', path: url, status } });
    17→    17→};
    18→    18→
    19→    19→async function apiFetch(url, options = {}) {
    20→    20→  let res;
    21→    21→  try {
    22→    22→    res = await fetch(url, {
    23→    23→      headers: { 'Content-Type': 'application/json' },
    24→    24→      ...options,
    25→    25→    });
    26→    26→  } catch (e) {
    27→    27→    logApiProblem(options, url, 0, e?.message || 'Network error');
    28→    28→    if (e && typeof e === 'object') e.problemLogged = true;
    29→    29→    throw e;
    30→    30→  }
    31→    31→  if (!res.ok) {
    32→    32→    // Preserve server-provided error message (used e.g. for 409 duplicate
    33→    33→    // invoice number). Attach status so callers can branch on it.
    34→    34→    let body = null;
    35→    35→    try { body = await res.json(); } catch { /* non-JSON body */ }
    36→    36→    const err = new Error(body?.error || `API error: ${res.status}`);
    37→    37→    err.status = res.status;
    38→    38→    err.body = body;
    39→    39→    if (res.status >= 500) {
    40→    40→      logApiProblem(options, url, res.status, err.message);
    41→    41→      err.problemLogged = true;
    42→    42→    }
    43→    43→    throw err;
    44→    44→  }
    45→    45→  return res.json();
    46→    46→}

   229→   229→// ---- Invoice counter ----
   230→   230→// Uses the atomic /meta/:key/increment endpoint so two concurrent saves can't both
   231→   231→// read 5 and both write 6 (= duplicate invoice numbers, which is a GST audit failure).
   232→   232→//
   233→   233→// Since v1.6.8: pass { peek: true } to preview the number WITHOUT reserving it.
   234→   234→// Used by InvoiceGenerator to show the next number on form mount — the atomic
   235→   235→// reservation happens only when the user actually saves. Cancelled forms no
   236→   236→// longer burn counter values, so CA-audited businesses keep gapless sequences.
   237→   237→// v1.10.78 (#93) - pass invoiceType: a tax invoice starts at least from
   238→   238→// "Start numbering from"; other types (and receipts, which pass none) from 1.
   239→   239→// v1.10.79 (#95) - pass businessKey: a business with its own numbering counts
   240→   240→// in its own series, with its own prefix (Print Settings' per-type prefixes
   241→   241→// belong to the shared series).
   242→   242→// `invoiceDate` (YYYY-MM-DD) picks the financial year printed in the number;
   243→   243→// without it, today's. A back-dated invoice (31 March, made on 2 April) took
   244→   244→// the new year's label, so its number disagreed with its own date.
   245→   245→export const getNextInvoiceNumber = async (prefix = 'INV', { peek = false, explicitPrefix = false, invoiceType = null, businessKey = '', invoiceDate = '' } = {}) => {
   246→   246→  const settings = await getNumberSeries(businessKey);
   247→   247→  const typePrefix = settings.own ? (INVOICE_TYPES[invoiceType]?.prefix || prefix) : prefix;
   248→   248→  const key = counterKeyFor(typePrefix, settings, businessKey);
   249→   249→  const start = invoiceType ? startFor(invoiceType, settings) : 1;
   250→   250→  let next;
   251→   251→  if (peek) {
   252→   252→    // Encoded: a key can hold a business name ("M/S Sharma", "Shop #12").
   253→   253→    const { value: current } = await apiFetch(`${API}/meta/${encodeURIComponent(key)}`);
   254→   254→    // Same rule the server uses when the number is reserved.
   255→   255→    next = nextCounterValue(current, start);
   256→   256→  } else {
   257→   257→    const inc = await apiFetch(`${API}/meta/${encodeURIComponent(key)}/increment`, { method: 'POST', body: JSON.stringify({ start }) });
   258→   258→    next = inc.value;
   259→   259→  }
   260→   260→
   261→   261→  // v1.10.10 — When `explicitPrefix` is true, the caller (per-type
   262→   262→  // prefix override from Print Settings) wants THEIR prefix used as-is
   263→   263→  // instead of the global brandPrefix from Invoice Number Settings.
   264→   264→  // v1.10.53 — the financial year label is India's April-March FY.
   265→   265→  // v1.10.79 — formatting lives in utils (formatDocNumber), shared with the
   266→   266→  // server's recurring invoices.
   267→   267→  return formatDocNumber({
   268→   268→    settings, typePrefix, next,
   269→   269→    explicitPrefix: settings.own ? false : explicitPrefix,
   270→   270→    fyLabel: getFinancialYearLabel(invoiceDate ? parseLocalDate(invoiceDate) : new Date()),
   271→   271→  });
   272→   272→};

   511→   511→// Keys matched by prefix. Currently used for per-profile last-used payment
   512→   512→// account (gst_lastUsedAccountId_<profileId>) so multi-account preferences

   514→   514→const EXPORTABLE_LOCALSTORAGE_PREFIXES = [
   515→   515→  'gst_lastUsedAccountId_',
   516→   516→];

   518→   518→const collectLocalStorage = () => {
   519→   519→  const out = {};
   520→   520→  EXPORTABLE_LOCALSTORAGE_KEYS.forEach(k => {
   521→   521→    try { const v = localStorage.getItem(k); if (v !== null) out[k] = v; } catch { /* sandboxed */ }
   522→   522→  });
   523→   523→  // Prefix-matched keys — iterate all localStorage keys and include any hit.
   524→   524→  try {
   525→   525→    for (let i = 0; i < localStorage.length; i++) {
   526→   526→      const k = localStorage.key(i);
   527→   527→      if (!k) continue;
   528→   528→      if (EXPORTABLE_LOCALSTORAGE_PREFIXES.some(p => k.startsWith(p))) {
   529→   529→        out[k] = localStorage.getItem(k);
   530→   530→      }
   531→   531→    }
   532→   532→  } catch { /* sandboxed */ }
   533→   533→  return out;
   534→   534→};
   535→   535→
   536→   536→const isRestorable = (k) => EXPORTABLE_LOCALSTORAGE_KEYS.includes(k)
   537→   537→  || EXPORTABLE_LOCALSTORAGE_PREFIXES.some(p => k.startsWith(p));
   538→   538→
   539→   539→const restoreLocalStorage = (map) => {
   540→   540→  if (!map || typeof map !== 'object') return;
   541→   541→  Object.entries(map).forEach(([k, v]) => {
   542→   542→    if (!isRestorable(k)) return; // ignore foreign keys — never let a backup smuggle unrelated writes in
   543→   543→    try { localStorage.setItem(k, v); } catch { /* ignore */ }
   544→   544→  });
   545→   545→};
