     3→/**
     4→ * ForgeERP — API client.
     5→ * One helper for the {success,data,error} envelope; typed entities for every
     6→ * backend endpoint; tiny fetch hooks with loading/error state.
     7→ * All paths are relative (same origin) — Caddy-friendly by construction.
     8→ */

    13→// ---------- envelope ----------
    14→
    15→interface Envelope<T> {
    16→  success: boolean
    17→  data?: T
    18→  error?: string
    19→}
    20→
    21→/** API error carrying the HTTP status — lets callers distinguish auth loss
    22→ *  (401) from transient failures (409 races, 5xx, network). */
    23→export class ApiError extends Error {
    24→  status: number
    25→
    26→  constructor(message: string, status: number) {
    27→    super(message)
    28→    this.name = 'ApiError'
    29→    this.status = status
    30→  }
    31→}
    32→
    33→export async function api<T>(path: string, init?: RequestInit): Promise<T> {
    34→  let res: Response
    35→  try {
    36→    res = await fetch(path, {
    37→      ...init,
    38→      headers: { 'Content-Type': 'application/json', ...(init?.headers ?? {}) },
    39→    })
    40→  } catch {
    41→    throw new ApiError('Network error — is the server running?', 0)
    42→  }
    43→  let json: Envelope<T>
    44→  try {
    45→    json = (await res.json()) as Envelope<T>
    46→  } catch {
    47→    throw new ApiError(`Request failed (${res.status})`, res.status)
    48→  }
    49→  if (!res.ok || !json.success) throw new ApiError(json.error ?? `Request failed (${res.status})`, res.status)
    50→  return json.data as T
    51→}
    52→
    53→export const apiGet = <T>(path: string): Promise<T> => api<T>(path)
    54→export const apiPost = <T>(path: string, body: unknown): Promise<T> =>
    55→  api<T>(path, { method: 'POST', body: JSON.stringify(body) })
    56→export const apiPut = <T>(path: string, body: unknown): Promise<T> =>
    57→  api<T>(path, { method: 'PUT', body: JSON.stringify(body) })
    58→export const apiDelete = <T>(path: string): Promise<T> => api<T>(path, { method: 'DELETE' })
    59→
    60→/** Toast any thrown error (sonner). */
    61→export function toastError(e: unknown): void {
    62→  toast.error(e instanceof Error ? e.message : 'Something went wrong')
    63→}

   704→/** Fetch + parse with loading/error state; `reload()` re-runs (stale-while-revalidate). `null` path skips. */
   705→export function useApi<T>(path: string | null) {
   706→  const [state, setState] = useState<{ forPath: string | null; data: T | null; error: string | null }>({
   707→    forPath: null,
   708→    data: null,
   709→    error: null,
   710→  })
   711→  const [tick, setTick] = useState(0)
   712→
   713→  useEffect(() => {
   714→    if (path === null) return
   715→    let alive = true
   716→    api<T>(path)
   717→      .then((d) => {
   718→        if (alive) setState({ forPath: path, data: d, error: null })
   719→      })
   720→      .catch((e: unknown) => {
   721→        if (alive)
   722→          setState({ forPath: path, data: null, error: e instanceof Error ? e.message : 'Request failed' })
   723→      })
   724→    return () => {
   725→      alive = false
   726→    }
   727→  }, [path, tick])
   728→
   729→  const current = state.forPath === path ? state : null
   730→  const data = current?.data ?? null
   731→  const error = current?.error ?? null
   732→  const loading = path !== null && current === null
   733→  const reload = useCallback(() => setTick((t) => t + 1), [])
   734→  return { data, loading, error, reload }
   735→}

   751→      items.reload()

   950→/**
   951→ * Opens the GST e-invoice portal's Bulk Upload page in a new tab, together with
   952→ * an e-Invoice JSON download. Popup-blocker-safe: the tab is opened
   953→ * synchronously inside the click gesture (browsers block window.open after
   954→ * awaits). `retry()` re-attempts in case the first open was blocked;
   955→ * `cancel()` closes the tab when pre-validation fails and no JSON is produced.
   956→ */
   957→export function openEInvoicePortal(): { retry: () => void; cancel: () => void } {
   958→  const tab = window.open(EINVOICE_PORTAL_URL, '_blank')
   959→  return {
   960→    retry: () => {
   961→      if (!tab) void window.open(EINVOICE_PORTAL_URL, '_blank', 'noopener')
   962→    },
   963→    cancel: () => tab?.close(),
   964→  }
   965→}

   967→/** Client-side CSV download (BOM + quote escaping). */
   968→export function downloadCsv(filename: string, rows: (string | number)[][]): void {
   969→  const csv = rows.map((r) => r.map((c) => `"${String(c).replace(/"/g, '""')}"`).join(',')).join('\r\n')
   970→  const blob = new Blob([`\uFEFF${csv}`], { type: 'text/csv;charset=utf-8' })
   971→  const url = URL.createObjectURL(blob)
   972→  const a = document.createElement('a')
   973→  a.href = url
   974→  a.download = filename
   975→  document.body.appendChild(a)
   976→  a.click()
   977→  a.remove()
   978→  URL.revokeObjectURL(url)
   979→}
