   281→export function ErpApp() {
   282→  const { view, params, setView } = useErpUi()
   283→  const company = useApi<Company | null>('/api/masters/company')
   284→  const [manualSetup, setManualSetup] = useState(false)
   285→  const [drawerOpen, setDrawerOpen] = useState(false)
   286→  const mounted = useMounted()
   287→
   288→  // ---------- organization login (SaaS) ----------
   289→  // undefined = session probe in flight · null = signed out · info = signed in.
   290→  // Platform Admin is a SEPARATE console (platformMode) — never inside the ERP.
   291→  // A SUPER_ADMIN session opens the console directly (like the reference SaaS).
   292→  const [session, setSession] = useState<SessionInfo | null | undefined>(undefined)
   293→  const [platformMode, setPlatformMode] = useState(false)
   294→
   295→  useEffect(() => {
   296→    let cancelled = false
   297→    api<SessionInfo | null>('/api/auth/session')
   298→      .then((s) => {
   299→        if (!cancelled) setSession(s)
   300→      })
   301→      .catch(() => {
   302→        if (!cancelled) setSession(null)
   303→      })
   304→    return () => {
   305→      cancelled = true
   306→    }
   307→  }, [])
   308→
   309→  const logout = async () => {
   310→    try {
   311→      await apiPost('/api/auth/logout', {})
   312→      toast.success('Signed out')
   313→    } catch {
   314→      toast.error('Could not sign out — try again')
   315→    }
   316→    setSession(null)
   317→    setView('dashboard')
   318→  }
   319→
   320→  useEffect(() => {
   321→    window.scrollTo({ top: 0 })
   322→  }, [view])
   323→
   324→  // ---------- global keyboard: palette (Ctrl/⌘+K, /), g+key goto, ? help, Alt+S/Alt+P ----------
   325→  const [paletteOpen, setPaletteOpen] = useState(false)
   326→  const [helpOpen, setHelpOpen] = useState(false)
   327→  const gotoArmed = useRef(0)
   328→
   329→  const openSalesCreate = useCallback(() => {
   330→    setCreatePreset({ type: 'sales-invoice' })
   331→    setView('sales')
   332→  }, [setView])
   333→  const openPurchaseCreate = useCallback(() => {
   334→    setCreatePreset({ type: 'purchase-invoice' })
   335→    setView('purchase')
   336→  }, [setView])
   337→
   338→  useEffect(() => {
   339→    const onKey = (e: KeyboardEvent) => {
   340→      // Ctrl/⌘ + K → palette (always)
   341→      if ((e.ctrlKey || e.metaKey) && e.key.toLowerCase() === 'k') {
   342→        e.preventDefault()
   343→        setPaletteOpen(true)
   344→        return
   345→      }
   346→      if (isTypingTarget(e)) return
   347→      // Alt+S / Alt+P → new documents from anywhere
   348→      if (e.altKey && e.key.toLowerCase() === 's') {
   349→        e.preventDefault()
   350→        openSalesCreate()
   351→        return
   352→      }
   353→      if (e.altKey && e.key.toLowerCase() === 'p') {
   354→        e.preventDefault()
   355→        openPurchaseCreate()
   356→        return
   357→      }
   358→      // "?" → shortcuts help
   359→      if (e.key === '?') {
   360→        e.preventDefault()
   361→        setHelpOpen(true)
   362→        return
   363→      }
   364→      // "/" → palette
   365→      if (e.key === '/') {
   366→        e.preventDefault()
   367→        setPaletteOpen(true)
   368→        return
   369→      }
   370→      // g + key → module (Gmail style, 1.5 s window)
   371→      if (e.key.toLowerCase() === 'g' && !e.ctrlKey && !e.metaKey && !e.altKey) {
   372→        gotoArmed.current = Date.now()
   373→        return
   374→      }
   375→      const goto = GOTO_KEYS[e.key.toLowerCase()]
   376→      if (goto && Date.now() - gotoArmed.current < 1500 && !e.ctrlKey && !e.metaKey && !e.altKey) {
   377→        e.preventDefault()
   378→        gotoArmed.current = 0
   379→        setView(goto.view)
   380→      }
   381→    }
   382→    window.addEventListener('keydown', onKey)
   383→    return () => window.removeEventListener('keydown', onKey)
   384→  }, [setView, openSalesCreate, openPurchaseCreate])
   385→
   386→  // Track recent views for the palette
   387→  useEffect(() => {
   388→    pushRecent(view)
   389→  }, [view])
   390→
   391→  // Boot probe: company fetch + session fetch must both settle before deciding.
   392→  if (company.loading || session === undefined) {
   393→    return (
   394→      <div className="flex min-h-screen flex-col gap-4 p-6">
   395→        <Skeleton className="h-14 w-full rounded-lg" />
   396→        <div className="flex flex-1 gap-4">
   397→          <Skeleton className="hidden h-96 w-60 rounded-lg md:block" />
   398→          <Skeleton className="h-96 flex-1 rounded-lg" />
   399→        </div>
   400→      </div>
   401→    )
   402→  }
   403→
   404→  // First boot, no company yet → demo seed or manual setup wizard.
   405→  if (company.data === null) {
   406→    if (manualSetup) {
   407→      return (
   408→        <ManualSetupView
   409→          onDone={() => {
   410→            setManualSetup(false)
   411→            company.reload()
   412→          }}
   413→          onBack={() => setManualSetup(false)}
   414→        />
   415→      )
   416→    }
   417→    return (
   418→      <SetupScreen
   419→        onDone={() => {
   420→          setManualSetup(false)
   421→          company.reload()
   422→        }}
   423→        onManual={() => setManualSetup(true)}
   424→      />
   425→    )
   426→  }
   427→
   428→  // Platform Admin — SEPARATE console (own header + pill nav), opened from the
   429→  // login screen or a SUPER_ADMIN sign-in; never inside an organization's ERP.
   430→  // Exiting while a SUPER_ADMIN session is active signs the admin out — the
   431→  // console is their home, not the org ERP.
   432→  const exitPlatform = async () => {
   433→    const wasSuperAdmin = !!session && session.user.role === 'SUPER_ADMIN'
   434→    setPlatformMode(false)
   435→    if (wasSuperAdmin) {
   436→      try {
   437→        await apiPost('/api/auth/logout', {})
   438→      } catch {
   439→        /* best effort — the console may already have logged out */
   440→      }
   441→      setSession(null)
   442→      return
   443→    }
   444→    // Key-mode exit: stay honest about the cookie the console may have swapped.
   445→    api<SessionInfo | null>('/api/auth/session')
   446→      .then((s) => setSession(s))
   447→      .catch(() => setSession(null))
   448→  }
   449→
   450→  const platformConsole = platformMode || (!!session && session.user.role === 'SUPER_ADMIN')
   451→
   452→  if (platformConsole) {
   453→    return (
   454→      <Suspense
   455→        fallback={
   456→          <div className="flex min-h-dvh items-center justify-center p-6">
   457→            <Skeleton className="h-96 w-full max-w-6xl rounded-xl" />
   458→          </div>
   459→        }
   460→      >
   461→        <PlatformView
   462→          onExit={() => void exitPlatform()}
   463→          onLoginAs={(s) => {
   464→            // Impersonation handover: the console set the session cookie for the
   465→            // target org's owner — hand the browser to their workspace.
   466→            setSession(s)
   467→            setPlatformMode(false)
   468→          }}
   469→        />
   470→      </Suspense>
   471→    )
   472→  }
   473→
   474→  // Not signed in → organization login screen (SaaS entry).
   475→  if (!session) {
   476→    return (
   477→      <LoginView
   478→        onLogin={(s) => setSession(s)}
   479→        onOpenPlatform={() => setPlatformMode(true)}
   480→        onPlatformLogin={(s) => {
   481→          setSession(s)
   482→          setPlatformMode(true)
   483→        }}
   484→      />
   485→    )
   486→  }
   487→
   488→  // Signed in to an org that is NOT the operating org → honest org portal
   489→  // (real profile/subscription data, never another org's books).
   490→  if (!session.operating) {
   491→    return <OrgPortal session={session} onLogout={() => void logout()} />
   492→  }
   493→
   494→  const fy = mounted ? fyCode(new Date()) : '····'
   495→  const today = mounted ? dateStr(new Date()) : ''
   496→
   497→  const chip = (children: React.ReactNode) => (
   498→    <span className="hidden items-center gap-1.5 rounded-md border bg-card px-2 py-1 text-xs text-muted-foreground lg:inline-flex">
   499→      {children}
   500→    </span>
   501→  )
   502→
   503→  let content: React.ReactNode
   504→  switch (view) {
   505→    case 'sales':
   506→      content = <SalesView />
   507→      break
   508→    case 'sales-detail':
   509→      content = <SalesDetailView id={params.id ?? ''} />
   510→      break
   511→    case 'purchase':
   512→      content = <PurchaseView />
   513→      break
   514→    case 'purchase-detail':
   515→      content = <PurchaseDetailView id={params.id ?? ''} />
   516→      break
   517→    case 'jobwork':
   518→      content = <JobworkView />
   519→      break
   520→    case 'jobwork-detail':
   521→      content = <JobworkDetailView id={params.id ?? ''} />
   522→      break
   523→    case 'inventory':
   524→      content = <InventoryView />
   525→      break
   526→    case 'accounting':
   527→      content = <AccountingView />
   528→      break
   529→    case 'reports':
   530→      content = <ReportsView />
   531→      break
   532→    case 'masters':
   533→      content = <MastersView />
   534→      break
   535→    case 'tally':
   536→      content = <TallyView />
   537→      break
   538→    case 'audit':
   539→      content = <AuditView />
   540→      break
   541→    case 'docs':
   542→      content = <DocsView />
   543→      break
   544→    default:
   545→      content = <DashboardView />
   546→  }
   547→
