   281→   281→export function ErpApp() {
   282→   282→  const { view, params, setView } = useErpUi()
   283→   283→  const company = useApi<Company | null>('/api/masters/company')
   284→   284→  const [manualSetup, setManualSetup] = useState(false)
   285→   285→  const [drawerOpen, setDrawerOpen] = useState(false)
   286→   286→  const mounted = useMounted()
   287→   287→
   288→   288→  // ---------- organization login (SaaS) ----------
   289→   289→  // undefined = session probe in flight · null = signed out · info = signed in.
   290→   290→  // Platform Admin is a SEPARATE console (platformMode) — never inside the ERP.
   291→   291→  // A SUPER_ADMIN session opens the console directly (like the reference SaaS).
   292→   292→  const [session, setSession] = useState<SessionInfo | null | undefined>(undefined)
   293→   293→  const [platformMode, setPlatformMode] = useState(false)
   294→   294→
   295→   295→  useEffect(() => {
   296→   296→    let cancelled = false
   297→   297→    api<SessionInfo | null>('/api/auth/session')
   298→   298→      .then((s) => {
   299→   299→        if (!cancelled) setSession(s)
   300→   300→      })
   301→   301→      .catch(() => {
   302→   302→        if (!cancelled) setSession(null)
   303→   303→      })
   304→   304→    return () => {
   305→   305→      cancelled = true
   306→   306→    }
   307→   307→  }, [])
   308→   308→
   309→   309→  const logout = async () => {
   310→   310→    try {
   311→   311→      await apiPost('/api/auth/logout', {})
   312→   312→      toast.success('Signed out')
   313→   313→    } catch {
   314→   314→      toast.error('Could not sign out — try again')
   315→   315→    }
   316→   316→    setSession(null)
   317→   317→    setView('dashboard')
   318→   318→  }
   319→   319→
   320→   320→  useEffect(() => {
   321→   321→    window.scrollTo({ top: 0 })
   322→   322→  }, [view])
   323→   323→
   324→   324→  // ---------- global keyboard: palette (Ctrl/⌘+K, /), g+key goto, ? help, Alt+S/Alt+P ----------
   325→   325→  const [paletteOpen, setPaletteOpen] = useState(false)
   326→   326→  const [helpOpen, setHelpOpen] = useState(false)
   327→   327→  const gotoArmed = useRef(0)
   328→   328→
   329→   329→  const openSalesCreate = useCallback(() => {
   330→   330→    setCreatePreset({ type: 'sales-invoice' })
   331→   331→    setView('sales')
   332→   332→  }, [setView])
   333→   333→  const openPurchaseCreate = useCallback(() => {
   334→   334→    setCreatePreset({ type: 'purchase-invoice' })
   335→   335→    setView('purchase')
   336→   336→  }, [setView])
   337→   337→
   338→   338→  useEffect(() => {
   339→   339→    const onKey = (e: KeyboardEvent) => {
   340→   340→      // Ctrl/⌘ + K → palette (always)
   341→   341→      if ((e.ctrlKey || e.metaKey) && e.key.toLowerCase() === 'k') {
   342→   342→        e.preventDefault()
   343→   343→        setPaletteOpen(true)
   344→   344→        return
   345→   345→      }
   346→   346→      if (isTypingTarget(e)) return
   347→   347→      // Alt+S / Alt+P → new documents from anywhere
   348→   348→      if (e.altKey && e.key.toLowerCase() === 's') {
   349→   349→        e.preventDefault()
   350→   350→        openSalesCreate()
   351→   351→        return
   352→   352→      }
   353→   353→      if (e.altKey && e.key.toLowerCase() === 'p') {
   354→   354→        e.preventDefault()
   355→   355→        openPurchaseCreate()
   356→   356→        return
   357→   357→      }
   358→   358→      // "?" → shortcuts help
   359→   359→      if (e.key === '?') {
   360→   360→        e.preventDefault()
   361→   361→        setHelpOpen(true)
   362→   362→        return
   363→   363→      }
   364→   364→      // "/" → palette
   365→   365→      if (e.key === '/') {
   366→   366→        e.preventDefault()
   367→   367→        setPaletteOpen(true)
   368→   368→        return
   369→   369→      }
   370→   370→      // g + key → module (Gmail style, 1.5 s window)
   371→   371→      if (e.key.toLowerCase() === 'g' && !e.ctrlKey && !e.metaKey && !e.altKey) {
   372→   372→        gotoArmed.current = Date.now()
   373→   373→        return
   374→   374→      }
   375→   375→      const goto = GOTO_KEYS[e.key.toLowerCase()]
   376→   376→      if (goto && Date.now() - gotoArmed.current < 1500 && !e.ctrlKey && !e.metaKey && !e.altKey) {
   377→   377→        e.preventDefault()
   378→   378→        gotoArmed.current = 0
   379→   379→        setView(goto.view)
   380→   380→      }
   381→   381→    }
   382→   382→    window.addEventListener('keydown', onKey)
   383→   383→    return () => window.removeEventListener('keydown', onKey)
   384→   384→  }, [setView, openSalesCreate, openPurchaseCreate])
   385→   385→
   386→   386→  // Track recent views for the palette
   387→   387→  useEffect(() => {
   388→   388→    pushRecent(view)
   389→   389→  }, [view])
   390→   390→
   391→   391→  // Boot probe: company fetch + session fetch must both settle before deciding.
   392→   392→  if (company.loading || session === undefined) {
   393→   393→    return (
   394→   394→      <div className="flex min-h-screen flex-col gap-4 p-6">
   395→   395→        <Skeleton className="h-14 w-full rounded-lg" />
   396→   396→        <div className="flex flex-1 gap-4">
   397→   397→          <Skeleton className="hidden h-96 w-60 rounded-lg md:block" />
   398→   398→          <Skeleton className="h-96 flex-1 rounded-lg" />
   399→   399→        </div>
   400→   400→      </div>
   401→   401→    )
   402→   402→  }
   403→   403→
   404→   404→  // First boot, no company yet → demo seed or manual setup wizard.
   405→   405→  if (company.data === null) {
   406→   406→    if (manualSetup) {
   407→   407→      return (
   408→   408→        <ManualSetupView
   409→   409→          onDone={() => {
   410→   410→            setManualSetup(false)
   411→   411→            company.reload()
   412→   412→          }}
   413→   413→          onBack={() => setManualSetup(false)}
   414→   414→        />
   415→   415→      )
   416→   416→    }
   417→   417→    return (
   418→   418→      <SetupScreen
   419→   419→        onDone={() => {
   420→   420→          setManualSetup(false)
   421→   421→          company.reload()
   422→   422→        }}
   423→   423→        onManual={() => setManualSetup(true)}
   424→   424→      />
   425→   425→    )
   426→   426→  }
   427→   427→
   428→   428→  // Platform Admin — SEPARATE console (own header + pill nav), opened from the
   429→   429→  // login screen or a SUPER_ADMIN sign-in; never inside an organization's ERP.
   430→   430→  // Exiting while a SUPER_ADMIN session is active signs the admin out — the
   431→   431→  // console is their home, not the org ERP.
   432→   432→  const exitPlatform = async () => {
   433→   433→    const wasSuperAdmin = !!session && session.user.role === 'SUPER_ADMIN'
   434→   434→    setPlatformMode(false)
   435→   435→    if (wasSuperAdmin) {
   436→   436→      try {
   437→   437→        await apiPost('/api/auth/logout', {})
   438→   438→      } catch {
   439→   439→        /* best effort — the console may already have logged out */
   440→   440→      }
   441→   441→      setSession(null)
   442→   442→      return
