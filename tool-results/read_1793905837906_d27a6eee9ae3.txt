    67→    67→export function PlatformView() {
    68→    68→  const [key, setKey] = useState('')
    69→    69→  const [authed, setAuthed] = useState(false)
    70→    70→  const [checking, setChecking] = useState(false)
    71→    71→
    72→    72→  useEffect(() => {
    73→    73→    const saved = sessionStorage.getItem(KEY_STORE)
    74→    74→    if (saved) {
    75→    75→      setKey(saved)
    76→    76→      void verify(saved)
    77→    77→    }
    78→    78→  }, [])
    79→    79→
    80→    80→  const verify = async (k: string) => {
    81→    81→    setChecking(true)
    82→    82→    try {
    83→    83→      await api('/api/platform/auth', { method: 'POST', body: JSON.stringify({ key: k }) })
    84→    84→      sessionStorage.setItem(KEY_STORE, k)
    85→    85→      setAuthed(true)
    86→    86→    } catch (e) {
    87→    87→      sessionStorage.removeItem(KEY_STORE)
    88→    88→      toastError(e)
    89→    89→    } finally {
    90→    90→      setChecking(false)
    91→    91→    }
    92→    92→  }
    93→    93→
    94→    94→  if (!authed) {
    95→    95→    return (
    96→    96→      <div className="mx-auto max-w-md space-y-4 py-10">
    97→    97→        <SectionHeading title="Platform Admin" description="SaaS control plane — plans, organizations, subscriptions, backups." />
    98→    98→        <Card className="p-0">
    99→    99→          <CardContent className="space-y-4 p-6">
   100→   100→            <div className="flex size-12 items-center justify-center rounded-xl bg-primary text-primary-foreground">
   101→   101→              <KeyRound className="size-6" aria-hidden />
   102→   102→            </div>
   103→   103→            <div className="space-y-1.5">
   104→   104→              <Label htmlFor="platform-key">Platform admin key</Label>
   105→   105→              <Input
   106→   106→                id="platform-key"
   107→   107→                type="password"
   108→   108→                value={key}
   109→   109→                onChange={(e) => setKey(e.target.value)}
   110→   110→                onKeyDown={(e) => {
   111→   111→                  if (e.key === 'Enter' && key) void verify(key)
   112→   112→                }}
   113→   113→                placeholder="forge-admin"
   114→   114→                className="h-11"
   115→   115→              />
   116→   116→              <p className="text-xs text-muted-foreground">
   117→   117→                Default is <span className="font-mono">forge-admin</span> — change it with the <span className="font-mono">PLATFORM_ADMIN_KEY</span> environment variable.
   118→   118→              </p>
   119→   119→            </div>
   120→   120→            <Button onClick={() => void verify(key)} disabled={!key || checking} className="h-11 w-full gap-2">
   121→   121→              {checking ? <Loader2 className="size-4 animate-spin" aria-hidden /> : <KeyRound className="size-4" aria-hidden />}
   122→   122→              Unlock console
   123→   123→            </Button>
   124→   124→          </CardContent>
   125→   125→        </Card>
   126→   126→      </div>
   127→   127→    )
   128→   128→  }
   129→   129→
   130→   130→  return <Console key={key} apiKey={key} onAuthLost={() => setAuthed(false)} />
   131→   131→}

   135→   135→function Console({ apiKey, onAuthLost }: { apiKey: string; onAuthLost: () => void }) {
   136→   136→  const [tab, setTab] = useState<'orgs' | 'plans'>('orgs')
   137→   137→  const [data, setData] = useState<OrgsResponse | null>(null)
   138→   138→  const [plans, setPlans] = useState<Plan[] | null>(null)
   139→   139→  const [loading, setLoading] = useState(true)
   140→   140→
   141→   141→  const reload = useCallback(async () => {
   142→   142→    setLoading(true)
   143→   143→    try {
   144→   144→      const [orgs, pl] = await Promise.all([
   145→   145→        platformApi<OrgsResponse>('/api/platform/orgs', undefined, apiKey),
   146→   146→        platformApi<Plan[]>('/api/platform/plans', undefined, apiKey),
   147→   147→      ])
   148→   148→      setData(orgs)
   149→   149→      setPlans(pl)
   150→   150→    } catch (e) {
   151→   151→      toastError(e)
   152→   152→      onAuthLost()
   153→   153→    } finally {
   154→   154→      setLoading(false)
   155→   155→    }
   156→   156→  }, [apiKey, onAuthLost])
   157→   157→
   158→   158→  useEffect(() => {
   159→   159→    void reload()
   160→   160→  }, [reload])
   161→   161→
   162→   162→  const orgAction = async (id: string, body: Record<string, unknown>, okMsg: string) => {
   163→   163→    try {
   164→   164→      await platformApi(`/api/platform/orgs/${id}`, { method: 'PATCH', body: JSON.stringify(body) }, apiKey)
   165→   165→      toast.success(okMsg)
   166→   166→      void reload()
   167→   167→    } catch (e) {
   168→   168→      toastError(e)
   169→   169→    }
   170→   170→  }
   171→   171→
   172→   172→  const stats = data?.stats
   173→   173→
   174→   174→  return (
   175→   175→    <div className="space-y-6">
   176→   176→      <SectionHeading
   177→   177→        title="Platform Admin"
   178→   178→        description="Plans, organizations, subscriptions, backups. The ERP operates the first organization; suspension/expiry blocks its business mutations for real."
   179→   179→        actions={
   180→   180→          <Button variant="outline" onClick={() => void reload()} className="h-11 gap-2">
   181→   181→            <RefreshCw className="size-4" aria-hidden /> Refresh
   182→   182→          </Button>
   183→   183→        }
   184→   184→      />
   185→   185→
   186→   186→      {loading && !data ? (
   187→   187→        <TableSkeleton rows={4} cols={4} />
   188→   188→      ) : stats ? (
   189→   189→        <div className="grid gap-4 sm:grid-cols-2 lg:grid-cols-4">
   190→   190→          <KpiCard title="Organizations" value={String(stats.total)} icon={Building2} hint={`${stats.active} active · ${stats.suspended} suspended`} />
   191→   191→          <KpiCard title="MRR" value={`₹ ${money(stats.mrr)}`} icon={IndianRupee} tone="positive" hint="Active orgs × monthly plan price" />
   192→   192→          <KpiCard title="Documents" value={String(stats.salesInvoices)} icon={DatabaseBackup} hint={`${stats.parties} parties · ${stats.items} items`} />
   193→   193→          <KpiCard title="Warehouses" value={String(stats.warehouses)} icon={Building2} hint="Across the operating org" />
   194→   194→        </div>
   195→   195→      ) : null}
   196→   196→
   197→   197→      <div className="flex flex-wrap gap-2">
   198→   198→        <Button variant={tab === 'orgs' ? 'default' : 'outline'} onClick={() => setTab('orgs')} className="h-11">
   199→   199→          Organizations ({data?.orgs.length ?? 0})
   200→   200→        </Button>
   201→   201→        <Button variant={tab === 'plans' ? 'default' : 'outline'} onClick={() => setTab('plans')} className="h-11">
   202→   202→          Plans ({plans?.length ?? 0})
   203→   203→        </Button>
   204→   204→      </div>
   205→   205→
   206→   206→      {tab === 'orgs' ? <OrgsTable data={data} plans={plans ?? []} apiKey={apiKey} onAction={orgAction} reload={reload} /> : null}
   207→   207→      {tab === 'plans' ? <PlansTable plans={plans ?? []} apiKey={apiKey} reload={reload} /> : null}
   208→   208→
   209→   209→      <BackupCard />
   210→   210→    </div>
   211→   211→  )
   212→   212→}

   240→   240→        '/api/platform/orgs',
