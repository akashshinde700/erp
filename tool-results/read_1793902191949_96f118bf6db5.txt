    91→function InwardsTab() {
    92→  const setView = useErpUi((s) => s.setView)
    93→  const { data, loading } = useApi<JobworkInward[]>('/api/jobwork/inwards')
    94→  const [createOpen, setCreateOpen] = useState(false)
    95→  const inwards = data ?? []
    96→
    97→  const progressOf = (iw: JobworkInward) => {
    98→    const total = iw.items.reduce((s, i) => s + i.qty, 0)
    99→    const done = iw.items.reduce((s, i) => s + i.processedQty, 0)
   100→    return { total, done, pct: total > 0 ? (done / total) * 100 : 0 }
   101→  }
   102→  const cardValue = (iw: JobworkInward) => iw.jobCards.reduce((s, c) => s + c.totalAmount, 0)
   103→
   104→  return (
   105→    <div className="space-y-4">
   106→      <div className="flex items-center justify-between gap-2">
   107→        <p className="text-sm text-muted-foreground">
   108→          {inwards.length} inward{inwards.length === 1 ? '' : 's'} — customer-owned material, value-neutral stock.
   109→        </p>
   110→        <Button onClick={() => setCreateOpen(true)} className="h-11 gap-2">
   111→          <Plus className="size-4" aria-hidden /> New Inward
   112→        </Button>
   113→      </div>
   114→
   115→      <Card className="p-0">
   116→        <CardContent className="p-0">
   117→          {loading ? (
   118→            <TableSkeleton rows={5} cols={6} />
   119→          ) : inwards.length === 0 ? (
   120→            <div className="p-6">
   121→              <EmptyState
   122→                title="No inwards yet"
   123→                hint="Record a customer challan inward to start the jobwork chain."
   124→                icon={ClipboardList}
   125→                action={
   126→                  <Button variant="outline" onClick={() => setCreateOpen(true)} className="mt-2 h-11 gap-2">
   127→                    <Plus className="size-4" aria-hidden /> New Inward
   128→                  </Button>

   132→          ) : (

   134→              <Table>
   135→                <TableHeader className="sticky top-0 z-10 bg-card">
   136→                  <TableRow>
   137→                    <TableHead>Inward No</TableHead>
   138→                    <TableHead>Date</TableHead>
   139→                    <TableHead>Party</TableHead>
   140→                    <TableHead className="text-right">Items</TableHead>
   141→                    <TableHead className="min-w-36">Progress</TableHead>
   142→                    <TableHead className="text-right">Card Value</TableHead>
   143→                    <TableHead>Status</TableHead>
   144→                  </TableRow>
   145→                </TableHeader>

   156→                        </TableCell>
   157→                        <TableCell className="text-right tabular-nums">{iw.items.length}</TableCell>
   158→                        <TableCell>
   159→                          <div className="space-y-1">
   160→                            <ProgressBar value={p.pct} />
   161→                            <p className="text-xs tabular-nums text-muted-foreground">
   162→                              {qtyFmt(p.done)} / {qtyFmt(p.total)} pcs

   197→function NewInwardDialog({
   198→  open,
   199→  onOpenChange,
   200→  onCreated,
   201→}: {
   202→  open: boolean
   203→  onOpenChange: (o: boolean) => void
   204→  onCreated: () => void
   205→}) {
   206→  const { parties, items, warehouses } = useMasters()
   207→  const customers = parties.filter((p) => p.active && partyTypes(p).includes('CUSTOMER'))
   208→
   209→  const [partyId, setPartyId] = useState('')
   210→  const [date, setDate] = useState(todayISO())
   211→  const [challanNo, setChallanNo] = useState('')
   212→  const [warehouseId, setWarehouseId] = useState('')
   213→  const [notes, setNotes] = useState('')
   214→  const [rows, setRows] = useState<InwardRowDraft[]>([{ key: 1, itemId: '', batchNo: '', qty: '', weight: '' }])
   215→  const [saving, setSaving] = useState(false)
   216→
   217→  const seq = rows.length ? Math.max(...rows.map((r) => r.key)) : 0
   218→  const setRow = (key: number, patch: Partial<InwardRowDraft>) =>
   219→    setRows((rs) => rs.map((r) => (r.key === key ? { ...r, ...patch } : r)))
   220→
   221→  const validRows = rows.filter((r) => r.itemId && num(r.qty) > 0)
   222→
   223→  const submit = async () => {
   224→    if (!partyId) {
   225→      toast.error('Select a customer party')
   226→      return
   227→    }
   228→    if (validRows.length === 0) {
   229→      toast.error('Add at least one item row with quantity > 0')
   230→      return
   231→    }
   232→    setSaving(true)
   233→    try {
   234→      const res = await apiPost<JobworkInward>('/api/jobwork/inwards', {
   235→        partyId,
   236→        date,
   237→        challanNo: challanNo.trim() || undefined,
   238→        warehouseId: warehouseId || undefined,
   239→        notes: notes.trim() || undefined,
   240→        items: validRows.map((r) => ({
   241→          itemId: r.itemId,
   242→          batchNo: r.batchNo.trim() || undefined,
   243→          qty: num(r.qty),
   244→          weight: num(r.weight),
   245→        })),
   246→      })
   247→      toast.success(`${res.inwardNo} recorded`, {
   248→        description: `${validRows.length} item line(s) — material received into the jobwork store, status OPEN.`,
   249→      })
   250→      onCreated()
   251→      onOpenChange(false)
   252→      setRows([{ key: seq + 1, itemId: '', batchNo: '', qty: '', weight: '' }])
   253→    } catch (e) {
   254→      toastError(e)
   255→    } finally {
   256→      setSaving(false)
   257→    }
   258→  }
   259→
   260→  return (
   261→    <Dialog open={open} onOpenChange={onOpenChange}>
   262→      <DialogContent className="flex max-h-[92vh] w-full max-w-3xl flex-col overflow-hidden">
   263→        <DialogHeader>
   264→          <DialogTitle>New Jobwork Inward</DialogTitle>
   265→          <DialogDescription>
   266→            Record the customer challan — items are booked qty + weight into the jobwork store (customer-owned,
   267→            value-neutral).
   268→          </DialogDescription>
   269→        </DialogHeader>
   270→
   271→        <div className="min-h-0 flex-1 space-y-4 overflow-y-auto erp-scroll-y pr-1">
   272→          <div className="grid gap-4 sm:grid-cols-2">
   273→            <div className="space-y-1.5">
   274→              <Label>Customer</Label>
   275→              <Select value={partyId || undefined} onValueChange={setPartyId}>
   276→                <SelectTrigger className="h-11 w-full">
   277→                  <SelectValue placeholder="Select customer…" />
   278→                </SelectTrigger>
   279→                <SelectContent>
   280→                  {customers.map((p) => (
   281→                    <SelectItem key={p.id} value={p.id}>
   282→                      {p.code} — {p.name}
   283→                    </SelectItem>
   284→                  ))}
   285→                </SelectContent>
   286→              </Select>
   287→            </div>
   288→            <div className="space-y-1.5">
   289→              <Label htmlFor="inw-date">Date</Label>
   290→              <Input id="inw-date" type="date" value={date} onChange={(e) => setDate(e.target.value)} className="h-11" />
   291→            </div>
   292→            <div className="space-y-1.5">
   293→              <Label htmlFor="inw-challan">Challan No</Label>
   294→              <Input id="inw-challan" value={challanNo} onChange={(e) => setChallanNo(e.target.value)} placeholder="Customer challan ref" className="h-11" />
   295→            </div>

   297→              <Label>Warehouse</Label>
   298→              <Select value={warehouseId || undefined} onValueChange={(v) => setWarehouseId(v === 'none' ? '' : v)}>
   299→                <SelectTrigger className="h-11 w-full">
   300→                  <SelectValue placeholder="Optional — jobwork store" />
   301→                </SelectTrigger>
   302→                <SelectContent>
   303→                  <SelectItem value="none">— skip —</SelectItem>
   304→                  {warehouses.map((w) => (

   338→                                {i.code} — {i.name}

   506→export function JobworkDetailView({ id }: { id: string }) {
   507→  const setView = useErpUi((s) => s.setView)
   508→  const inv = useApi<JobworkInward | null>(`/api/jobwork/inwards/${id}`)
   509→  const { processes } = useMasters()
   510→  const [cardOpen, setCardOpen] = useState(false)
   511→  const [busy, setBusy] = useState<'cert' | 'invoice' | null>(null)
   512→  const [cancelling, setCancelling] = useState(false)
   513→
   514→  if (inv.loading) {
   515→    return <TableSkeleton rows={8} cols={5} />
   516→  }
   517→  if (inv.error || !inv.data) {
   518→    return <EmptyState title="Inward not found" hint={inv.error ?? 'Unknown inward.'} icon={XCircle} />
   519→  }
   520→  const iw = inv.data
   521→  const completedCards = iw.jobCards.filter((c) => c.status === 'COMPLETED')

   920→// ---------- new job card dialog ----------
   921→
   922→interface JcLineState {
   923→  included: boolean
   924→  qty: string
   925→  weight: string
   926→}
   927→
   928→function NewJobCardDialog({
   929→  inward,
   930→  processes,
   931→  open,
   932→  onOpenChange,
   933→  onCreated,
   934→}: {
   935→  inward: JobworkInward
   936→  processes: Process[]
   937→  open: boolean
   938→  onOpenChange: (o: boolean) => void
   939→  onCreated: () => void
   940→}) {
   941→  const [date, setDate] = useState(todayISO())
   942→  const [notes, setNotes] = useState('')
   943→  const [saving, setSaving] = useState(false)
   944→  const [lines, setLines] = useState<Record<string, JcLineState>>({})
   945→  const [selected, setSelected] = useState<Record<string, boolean>>({})
   946→
   947→  // reset when re-opened, from the latest inward state
   948→  useEffect(() => {
   949→    if (!open) return
   950→    const fresh: Record<string, JcLineState> = {}
   951→    for (const it of inward.items) {
   952→      const remQ = Math.max(0, it.qty - it.processedQty)
   953→      const remW = Math.max(0, it.weight - it.processedWeight)
   954→      fresh[it.id] = { included: remQ > 0, qty: String(remQ), weight: String(remW) }
   955→    }
   956→    setLines(fresh)
   957→    setSelected({})
   958→    setDate(todayISO())
   959→    setNotes('')
   960→  }, [open, inward])
   961→
   962→  const activeProcesses = processes.filter((p) => p.active)
   963→  const selectedIds = activeProcesses.filter((p) => selected[p.id]).map((p) => p.id)
   964→  const includedLines = inward.items.filter((it) => lines[it.id]?.included && num(lines[it.id]?.qty) > 0)
   965→
   966→  // live estimate using process default rates (server may use party rates)
   967→  const estimates = selectedIds.flatMap((pid) => {
   968→    const p = activeProcesses.find((x) => x.id === pid)
   969→    if (!p) return []
   970→    const qtySum = includedLines.reduce((s, it) => s + num(lines[it.id]?.qty), 0)
   971→    const weightSum = includedLines.reduce((s, it) => s + num(lines[it.id]?.weight), 0)
   972→    const amount = p.basis === 'WEIGHT' ? weightSum * p.rate : qtySum * p.rate
   973→    return [{ process: p, qtySum, weightSum, amount }]
   974→  })
   975→  const estimateTotal =
   976→    estimates.reduce((s, e) => s + e.amount, 0) +
   977→    selectedIds.reduce((s, pid) => {
   978→      const p = activeProcesses.find((x) => x.id === pid)
   979→      return s + (p?.lotCharge ?? 0)
   980→    }, 0)
   981→
   982→  const setLine = (id: string, patch: Partial<JcLineState>) =>
   983→    setLines((ls) => ({
   984→      ...ls,
   985→      [id]: { ...(ls[id] ?? { included: false, qty: '', weight: '' }), ...patch },
   986→    }))
   987→
   988→  const toggleProcess = (id: string, on: boolean) => setSelected((s) => ({ ...s, [id]: on }))
   989→
   990→  const submit = async () => {
   991→    if (includedLines.length === 0) {
   992→      toast.error('Include at least one item line with qty > 0')
   993→      return
   994→    }
   995→    if (selectedIds.length === 0) {
   996→      toast.error('Select at least one process')
   997→      return
   998→    }
   999→    setSaving(true)
  1000→    try {
  1001→      const res = await apiPost<{ cardNo: string; totalAmount: number }>('/api/jobwork/job-cards', {
  1002→        inwardId: inward.id,
  1003→        date,
  1004→        notes: notes.trim() || undefined,
  1005→        lines: includedLines.map((it) => ({
