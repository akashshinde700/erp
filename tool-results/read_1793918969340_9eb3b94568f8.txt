    90→  const setView = useErpUi((s) => s.setView)

    92→  const path = isSales ? '/api/sales-invoices' : '/api/purchase-invoices'
    93→  const list = useApi<AnyInvoice[]>(path)
    94→  const [search, setSearch] = useState('')
    95→  const [statusFilter, setStatusFilter] = useState<'ALL' | 'POSTED' | 'CANCELLED'>('ALL')
    96→  const [createOpen, setCreateOpen] = useState(false)
    97→  const [presetPartyId, setPresetPartyId] = useState('')
    98→  const [prefill, setPrefill] = useState<AnyInvoice | null>(null)
    99→  const [reloadTick, setReloadTick] = useState(0)
   100→
   101→  // Cross-module create preset (dashboard / palette / party view / Alt+S) —
   102→  // opens the create dialog prefilled with the party.
   103→  useEffect(() => {
   104→    const apply = (p: { type?: string; partyId?: string }) => {
   105→      if (p.type !== (isSales ? 'sales-invoice' : 'purchase-invoice')) return
   106→      setPrefill(null)
   107→      setPresetPartyId(p.partyId ?? '')
   108→      setCreateOpen(true)
   109→    }
   110→    // Preset may have been written a moment before mount (sessionStorage).
   111→    const pending = takeCreatePreset()

   113→    return onPreset(apply)

   115→
   116→  const invoices = useMemo(() => {
   117→    const q = search.trim().toLowerCase()
   118→    return (list.data ?? []).filter((inv) => {
   119→      if (statusFilter !== 'ALL' && inv.status !== statusFilter) return false
   120→      if (!q) return true

   122→        inv.invoiceNo.toLowerCase().includes(q) ||
   123→        (inv.party?.name ?? '').toLowerCase().includes(q) ||

   132→    const paid = posted.reduce((s, i) => s + i.paidAmount, 0)
   133→    return { count: posted.length, grand, paid, outstanding: grand - paid }
   134→  }, [invoices])

   137→    try {
   138→      const full = await apiGet<AnyInvoice>(isSales ? `/api/sales-invoices/${inv.id}` : `/api/purchase-invoices/${inv.id}`)

   140→      setCreateOpen(true)
   141→    } catch (e) {
   142→      toastError(e)
   143→    }
   144→  }
   145→

   148→      ['Invoice No', 'Date', 'Party', 'Taxable', 'CGST', 'SGST', 'IGST', 'Grand Total', 'Paid', 'Status', 'Source'],
   149→      ...invoices.map((i) => [
   150→        i.invoiceNo,
   151→        dateStr(i.date),
   152→        i.party?.name ?? '',
   153→        i.taxable,
   154→        i.cgst,
   155→        i.sgst,
   156→        i.igst,
   157→        i.grandTotal,
   158→        i.paidAmount,
   159→        i.status,
   160→        isSales ? (i as SalesInvoice).source : '',
   161→      ]),
   162→    ]
   163→    downloadCsv(`${isSales ? 'sales' : 'purchase'}-invoices-${todayISO()}.csv`, rows)
   164→  }

   168→      <SectionHeading
   169→        title={isSales ? 'Sales Invoices' : 'Purchase Invoices'}

   171→          isSales

   174→        }
   175→        actions={
   176→          <div className="flex flex-wrap items-center gap-2">
   177→            <Button variant="outline" onClick={() => list.reload()} className="h-11 gap-2">

   180→            <Button variant="outline" onClick={exportCsv} disabled={invoices.length === 0} className="h-11 gap-2">
   181→              <FileDown className="size-4" aria-hidden /> CSV
   182→            </Button>
   183→            <Button onClick={() => { setPrefill(null); setCreateOpen(true) }} className="h-11 gap-2">
   184→              <Plus className="size-4" aria-hidden />
   185→              New {isSales ? 'Invoice' : 'Purchase'}
   186→            </Button>
   187→          </div>
   188→        }
   189→      />
   190→
   191→      <div className="flex flex-wrap items-center gap-2">
   192→        <div className="relative min-w-52 flex-1">
   193→          <Search className="absolute left-2.5 top-1/2 size-4 -translate-y-1/2 text-muted-foreground" aria-hidden />
   194→          <Input
   195→            value={search}
   196→            onChange={(e) => setSearch(e.target.value)}
   197→            placeholder="Search invoice no / party…"
   198→            className="h-11 pl-8"
   199→            aria-label="Search invoices"
   200→          />
   201→        </div>
   202→        <Select value={statusFilter} onValueChange={(v) => setStatusFilter(v as typeof statusFilter)}>
   203→          <SelectTrigger className="h-11 w-36" aria-label="Filter by status">
   204→            <SelectValue />
   205→          </SelectTrigger>
   206→          <SelectContent>
   207→            <SelectItem value="ALL">All statuses</SelectItem>
   208→            <SelectItem value="POSTED">Posted</SelectItem>
   209→            <SelectItem value="CANCELLED">Cancelled</SelectItem>
   210→          </SelectContent>
   211→        </Select>
   212→        <div className="flex flex-wrap items-center gap-2 text-xs text-muted-foreground">
   213→          <Badge variant="secondary" className="tabular-nums">{totals.count} posted</Badge>
   214→          <span className="tabular-nums">Σ ₹ {money(totals.grand)}</span>
   215→          <span className="tabular-nums">outstanding ₹ {money(totals.outstanding)}</span>
   216→        </div>
   217→      </div>
   218→
   219→      <Card className="p-0">
   220→        <CardContent className="p-0">
   221→          {list.loading ? (
   222→            <TableSkeleton rows={6} cols={7} />
   223→          ) : invoices.length === 0 ? (
   224→            <div className="p-6">
   225→              <EmptyState
   226→                title="No invoices found"
   227→                hint={isSales ? 'Post your first sales invoice to see it here.' : 'Record your first purchase invoice.'}
   228→                action={
   229→                  <Button variant="outline" onClick={() => setCreateOpen(true)} className="mt-2 h-11 gap-2">
   230→                    <Plus className="size-4" aria-hidden /> New {isSales ? 'Invoice' : 'Purchase'}
   231→                  </Button>
   232→                }
   233→              />
   234→            </div>
   235→          ) : (
   236→            <div className="max-h-[62dvh] overflow-auto erp-scroll [&_[data-slot=table-container]]:overflow-visible">
   237→              <Table>
   238→                <TableHeader className="sticky top-0 z-10 bg-card">
   239→                  <TableRow>
   240→                    <TableHead>Invoice No</TableHead>
   241→                    <TableHead>Date</TableHead>
   242→                    <TableHead>Party</TableHead>
   243→                    <TableHead className="hidden text-right md:table-cell">Taxable</TableHead>
   244→                    <TableHead className="hidden text-right md:table-cell">GST</TableHead>
   245→                    <TableHead className="text-right">Grand Total</TableHead>
   246→                    <TableHead className="hidden text-right md:table-cell">Paid</TableHead>
   247→                    <TableHead className="hidden text-right md:table-cell">Balance</TableHead>
   248→                    {isSales ? <TableHead className="hidden lg:table-cell">Source</TableHead> : null}

   250→                    <TableHead className="text-right">Actions</TableHead>
   251→                  </TableRow>
   252→                </TableHeader>
   253→                <TableBody>
   254→                  {invoices.map((inv) => (
   255→                    <TableRow
   256→                      key={inv.id}
   257→                      className="cursor-pointer"
   258→                      onClick={() => setView(isSales ? 'sales-detail' : 'purchase-detail', { id: inv.id })}
   259→                    >
   260→                      <TableCell className="font-medium tabular-nums">{inv.invoiceNo}</TableCell>
   261→                      <TableCell className="whitespace-nowrap">{dateStr(inv.date)}</TableCell>
   262→                      <TableCell className="max-w-44 truncate">
   263→                        <span className="block truncate">{inv.party?.name ?? '—'}</span>

   272→                      <TableCell className="text-right font-semibold tabular-nums">
   273→                        <Money value={inv.grandTotal} plain />

   276→                        <Money value={inv.paidAmount} plain />

   279→                        <Money value={inv.grandTotal - inv.paidAmount} plain />
   280→                      </TableCell>
   281→                      {isSales ? (

   283→                          <SourceBadge source={(inv as SalesInvoice).source} />
   284→                        </TableCell>
   285→                      ) : null}
   286→                      <TableCell>
   287→                        <StatusBadge status={inv.status} />
   288→                      </TableCell>
   289→                      <TableCell>
   290→                        <div
   291→                          className="flex items-center justify-end gap-0.5"
   292→                          onClick={(e) => e.stopPropagation()}
   293→                          onKeyDown={(e) => e.stopPropagation()}
   294→                        >
   295→                          <Button
   296→                            variant="ghost"
   297→                            size="icon"
   298→                            className="hidden size-9 text-muted-foreground hover:text-foreground md:inline-flex md:size-8"
   299→                            onClick={() => setView(isSales ? 'sales-detail' : 'purchase-detail', { id: inv.id })}
   300→                            aria-label={`View ${inv.invoiceNo}`}
   301→                            title="View detail"
   302→                          >
   303→                            <Eye className="size-4" aria-hidden />
   304→                          </Button>
   305→                          <Button
   306→                            variant="ghost"
   307→                            size="icon"
   308→                            className="size-9 text-muted-foreground hover:text-foreground md:size-8"
   309→                            onClick={() => void duplicate(inv)}

   314→                            <Copy className="size-4" aria-hidden />
   315→                          </Button>
   316→                          <DropdownMenu>
   317→                            <DropdownMenuTrigger asChild>

   319→                                variant="ghost"
   320→                                size="icon"
   321→                                className="size-9 text-muted-foreground hover:text-foreground md:size-8"
   322→                                disabled={inv.status === 'CANCELLED'}
   323→                                aria-label={`Tally export ${inv.invoiceNo}`}
   324→                                title="Tally export"
   325→                              >
   326→                                <FileDown className="size-4" aria-hidden />

   328→                            </DropdownMenuTrigger>
   329→                            <DropdownMenuContent align="end">
   330→                              <DropdownMenuItem onClick={() => void exportTallyVoucher(isSales ? 'SALES' : 'PURCHASE', inv.id, 'xml')}>
   331→                                <FileCode2 className="size-4" aria-hidden /> Tally XML (import file)
   332→                              </DropdownMenuItem>
   333→                              <DropdownMenuItem onClick={() => void exportTallyVoucher(isSales ? 'SALES' : 'PURCHASE', inv.id, 'json')}>
   334→                                <Braces className="size-4" aria-hidden /> Tally JSON
   335→                              </DropdownMenuItem>
   336→                              <DropdownMenuItem onClick={() => void exportTallyVoucher(isSales ? 'SALES' : 'PURCHASE', inv.id, 'csv')}>
   337→                                <FileSpreadsheet className="size-4" aria-hidden /> Tally Excel (CSV)
   338→                              </DropdownMenuItem>
   339→                              <DropdownMenuSeparator />
   340→                              <DropdownMenuItem onClick={() => downloadUrl('/downloads/tally/Tally-Import.bat')}>
   341→                                <Terminal className="size-4" aria-hidden /> Tally importer (.bat)
   342→                              </DropdownMenuItem>
   343→                              <DropdownMenuItem onClick={() => downloadUrl('/api/tally/masters?parties=1')}>
   344→                                <FileDown className="size-4" aria-hidden /> Tally masters (XML)
   345→                              </DropdownMenuItem>
   346→                              <DropdownMenuSeparator />
   347→                              <DropdownMenuItem onClick={() => void queueForConnector(isSales ? 'SALES' : 'PURCHASE', inv.id)}>
   348→                                Queue for Tally connector
   349→                              </DropdownMenuItem>
   350→                              <DropdownMenuItem onClick={() => void markTallySent(isSales ? 'SALES' : 'PURCHASE', inv.id, true)}>
   351→                                Mark as sent to Tally
   352→                              </DropdownMenuItem>
   353→                              {isSales ? (
   354→                                <>
   355→                                  <DropdownMenuSeparator />
   356→                                  <DropdownMenuItem onClick={() => void exportEInvoiceJson(inv.id)}>
   357→                                    <ScrollText className="size-4" aria-hidden /> e-Invoice JSON (GST portal)
   358→                                  </DropdownMenuItem>
   359→                                </>
   360→                              ) : null}
   361→                            </DropdownMenuContent>
   362→                          </DropdownMenu>
   363→                        </div>
   364→                      </TableCell>
   365→                    </TableRow>
   366→                  ))}
   367→                </TableBody>
   368→              </Table>
   369→            </div>
   370→          )}
   371→        </CardContent>
   372→      </Card>
   373→
   374→      <CreateInvoiceDialog
   375→        kind={kind}
   376→        open={createOpen}
   377→        prefill={prefill}
   378→        presetPartyId={presetPartyId}
   379→        onOpenChange={(o) => {
   380→          setCreateOpen(o)
   381→          if (!o) {
   382→            setPrefill(null)
   383→            setPresetPartyId('')
   384→            setReloadTick((t) => t + 1)
   385→          }
   386→        }}
   387→        onCreated={() => {
   388→          list.reload()
   389→        }}
   390→        key={`create-${reloadTick}`}
   391→      />
   392→    </div>

   400→function CreateInvoiceDialog({
   401→  kind,
   402→  open,
   403→  prefill,
   404→  presetPartyId,
   405→  onOpenChange,
   406→  onCreated,
   407→}: {
   408→  kind: InvoiceKind
   409→  open: boolean
   410→  prefill: AnyInvoice | null
   411→  presetPartyId?: string
   412→  onOpenChange: (o: boolean) => void
   413→  onCreated: () => void
   414→}) {
   415→  const isSales = kind === 'SALES'
   416→  const setView = useErpUi((s) => s.setView)
   417→  const { parties, items, warehouses } = useMasters()
   418→  const company = useCompany()
   419→  const itemMap = useMemo(() => new Map<string, Item>(items.map((i) => [i.id, i])), [items])
   420→
   421→  const eligibleParties = parties.filter(
   422→    (p) => p.active && partyTypes(p).includes(isSales ? 'CUSTOMER' : 'SUPPLIER')
   423→  )
   424→
   425→  const [partyId, setPartyId] = useState('')
   426→  const [date, setDate] = useState(todayISO())
   427→  const [pos, setPos] = useState('')
   428→  const [taxInclusive, setTaxInclusive] = useState(false)
   429→  const [poNo, setPoNo] = useState('')
   430→  const [remarks, setRemarks] = useState('')
   431→  const [warehouseId, setWarehouseId] = useState('')
   432→  const [lines, setLines] = useState<LineDraft[]>([emptyLine()])
   433→  const [saving, setSaving] = useState(false)
   434→
   435→  // Duplicate prefill: populate from an existing invoice (new number is assigned server-side).
   436→  useEffect(() => {
   437→    if (!open || !prefill) return
   438→    setPartyId(prefill.partyId)
   439→    setDate(String(prefill.date).slice(0, 10))
   440→    setPoNo(prefill.poNo ?? '')
   441→    setRemarks(prefill.remarks ?? '')
   442→    setTaxInclusive(prefill.taxInclusive)
   443→    if (isSales) setPos((prefill as SalesInvoice).pos ?? '')
   444→    const items = prefill.items ?? []
   445→    if (items.length > 0) {
   446→      setLines(
   447→        items.map((l) => ({
   448→          key: newKey(),
   449→          itemId: l.itemId,
   450→          detail: l.detail ?? '',
   451→          qty: String(l.qty),
   452→          rate: String(l.rate),
   453→          discPct: String(l.discPct ?? ''),
   454→        }))
   455→      )
   456→    }
   457→  }, [open, prefill, isSales])
   458→
   459→  // Cross-module preset: just the party (e.g. "New invoice for this customer").
   460→  useEffect(() => {
   461→    if (!open) return
   462→    if (presetPartyId) {
   463→      setPartyId(presetPartyId)
   464→      const p = parties.find((x) => x.id === presetPartyId)
   465→      if (isSales && p) setPos(p.stateCode)
   466→    }
   467→  }, [open, presetPartyId, parties, isSales])
   468→
   469→  const party: Party | undefined = parties.find((p) => p.id === partyId)
   470→  const companyState = company?.stateCode ?? '27'
   471→
   472→  const onPartyChange = (id: string) => {
   473→    setPartyId(id)
   474→    const p = parties.find((x) => x.id === id)
   475→    if (isSales && p) setPos(p.stateCode) // PoS defaults to the party's state
   476→  }
   477→
   478→  const interState = isSales
   479→    ? pos !== '' && pos !== companyState
   480→    : !!party && party.stateCode !== companyState
   481→
   482→  const validLines = lines.filter((l) => l.itemId && num(l.qty) > 0)
   483→
   484→  const taxLines = validLines.map((l) => ({
   485→    qty: num(l.qty),
   486→    rate: num(l.rate),
   487→    discPct: num(l.discPct) || 0,
   488→    gstRate: itemMap.get(l.itemId)?.gstRate ?? 0,
   489→  }))
   490→
   491→  // Live preview — same pure engine as the server (cheap, no memo needed)
   492→  const totals = computeInvoiceTotals(taxLines, { taxInclusive, interState })
   493→
   494→  const setLine = (key: string, patch: Partial<LineDraft>) =>
   495→    setLines((ls) => ls.map((l) => (l.key === key ? { ...l, ...patch } : l)))
   496→
   497→  const onItemPick = (key: string, itemId: string) => {
   498→    const it = itemMap.get(itemId)
   499→    setLine(key, {
   500→      itemId,
   501→      rate: it ? String(isSales ? it.saleRate : it.costRate) : '',
   502→    })
   503→  }
   504→
   505→  const linePreview = (l: LineDraft): { taxable: number; total: number } => {
   506→    if (!l.itemId || num(l.qty) <= 0) return { taxable: 0, total: 0 }
   507→    const t = computeInvoiceTotals(
   508→      [{ qty: num(l.qty), rate: num(l.rate), discPct: num(l.discPct) || 0, gstRate: itemMap.get(l.itemId)?.gstRate ?? 0 }],
   509→      { taxInclusive, interState }
   510→    )
   511→    return { taxable: t.lines[0].taxable, total: t.lines[0].total }
   512→  }
   513→
   514→  const submit = async () => {

   971→export function InvoiceDetail({ kind, id }: { kind: InvoiceKind; id: string }) {
   972→  const setView = useErpUi((s) => s.setView)
   973→  const isSales = kind === 'SALES'

   975→  const { items } = useMasters()

   989→  }
   990→  const invoice = inv.data
   991→  const cancelled = invoice.status === 'CANCELLED'
   992→
   993→  const cancel = async () => {
   994→    setCancelling(true)

   996→      await apiDelete(isSales ? `/api/sales-invoices/${id}` : `/api/purchase-invoices/${id}`)

  1000→      inv.reload()
  1001→    } catch (e) {
  1002→      toastError(e)
  1003→    } finally {
  1004→      setCancelling(false)
  1005→    }
  1006→  }
  1007→
  1008→  // One button, the whole next step: open the payment dialog on the Accounting
  1009→  // module, prefilled with this party and the balance allocated to THIS invoice.
  1010→  const payNow = () => {
  1011→    const balance = Math.max(0, r2(invoice.grandTotal - invoice.paidAmount))
  1012→    if (balance <= 0) {
  1013→      toast.info('This invoice is fully paid')
  1014→      return
