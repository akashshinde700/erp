   274→        <p className="max-w-md text-sm text-muted-foreground">
   275→          No company is configured yet. Load the seeded demo data (company &ldquo;Shree Forge Industries&rdquo;
   276→          with parties, items, processes, invoices and a full jobwork chain), or start with a blank company.

   288→    </div>
   289→  )

   294→export function ErpApp() {
   295→  const { view, params, setView } = useErpUi()
   296→
   297→  // ---------- organization login (SaaS) ----------
   298→  // undefined = session probe in flight · null = signed out · info = signed in.
   299→  // Platform Admin is a SEPARATE console (platformMode) — never inside the ERP.
   300→  // A SUPER_ADMIN session opens the console directly (like the reference SaaS).
   301→  const [session, setSession] = useState<SessionInfo | null | undefined>(undefined)
   302→  const [platformMode, setPlatformMode] = useState(false)
   303→
   304→  // Boot routing uses the PUBLIC status probe (no sensitive data): it tells
   305→  // first-boot (no company) apart from signed-out without ever touching the
   306→  // company profile. The full profile (bank details included) is fetched ONLY
   307→  // for a signed-in operating-org session — never anonymously.
   308→  const status = useApi<AuthStatus | null>('/api/auth/status')
   309→  const company = useApi<Company | null>(session === undefined ? null : session ? '/api/masters/company' : null)
   310→  const [manualSetup, setManualSetup] = useState(false)
   311→  const [drawerOpen, setDrawerOpen] = useState(false)
   312→  const mounted = useMounted()
   313→
   314→  useEffect(() => {
   315→    let cancelled = false
   316→    api<SessionInfo | null>('/api/auth/session')
   317→      .then((s) => {
   318→        if (!cancelled) setSession(s)
   319→      })
   320→      .catch(() => {
   321→        if (!cancelled) setSession(null)
   322→      })
   323→    return () => {
   324→      cancelled = true
   325→    }
   326→  }, [])
   327→
   328→  const logout = async () => {
   329→    try {
   330→      await apiPost('/api/auth/logout', {})
   331→      toast.success('Signed out')
   332→    } catch {
   333→      toast.error('Could not sign out — try again')
   334→    }
   335→    setSession(null)
   336→    setView('dashboard')
   337→  }
   338→
   339→  useEffect(() => {
   340→    window.scrollTo({ top: 0 })
   341→  }, [view])
   342→
   343→  // ---------- global keyboard: palette (Ctrl/⌘+K, /), g+key goto, ? help, Alt+S/Alt+P ----------
   344→  const [paletteOpen, setPaletteOpen] = useState(false)
   345→  const [helpOpen, setHelpOpen] = useState(false)
   346→  const gotoArmed = useRef(0)
   347→
   348→  const openSalesCreate = useCallback(() => {
   349→    setCreatePreset({ type: 'sales-invoice' })
   350→    setView('sales')
   351→  }, [setView])
   352→  const openPurchaseCreate = useCallback(() => {
   353→    setCreatePreset({ type: 'purchase-invoice' })
   354→    setView('purchase')
   355→  }, [setView])
   356→
   357→  useEffect(() => {
   358→    const onKey = (e: KeyboardEvent) => {
   359→      // Ctrl/⌘ + K → palette (always)
   360→      if ((e.ctrlKey || e.metaKey) && e.key.toLowerCase() === 'k') {
   361→        e.preventDefault()
   362→        setPaletteOpen(true)
   363→        return
   364→      }
   365→      if (isTypingTarget(e)) return
   366→      // Alt+S / Alt+P → new documents from anywhere
   367→      if (e.altKey && e.key.toLowerCase() === 's') {
   368→        e.preventDefault()
   369→        openSalesCreate()
   370→        return
   371→      }
   372→      if (e.altKey && e.key.toLowerCase() === 'p') {
   373→        e.preventDefault()
   374→        openPurchaseCreate()
   375→        return
   376→      }
   377→      // "?" → shortcuts help
   378→      if (e.key === '?') {
   379→        e.preventDefault()
   380→        setHelpOpen(true)
   381→        return
   382→      }
   383→      // "/" → palette
   384→      if (e.key === '/') {
   385→        e.preventDefault()
   386→        setPaletteOpen(true)
   387→        return
   388→      }
   389→      // g + key → module (Gmail style, 1.5 s window)
   390→      if (e.key.toLowerCase() === 'g' && !e.ctrlKey && !e.metaKey && !e.altKey) {
   391→        gotoArmed.current = Date.now()
   392→        return
   393→      }
   394→      const goto = GOTO_KEYS[e.key.toLowerCase()]
   395→      if (goto && Date.now() - gotoArmed.current < 1500 && !e.ctrlKey && !e.metaKey && !e.altKey) {
   396→        e.preventDefault()
   397→        gotoArmed.current = 0
   398→        setView(goto.view)
   399→      }
   400→    }
   401→    window.addEventListener('keydown', onKey)
   402→    return () => window.removeEventListener('keydown', onKey)
   403→  }, [setView, openSalesCreate, openPurchaseCreate])
   404→
   405→  // Track recent views for the palette
   406→  useEffect(() => {
   407→    pushRecent(view)
   408→  }, [view])
   409→
   410→  // Boot probe: session + status must both settle before deciding. The
   411→  // company profile loads lazily behind the session.
   412→  if (session === undefined || (status.data === null && !status.error)) {
   413→    return (
   414→      <div className="flex min-h-screen flex-col gap-4 p-6">
   415→        <Skeleton className="h-14 w-full rounded-lg" />
   416→        <div className="flex flex-1 gap-4">
   417→          <Skeleton className="hidden h-96 w-60 rounded-lg md:block" />
   418→          <Skeleton className="h-96 flex-1 rounded-lg" />
   419→        </div>
   420→      </div>
   421→    )
   422→  }
   423→
   424→  // Status endpoint unreachable → honest offline/error screen (never a
   425→  // wrong setup wizard).
   426→  if (status.error && status.data === null) {
   427→    return (
   428→      <div className="flex min-h-screen flex-col items-center justify-center gap-4 p-6 text-center">
   429→        <ShieldAlert className="size-10 text-destructive" aria-hidden />
   430→        <h1 className="text-lg font-semibold">Cannot reach the server</h1>
   431→        <p className="max-w-md text-sm text-muted-foreground">{status.error}</p>
   432→      </div>
   433→    )
   434→  }
   435→
   436→  // First boot, no company yet → demo seed or manual setup wizard.
   437→  if (status.data && !status.data.companyExists) {
   438→    if (manualSetup) {
   439→      return (
   440→        <ManualSetupView
   441→          onDone={() => {
   442→            setManualSetup(false)
   443→            status.reload()
   444→          }}
   445→          onBack={() => setManualSetup(false)}
   446→        />
   447→      )
   448→    }
   449→    return (
   450→      <SetupScreen
   451→        onDone={() => {
   452→          setManualSetup(false)
   453→          status.reload()
   454→        }}
   455→        onManual={() => setManualSetup(true)}
   456→      />
   457→    )
   458→  }
   459→
   460→  // Platform Admin — SEPARATE console (own header + pill nav), opened from the
   461→  // login screen or a SUPER_ADMIN sign-in; never inside an organization's ERP.
   462→  // Exiting while a SUPER_ADMIN session is active signs the admin out — the
   463→  // console is their home, not the org ERP.
   464→  const exitPlatform = async () => {
   465→    const wasSuperAdmin = !!session && session.user.role === 'SUPER_ADMIN'
   466→    setPlatformMode(false)
   467→    if (wasSuperAdmin) {
   468→      try {
   469→        await apiPost('/api/auth/logout', {})
   470→      } catch {
   471→        /* best effort — the console may already have logged out */
   472→      }
   473→      setSession(null)
   474→      return
   475→    }
   476→    // Key-mode exit: stay honest about the cookie the console may have swapped.
   477→    api<SessionInfo | null>('/api/auth/session')
   478→      .then((s) => setSession(s))
   479→      .catch(() => setSession(null))
   480→  }
   481→
   482→  const platformConsole = platformMode || (!!session && session.user.role === 'SUPER_ADMIN')
   483→
   484→  if (platformConsole) {
   485→    return (
   486→      <Suspense
   487→        fallback={
   488→          <div className="flex min-h-dvh items-center justify-center p-6">
   489→            <Skeleton className="h-96 w-full max-w-6xl rounded-xl" />
   490→          </div>
   491→        }
   492→      >
   493→        <PlatformView
   494→          onExit={() => void exitPlatform()}
   495→          onLoginAs={(s) => {
   496→            // Impersonation handover: the console set the session cookie for the
   497→            // target org's owner — hand the browser to their workspace.
   498→            setSession(s)
   499→            setPlatformMode(false)
   500→          }}
   501→        />
   502→      </Suspense>
   503→    )
   504→  }
   505→
   506→  // Not signed in → organization login screen (SaaS entry).
   507→  if (!session) {
   508→    return (
   509→      <LoginView
   510→        onLogin={(s) => setSession(s)}
   511→        onOpenPlatform={() => setPlatformMode(true)}
   512→        onPlatformLogin={(s) => {
   513→          setSession(s)
   514→          setPlatformMode(true)
   515→        }}
   516→      />
   517→    )
   518→  }
   519→
   520→  // Signed in to an org that is NOT the operating org → honest org portal
   521→  // (real profile/subscription data, never another org's books).
   522→  if (!session.operating) {
   523→    return <OrgPortal session={session} onLogout={() => void logout()} />
   524→  }
   525→
   526→  const fy = mounted ? fyCode(new Date()) : '····'
   527→  const today = mounted ? dateStr(new Date()) : ''
   528→
